function createChannel() {
  let senderResolve;
  let receiverResolve;
  
  return {
    send: async function(message) {
      if (receiverResolve) {
        const resolve = receiverResolve;
        receiverResolve = null;
        resolve(message);
        return;
      } else {
        return new Promise(resolve => { senderResolve = resolve; });
      }
    },
    receive: async function() {
      if (senderResolve) {
        const resolve = senderResolve;
        senderResolve = null;
        return resolve();
      } else {
        return new Promise(resolve => { receiverResolve = resolve; });
      }
    }
  };
}

async function* genA(channel) {
  const data = [1, 2, 3, 4, 5];

}
