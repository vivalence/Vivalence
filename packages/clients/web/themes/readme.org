# Theme System Documentation

This document outlines a systematic approach to managing themes using CSS variables with Tailwind CSS, while maintaining a structured JavaScript color system.

## Overview
The system consists of:
1. JavaScript color definitions (palette, tints, semantic colors)
2. Theme definitions using these colors
3. CSS variable generation
4. Tailwind configuration
5. Theme switching mechanism

## 1. Color System Structure

### Base Palette
```javascript
const palette = {
  white: "#F2F4F6",
  black: "#030D0D",
  gray: {

    10: "#F3F5F7",
    // ... scales
  },
  aqua: {
    50: "#adfffa",
    100: "#80ede6",
    // ... scales
  }
  // ... other colors
}
```

## 2. Theme Definitions

Themes are defined using the color system structure:

```javascript
const themes = {
  carbon: {
    colors: {
      theme: {
        background: palette.gray[900],
        text: {
          1: palette.gray[10],
          2: palette.gray[60]
        },
        accent: palette.pink[400]
      },
      system: {
        info: palette.indigo[400],
        // ... system colors
      }
    }
  },
  light: {
    colors: {
      theme: {
        background: palette.gray[10],
        text: {
          1: palette.gray[900],
          2: palette.gray[600]
        },
        accent: palette.aqua[400]
      }
      // ... other colors
    }
  }
  // ... other themes
}
```

## 3. CSS Variable Generation

Convert theme objects to CSS variables using a flattening function:

```javascript
function flattenTheme(obj, prefix = '') {
  return Object.entries(obj).reduce((acc, [key, value]) => {
    const newPrefix = prefix ? `${prefix}-${key}` : key

    if (typeof value === 'string') {
      acc[`--${newPrefix}`] = value
    } else if (typeof value === 'object') {
      Object.assign(acc, flattenTheme(value, newPrefix))
    }
    return acc
  }, {})
}

function generateThemeCSS() {
  let css = ''
  
  Object.entries(themes).forEach(([themeName, theme]) => {
    const variables = flattenTheme(theme)
    
    css += `[data-theme='${themeName}'] {\n`
    Object.entries(variables).forEach(([variable, value]) => {
      css += `  ${variable}: ${value};\n`
    })
    css += `}\n\n`
  })

  return css
}
```

This generates CSS like:
```css
[data-theme='carbon'] {
  --colors-theme-background: #030D0D;
  --colors-theme-text-1: #F2F4F6;
  --colors-theme-text-2: #BEC8D2;
  --colors-theme-accent: #FF00AA;
  --colors-system-info: #00006a;
}

[data-theme='light'] {
  --colors-theme-background: #F3F5F7;
  --colors-theme-text-1: #020508;
  --colors-theme-text-2: #354556;
  --colors-theme-accent: #00FFFB;
}
```

## 4. Tailwind Configuration

Configure Tailwind to use the CSS variables:

```javascript
// tailwind.config.js
module.exports = {
  theme: {
    colors: {
      background: 'var(--colors-theme-background)',
      text: {
        1: 'var(--colors-theme-text-1)',
        2: 'var(--colors-theme-text-2)'
      },
      accent: 'var(--colors-theme-accent)',
      // ... other color mappings
    }
  }
}
```

## 5. Theme Switching

Simple theme switching mechanism:

```javascript
// theme.js
export function setTheme(themeName) {
  document.documentElement.setAttribute('data-theme', themeName)
  localStorage.setItem('theme', themeName)
}

// Initialize theme
const savedTheme = localStorage.getItem('theme') || 'carbon'
setTheme(savedTheme)
```

## Usage

### In Components
```html
<div class="bg-background text-text-1">
  <h1 class="text-text-2">Title</h1>
  <button class="bg-accent">
    Action
  </button>
</div>
```

### Theme Switcher Component
```javascript
function ThemeSelector() {
  return (
    <select 
      value={currentTheme} 
      onChange={(e) => setTheme(e.target.value)}
    >
      <option value="carbon">Carbon</option>
      <option value="light">Light</option>
    </select>
  )
}
```

## Build Process

1. Create a build script that runs the CSS generation:
```javascript
// build-themes.js
const fs = require('fs')
const css = generateThemeCSS()
fs.writeFileSync('src/themes.css', css)
```

2. Add to your build process:
```json
{
  "scripts": {
    "build:themes": "node build-themes.js",
    "build": "npm run build:themes && npm run build:app"
  }
}
```

export const createStackLevel = (namespace = '', baseLevel = 0) => {
  let currentLevel = baseLevel;
  
  return {
    next: (label = '') => {
      currentLevel += 1;
      return {
        level: currentLevel,
        name: label ? `${namespace}-${label}` : `${namespace}-${currentLevel}`,
        zIndex: currentLevel
      };
    },
    reset: () => {
      currentLevel = baseLevel;
    },
    current: () => currentLevel
  };
};

export const createNamespacedStack = (namespace, config = {}) => {
  const {
    baseLevel = 0,
    increment = 1,
    max = 100
  } = config;
  
  return {
    baseLevel,
    increment,
    max,
    calculate: (level) => Math.min(baseLevel + (level * increment), max),
    namespace
  };
};

export const createMediaQuery = (breakpoint, type = 'min') => {
  const bpValue = breakpoints.values[breakpoint];
  return `@media (${type}-width: ${bpValue}${breakpoints.unit})`;
};

export const createMediaQuerySet = (queries) => {
  return Object.entries(queries).reduce((acc, [key, value]) => {
    acc[key] = `@media ${value}`;
    return acc;

export class TokenRuntime {
  private cache: Map<string, any>;
  private computedTokens: Map<string, () => any>;

  constructor() {
    this.cache = new Map();
    this.computedTokens = new Map();
  }

  registerComputed(name: string, compute: () => any) {
    this.computedTokens.set(name, compute);
  }

  resolve(tokenPath: string) {
    if (this.cache.has(tokenPath)) {
      return this.cache.get(tokenPath);
    }

    // Resolve and cache token value
    const value = this.computeTokenValue(tokenPath);
    this.cache.set(tokenPath, value);
    return value;
  }
}

class TokenInspector {
private observer: MutationObserver;

initialize() {
// Setup dev tools panel
this.injectDevTools();
this.setupElementInspection();
}

private setupElementInspection() {
this.observer = new MutationObserver((mutations) => {
for (const mutation of mutations) {
this.analyzeTokenUsage(mutation.target as Element);
}
});
}

private analyzeTokenUsage(element: Element) {
// Extract and display token usage
const usedTokens = this.extractTokens(element);
this.updateDevToolsPanel(usedTokens);
}
}
