export class ProcessManager {
  constructor(gaia) {
    this.gaia = gaia;
    this.processes = new Map();
    this.watchdog = null;
  }

  async start() {
    // Start watchdog
    this.watchdog = setInterval(() => this.healthCheck(), 5000);
    
    // Register signal handlers
    Deno.addSignalListener("SIGTERM", () => this.gracefulShutdown());
    Deno.addSignalListener("SIGHUP", () => this.reload());
    
    // Auto-start configured processes
    for (const die of this.gaia.terrans) {
      if (die.cake.autostart) {
        await this.spawn(die);
      }
    }
  }

  async spawn(die) {
    const process = new Deno.Command(die.cake.cmd[0], {
      args: die.cake.cmd.slice(1),
      env: { ...die.cake.env, VIVA_PARENT_PID: Deno.pid },
      stdout: "piped",

      this.processes.delete(die.slug);
    });
