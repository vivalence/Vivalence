// pattern-recognition/provision.js
export default async function provision(inputs, ctx) {
  const { tactic, scope } = inputs;
  const { games, tags } = tactic.relations;
  let blacklist = inputs.blacklist;

  // 1. Get pattern examples (e.g., stem-changing verbs)
  const patternUnits = await ctx.runtime.call("/units/fromTagIds", {
    tagIds: [
      tags.pattern[0].id,  // e.g., stem:e-ie
      ...tags.examples.map(t => t.id)
    ],
    blacklist,
    take: 3
  });

  // 2. Get contrastive examples (similar but different pattern)
  const exceptionUnits = await ctx.runtime.call("/units/fromTagIds", {
    tagIds: [
      tags.exceptions[0].id,  // e.g., stem:e-i
      ...tags.examples.map(t => t.id)
    ],
    take: 2
  });

  // 3. Generate pattern recognition exercises
  const comparisonExercises = await games.comparison.call("/provision", {
    pattern: patternUnits,
    exceptions: exceptionUnits,
    prompt: "Identify the pattern change in these verbs"
  });

  // 4. Practice correct forms in context
  const translations = await Promise.all(
    patternUnits.map(unit => {
      const constraints = [
        `Use verb: ${unit.data.learning}`,
        `Must trigger stem change`,
        `Keep context simple`
      ];
      return games.translations.call("/provision", { constraints });
    })
  );

  // 5. Reinforce with flashcards
  const flashcards = await games.flashcards.call("/provision/fromUnits", {
    units: [...patternUnits, ...exceptionUnits]
  });

  return [comparisonExercises, ...translations, ...flashcards];
}

// transformation/provision.js
export default async function provision(inputs, ctx) {
  const { tactic, scope } = inputs;
  const { games, tags } = tactic.relations;
  let blacklist = inputs.blacklist;

  // 1. Get base forms to transform
  const sourceUnits = await ctx.runtime.call("/units/fromTagIds", {
    tagIds: [
      tags.source[0].id,  // e.g., tense:pres
      ...tags.context.map(t => t.id)
    ],
    blacklist,
    take: 3
  });

  // 2. Get corresponding target forms
  const targetUnits = await Promise.all(
    sourceUnits.map(unit => 
      ctx.runtime.call("/units/fromTagIds", {
        tagIds: [
          tags.target[0].id,  // e.g., tense:past
          unit.lemma.id
        ],
        take: 1
      })
    )
  );

  // 3. Create transformation exercises
  const transformations = await games.transformation.call("/provision", {
    pairs: sourceUnits.map((source, i) => ({
      source,
      target: targetUnits[i],
      context: scope
    }))
  });

  // 4. Practice in context
  const translations = await Promise.all(
    targetUnits.map(unit => {
      const constraints = [
        `Use form: ${unit.data.learning}`,
        `Highlight temporal context change`,
        `Keep vocabulary simple`
      ];
      return games.translations.call("/provision", { constraints });
    })
  );

  // 5. Reinforce difficult transformations
  const flashcards = await games.flashcards.call("/provision/fromUnits", {
    units: targetUnits.filter(u => u.memory?.status === "UNKNOWN")
  });

  return [transformations, ...translations, ...flashcards];
}
