export default class Aperture {
  path: Path;
  descendants: Aperture[] = [];
  composed: function;
  router: Router;
  middleware: {
    native: Middleware[];
    pre: Middleware[];
    post: Middleware[];
  };

  constructor(path: Path) {
    this.path = path;
    this.router = new Router();
    this.middleware = {
      native: [],
      pre: [],
      post: [],
    };
  }

  use(middleware: Middleware): Aperture {
    this.middleware.native.push(middleware);
    return this;
  }

  pre(middleware: Middleware): Aperture {
    this.middleware.pre.push(middleware);
    return this;
  }

  post(middleware: Middleware): Aperture {
    this.middleware.post.push(middleware);
    return this;
  }

  open(path: string, handler: Handler): Aperture {
    const routePath = new Path(path);

    this.router.all(routePath.toString(), async (ctx: ApertureContext) => {
      ctx.response.body = await handler(await parser(ctx), ctx);
    });

    return this;
  }

  branch(path: string): Aperture {
    const aperture = new Aperture(new Path(path));
    this.descendants.push(aperture);
    return aperture;
  }

  compose(force = false) {
    if (force) this.composed = null;

    if (!this.composed) {
      this.router.use(async (ctx, next) => {
        for (const middleware of this.middleware.pre) {
          await new Promise<void>((resolve) => {
            middleware(ctx, () => {
              resolve();
              return Promise.resolve();
            });
          });
        }
        await next();
      });

      for (const middleware of this.middleware.native) {
        this.router.use(middleware);
      }

      this.router.use(async (ctx, next) => {
        await next();
        for (const middleware of this.middleware.post) {
          await new Promise<void>((resolve) => {
            middleware(ctx, () => {
              resolve();
              return Promise.resolve();
            });
          });
        }
      });

      for (const descendant of this.descendants) {
        descendant.serve(this.router);
      }

      this.composed = compose([this.router.routes(), this.router.allowedMethods()]);
    }

    return this.composed;
  }
  serve(router: Router) {
    this.compose();
    router.use(this.path.toString(), this.router.routes(), this.router.allowedMethods());
    return this;
  }
}

export default class Trajectory {
  constructor(path = "/") {
    this.path = path;
    this.handlers = new Map();
    this.branches = new Map();
    this.middlewares = [];
  }

  open(path, handler) {
    this.handlers.set(path, handler);
    return this;
  }

  branch(path) {

    if (!this.branches.has(path)) {
      this.branches.set(path, new Trajectory(path));
    }
    return this.branches.get(path);
  }

  use(middleware) {
    this.middlewares.push(middleware);
    return this;
  }

  getAvailablePaths() {
    return [...Array.from(this.handlers.keys()), ...Array.from(this.branches.keys())];
  }

  async traverse(path, ctx) {
    if (this.handlers.has(path)) {
      await this._applyMiddlewares(ctx);

      const result = await this.handlers.get(path)(ctx);

      return {
        handled: true,
        result,
        paths: this.getAvailablePaths(),
        currentPath: this.path,
      };
    }

    for (const [branchPath, branch] of this.branches) {
      if (path === branchPath) {
        await this._applyMiddlewares(ctx);

        return {
          handled: false,
          paths: branch.getAvailablePaths().map((p) => `${branchPath}${p === "/" ? "" : "/" + p}`),
          currentPath: branchPath,
        };
      }

      if (path.startsWith(branchPath + "/")) {
        await this._applyMiddlewares(ctx);

        const subPath = path.substring(branchPath.length) || "/";
        const branchResult = await branch.traverse(subPath, ctx);

        if (branchResult.paths) {
          branchResult.paths = branchResult.paths.map((p) =>
            p.startsWith("/") ? `${branchPath}${p}` : `${branchPath}/${p}`,
          );
        }

        return branchResult;
      }
    }

    return {
      handled: false,
      paths: this.getAvailablePaths(),
      currentPath: this.path,
    };
  }

  async _applyMiddlewares(ctx) {
    let index = 0;

    const next = async () => {
      if (index < this.middlewares.length) {
        const middleware = this.middlewares[index++];
        return await middleware(ctx, next);
      }
    };

    await next();
  }
}

Deep Dive: How Routers Match Paths & HTTP Request Flow

The Core Router Path Matching Algorithm
At the heart of every web framework is the router's path matching algorithm. Here's how it works in detail:
Trie-Based Routing for Efficiency
Most modern routers use a trie (prefix tree) data structure for efficient path matching:
javascriptCopyclass Router {
  constructor() {
    // The root of our trie
    this.tree = {
      children: new Map(),
      handlers: new Map() // HTTP method -> handler
    };
  }
  
  // Adding a route to the trie
  add(method, path, handler) {
    const pathSegments = path.split('/').filter(Boolean);
    let node = this.tree;
    
    // Build the trie structure
    for (const segment of pathSegments) {
      // Create node for this segment if it doesn't exist
      if (!node.children.has(segment)) {
        node.children.set(segment, {
          paramName: segment.startsWith(':') ? segment.slice(1) : null,
          isWildcard: segment === '*',
          children: new Map(),
          handlers: new Map()
        });
      }
      
      node = node.children.get(segment);
    }
    
    // Store the handler for this path and method
    node.handlers.set(method, handler);
  }
}
Path Parameter Extraction
When matching, parameters are carefully extracted:
javascriptCopymatch(method, path) {
  const segments = path.split('/').filter(Boolean);
  const params = {};
  let matchFound = false;
  let matchedHandler = null;
  
  // Recursive function to walk the trie
  const walk = (node, index) => {
    // If we've processed all segments, check if this node has a handler
    if (index >= segments.length) {
      if (node.handlers.has(method)) {
        matchFound = true;
        matchedHandler = node.handlers.get(method);
        return true;
      }
      return false;
    }
    
    const currentSegment = segments[index];
    
    // Try exact match first (highest priority)
    if (node.children.has(currentSegment)) {
      const exactMatch = walk(node.children.get(currentSegment), index + 1);
      if (exactMatch) return true;
    }
    
    // Try parameter matches (medium priority)
    for (const [segment, childNode] of node.children.entries()) {
      if (childNode.paramName) {
        // This is a parameter node (e.g., :id)
        params[childNode.paramName] = currentSegment;
        const paramMatch = walk(childNode, index + 1);
        if (paramMatch) return true;
        // If no match, remove the param since it's not the right path
        delete params[childNode.paramName];
      }
    }
    
    // Try wildcard match (lowest priority)
    for (const [segment, childNode] of node.children.entries()) {
      if (childNode.isWildcard) {
        // Capture remaining path
        params['*'] = segments.slice(index).join('/');
        matchFound = true;
        matchedHandler = childNode.handlers.get(method);
        return true;
      }
    }
    
    return false;
  };
  
  walk(this.tree, 0);
  
  return matchFound ? { handler: matchedHandler, params } : null;
}
Path Matching Priority Order
The router follows a specific priority order:

Exact matches - /users/profile matches exactly before parameters
Parameter matches - /users/:id extracts the id parameter
Wildcard matches - /files/* catches any path under /files
Optional parameters - /docs/:section? makes section optional

2. Middleware Composition

javascriptCopy// Inside Application.handleRequest()

  const composedMiddleware = compose(this.middleware);

// The actual compose function
function compose(middleware) {
  return function(context, next) {
    let index = -1;
    
    async function dispatch(i) {
      // Ensure next() isn't called multiple times at same level
      if (i <= index) throw new Error('next() called multiple times');
      
      index = i;
      let fn = middleware[i];
      
      // If we've run out of middleware, use provided next() if any
      if (i === middleware.length) fn = next;
      
      // End of middleware chain
      if (!fn) return;
      
      // Execute current middleware with next in chain
      return await fn(context, dispatch.bind(null, i + 1));
    }
    
    return dispatch(0);
  };
}
This composition creates a nested promise chain where each middleware either:

Calls next() to continue to the next middleware
Returns without calling next() to stop the chain
Throws an exception to trigger error handling
