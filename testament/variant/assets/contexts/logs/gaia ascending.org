// systems/gaia/prototypes/process.js
import { Status, Connection } from "@vivalence/typology";
import { Vector } from "@vivalence/vector";

export class Process {
  constructor(cake, gaia) {
    this.cake = cake;           // ProcessConfiguration
    this.gaia = gaia;           // Gaia instance
    this.runner = null;         // Execution context
    this.blocked = false;       // Manual override
    this.restarts = 0;          // Restart counter
    
    // Reactive state management
    this.status = Status("created");
    this.connection = new Connection();
    this.events = new Vector();
    
    // Process metadata
    this.pid = null;
    this.started = null;
    this.exited = null;
    this.reason = null;         // Last action reason
  }

  // State transition guards
  canStart() {
    if (this.blocked) return { can: false, reason: "blocked" };
    if (this.status.get().code === "running" && !this.cake.overrun) {
      return { can: false, reason: "already_running" };
    }
    if (this.restarts >= (this.cake.restartLimit ?? Infinity)) {
      return { can: false, reason: "exhausted" };
    }
    return { can: true };
  }

  canStop() {
    const status = this.status.get().code;
    return status === "running" || status === "starting";
  }

  // Core lifecycle methods
  async start(reason = "manual", restart = false) {
    const guard = this.canStart();
    if (!guard.can) {
      this.gaia.logger.warn("process", `Cannot start ${this.cake.id}: ${guard.reason}`);
      return false;
    }

    this.status.update({ code: "starting" });
    this.reason = reason;
    
    if (restart) this.restarts++;

    try {
      // Create runner based on cake type
      this.runner = await this.createRunner();
      
      // Execute process
      const result = await this.runner.run(() => {
        this.status.update({ code: "running" });
        this.started = new Date();
        this.connection.setActive();
        this.events.emit("started", { reason, restart });
      });

      // Handle completion
      if (result.success) {
        this.status.update({ code: "finished" });
        this.connection.setHealthy();
      } else {
        this.status.update({ code: "errored", error: result.error });
        this.connection.setFaulty(result.error);
      }
      
      this.exited = new Date();
      this.events.emit("exited", { code: result.code, success: result.success });
      
    } catch (error) {
      this.status.update({ code: "errored", error });
      this.connection.setFaulty(error);
      this.events.emit("crashed", { error });
    }
 
    return true;
  }

  async stop(reason = "manual") {
    if (!this.canStop()) return false;

    this.status.update({ code: "stopping" });
    this.reason = reason;
    this.restarts = 0; // Reset restart counter on manual stop

    if (!this.runner) return true;

    const gracePeriod = this.cake.terminateGracePeriod ?? 5;
    const timeout = this.cake.terminateTimeout ?? 30;

    // Graceful termination sequence
    setTimeout(() => {
      this.runner?.kill("SIGTERM");
    }, gracePeriod * 1000);

    setTimeout(() => {
      this.runner?.kill("SIGKILL");
    }, timeout * 1000);

    this.events.emit("stopping", { reason });
    return true;
  }

  restart(reason = "manual") {
    this.stop(`${reason}_restart`);
    // Actual restart happens in watchdog when status becomes "stopped"
    this.pendingRestart = reason;
  }

  block(reason = "manual") {
    this.blocked = true;
    this.events.emit("blocked", { reason });
  }

  unblock(reason = "manual") {
    this.blocked = false;
    this.events.emit("unblocked", { reason });
  }

  // Process factory
  async createRunner() {
    if (this.cake.worker) {
      const { WorkerRunner } = await import("./runners/worker.js");
      return new WorkerRunner(this.cake, this.gaia);
    } else if (this.cake.cmd) {
      const { ProcessRunner } = await import("./runners/process.js");
      return new ProcessRunner(this.cake, this.gaia);
    } else {
      throw new Error(`Process ${this.cake.id}: No cmd or worker specified`);
    }
  }

  // Introspection
  getStatus() {
    return {
      id: this.cake.id,
      status: this.status.get().code,
      blocked: this.blocked,
      restarts: this.restarts,
      pid: this.pid,
      started: this.started,
      exited: this.exited,
      connection: this.connection.isHealthy.get(),
      reason: this.reason
    };
  }

  isPendingRestart() {
    return !!this.pendingRestart;
  }
}

// systems/gaia/lifecycle/watchdog.js
import { delay } from "@std/async";

const WATCHDOG_INTERVAL = 5000; // 5 seconds

export async function pup(gaia) {
  gaia.watchdog = {
    running: false,
    interval: null,
    tick: 0
  };

  gaia.watchdog.start = () => {
    if (gaia.watchdog.running) return;
    
    gaia.watchdog.running = true;
    gaia.watchdog.interval = setInterval(async () => {
      await watchdogTick(gaia);
    }, WATCHDOG_INTERVAL);
    
    gaia.logger.info("watchdog", "Pup watchdog started");
  };

  gaia.watchdog.stop = () => {
    if (!gaia.watchdog.running) return;
    
    clearInterval(gaia.watchdog.interval);
    gaia.watchdog.running = false;
    gaia.logger.info("watchdog", "Pup watchdog stopped");
  };

  // Auto-start watchdog
  gaia.watchdog.start();
}

async function watchdogTick(gaia) {
  gaia.watchdog.tick++;
  
  try {
    for (const process of gaia.processes()) {
      await processDecisions(process, gaia);
    }
    
    await systemMaintenance(gaia);
    
  } catch (error) {
    gaia.logger.error("watchdog", `Watchdog error: ${error.message}`);
  }
}

async function processDecisions(process, gaia) {
  const status = process.status.get();
  const cake = process.cake;
  
  // 1. Autostart decisions
  if (cake.autostart && status.code === "created") {
    await process.start("autostart");
    return;
  }
  
  // 2. Pending restart handling
  if (process.isPendingRestart() && status.code === "stopped") {
    await process.start(process.pendingRestart, true);
    process.pendingRestart = null;
    return;
  }
  
  // 3. Restart policy enforcement
  if (status.code === "finished" || status.code === "errored") {
    const msSinceExit = process.exited ? 
      Date.now() - process.exited.getTime() : Infinity;
    
    const restartDelay = cake.restartDelayMs ?? 
      (cake.watch ? 500 : 10000); // Faster for file watchers
    
    if (msSinceExit > restartDelay) {
      const restartPolicy = cake.restart ?? 
        (cake.autostart ? "always" : "never");
      
      if (restartPolicy === "always") {
        await process.start("policy_restart", true);
      } else if (restartPolicy === "error" && status.code === "errored") {
        await process.start("error_restart", true);
      }
    }
  }
  
  // 4. Timeout enforcement
  if (status.code === "running" && cake.timeout && process.started) {
    const secondsRunning = (Date.now() - process.started.getTime()) / 1000;
    if (secondsRunning > cake.timeout) {
      await process.stop("timeout");
    }
  }
  
  // 5. Health monitoring
  if (status.code === "running" && !process.connection.isHealthy.get()) {
    gaia.logger.warn("watchdog", `Process ${process.cake.id} unhealthy`);
    // Could trigger restart or alert
  }
}

async function systemMaintenance(gaia) {
  // Every 60 ticks (5 minutes at 5s intervals)
  if (gaia.watchdog.tick % 60 === 0) {
    gaia.logger.info("watchdog", `Heartbeat: ${gaia.processes().length} processes`);
    
    // Cleanup dead processes, log rotation, etc.
    await gaia.cleanup?.();
  }
}
