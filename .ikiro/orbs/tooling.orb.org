#+BEGIN_SRC javascript
import { Vector, shape, steer, v } from "@vivalence/typology";

const emitterInput = v.object({
  where: v.object({}, { additionalProperties: true }),
  thread: v.string().optional(),
  blacklist: v.unknown().optional(),
});

export const SURVEYING = (mode, daemon) => {
  const tools = new Vector();

  for (const peerSet of Object.values(daemon.modes ?? {})) {
    for (const peer of Object.values(peerSet ?? {})) {
      if (peer === mode || !peer.cake.emitter) continue;
      const { type, slug } = peer.manifest;

      for (const { pattern } of steer.rollup(peer.cake.emitter)) {
        if (!pattern.valence) continue;
        tools.open(
          {
            ...pattern,
            nature: `/${slug}/${pattern.nature}`,
            input: pattern.input ?? emitterInput,
          },
          async (ctx) => ctx.daemon.modes[type][slug].emit[pattern.nature](ctx.input),
        );
      }
    }
  }

  const compiled = shape.agentic(tools);
  mode.cake.harness ??= new Vector();
  mode.cake.harness.branch("/dialogue").use(async (ctx, next) => {
    ctx.hallucination.absorb(compiled);
    await next();
  });
};
#+END_SRC

import { Vector } from "@vivalence/typology";

export function proxy(refs) {
  const vector = new Vector();
  for (const ref of refs) vector.open({ nature: ref.slug, proxy: ref });
  return vector;
}
