#+begin_src javascript
import { Vector, shape, shard, soma, Socket, Session } from "@vivalence/typology";

export const CONVERSATIONAL = async (mode, daemon) => {
  if (!mode.cake.dialogue) return;

  const inbound = new Vector();

  inbound.open("/dialogue/engage", async (ctx) => {
    const session = ctx.socket.state.session;
    const thread = await daemon.entities.thread.findOne(ctx.input.thread);

    const history = await daemon.entities.turn.find(
      { thread },
      { orderBy: { createdAt: "ASC" } },
    );

    const userTurn = daemon.entities.turn.create({
      role: "user",
      parts: [{ type: "text", text: ctx.input.message }],
      parent: history.at(-1) ?? null,
      thread,
      mode: mode.id,
    });
    await daemon.entities.em.flush();

    const hallucinate = daemon.cortex.spawn()
      .add(...history, userTurn)
      .tune(ctx.input.tune ?? mode.cake.tune ?? "balanced");

    const dialogue = new Vector();
    dialogue.slurp(mode.cake.dialogue);
    const context = { hallucinate, input: ctx.input };
    await shape.object(dialogue)(context);
    const stream = context.output ?? await hallucinate.conversation.stream();

    let assistantTurn = null;
    let parent = userTurn;
    const created = [];

    try {
      for await (const packet of stream) {
        assistantTurn = soma.pour(assistantTurn, packet);
        session.send.dialogue.packet(packet);
        if (packet.event === "turn.close") {
          parent = daemon.entities.turn.create({
            role: assistantTurn.role,
            parts: assistantTurn.parts,
            meta: assistantTurn.meta,
            parent,
            thread,
            mode: mode.id,
          });
          created.push(parent);
          assistantTurn = null;
        }
      }
      await daemon.entities.em.flush();
    } catch (error) {
      for (const entity of created) daemon.entities.em.remove(entity);
      throw error;
    }

    session.send.dialogue.terminate({});
  });

  mode.aperture.open("/connect", shard.serve.websocket((ws) => {
    const socket = new Socket(ws, inbound);
    socket.state.session = new Session(inbound, socket);
  }));

  mode.dialogue = { inbound };
};

export const TerminalDossier = {
  name: "terminal",
  kind: () => Terminal,
  repository: (schema) => new LocalRepository({ kind: schema.kind(), persist: "viva.quarters" }),

  use: [
    async (ctx, next) => {
      await next();
      const terminal = ctx.entity;
      terminal.streams = { dialogue: new Queue(), error: new Queue() };
      terminal.teardowns = [];

      const openSession = (thread) => {
        const mode = thread.mode;
        if (!mode?.traits?.includes("CONVERSATIONAL")) return;
        if (!thread.traits?.includes("INSITU")) return;
        if (terminal.session) return;

        const inbound = new Vector();
        inbound.open("/dialogue/packet",    (frame) => terminal.streams.dialogue.enqueue(frame.input));
        inbound.open("/dialogue/terminate", () => terminal.streams.dialogue.enqueue({ event: "turn.close" }));

        const token = ctx.lighthouse?.$authority?.get()?.access;
        const socket = mode.connection.socket("/connect", inbound, { token });
        terminal.session = new Session(inbound, socket);
        terminal.session.moin().catch((error) => console.error("[terminal] moin failed", error));
      };

      const closeSession = () => {
        terminal.session?.close();
        terminal.session = null;
      };

      const teardown = terminal.$thread.subscribe((thread) => {
        closeSession();
        if (thread) openSession(thread);
      });
      terminal.teardowns.push(teardown);
    },
  ],
};

<script>
  import { onDestroy } from "svelte";

  let { terminal, embedded = false } = $props();

  let sessionState = $state(terminal.session?.$state.get() ?? "IDLE");
  let thread = $state(terminal.thread);
  let port = $state(terminal.port);
  let turns = $state([]);
  let input = $state("");

  const teardowns = [];

  teardowns.push(terminal.$thread.subscribe((value) => { thread = value; }));
  teardowns.push(terminal.$port.subscribe((value) => { port = value; }));

  $effect(() => {
    if (!terminal.session) { sessionState = "IDLE"; return; }
    const teardown = terminal.session.$state.subscribe((state) => { sessionState = state; });
    return teardown;
  });

  $effect(() => {
    if (!thread?.$turns) { turns = []; return; }
    const teardown = thread.$turns.subscribe((value) => { turns = [...value]; });
    return teardown;
  });

  onDestroy(() => { for (const teardown of teardowns) teardown(); });

  const docks = ["top", "right", "bottom", "left"];

  function cycleDock(delta) {
    const index = (docks.indexOf(port.dock) + delta + docks.length) % docks.length;
    terminal.port = { dock: docks[index] };
  }

  function send() {
    const message = input.trim();
    if (!message) return;
    terminal.session?.send?.dialogue?.engage?.({ thread: thread?.id, message });
    input = "";
  }

  function endSession() {
    terminal.session?.close();
    terminal.session = null;
    terminal.thread = null;
  }

  function resetSession() {
    terminal.session?.close();
    terminal.session = null;
    if (thread) terminal.thread = thread;
  }
</script>
