** decisions (mine — every one overturnable)

- the App splits into a shell and parts under =buffer/parts/= (the dojo precedent); the JD grammar the
  client needs lives in =buffer/jdex.js= as pure functions with their own suite over the rows snapshot —
  the App never imports server code (=jd.js=) and nothing leaves the mode.
- sections are cut from the TEXT at the shallowest heading below the title (=##= · =*=), not from the
  renderer's AST: a section is a line range, so fold · reorder · the outline need no serializer and the
  drapes parsers stay internal. The renderer draws each section's body through =Markdown= · =Org= · =Json=.
- a reference is =→ NN.NN= or =[[NN.NN …]]=; the reader rewrites it into the format's own link grammar
  (=[→ 22.04](#vdex:22.04)= · =[[#vdex:22.04][→ 22.04]]=) and one delegated click on the prose jumps. The
  file on disk keeps =→ 22.04=. The real JDex carries no arrows today — this is the reader side of the
  design's convention.
- read · edit only. The design's three segments (read · edit · raw) collapse: edit IS the source until a
  block editor exists; a raw tab beside an identical edit tab would lie.
- keys live on the App's root (=tabindex="-1"= · =role="presentation"=), never on the window: a buffer App
  is one of several panels; focus inside the App is the gate. Inputs and textareas swallow keys; ⌥ reads
  =event.code= (the mac's ⌥1 is =¡=).
- every wire call lands in the status bar; a refusal is shown there in danger ink, never thrown at the
  screen — the aperture's messages name the next move and are meant to be read.
- drafts survive a switch: a dirty text is kept per slug and restored when the document comes back; the
  chip says =unsaved · ⌘S=.
- the twitch replaces the document's row in place (never a refetch); an open document re-reads only when
  its hash moved and the draft is clean; a deleted open document closes.
- rows are =$state.raw= (2000 pojos from the wire — never deep-proxied); the cursor and chrome are =$state=.
- disabled entries in the ⋯ menu and the pdf chips carry the design's LATER verbs with =later= as their key
  label — the design's intent shows, nothing pretends to work.

// the client's view of the JD grammar. A row carries symbol.vdex = { area, category, id: { [category]: id }, slot, format };
// a coordinate symbol is vdex.(area|category|id).<coordinate> with LABELED { name, description: the path on the mount }.

export const COORDINATE = /^vdex\.(area|category|id)\./;
const DEPTH = { area: 0, category: 1, id: 2 };
const HEADING = { md: /^(#{1,6})\s+(.+?)\s*#*\s*$/, mdx: /^(#{1,6})\s+(.+?)\s*#*\s*$/, org: /^(\*+)\s+(.+)$/ };
const REFERENCE = /(?:→|->)\s?(\d\d(?:\.\d\d)?)\b|\[\[(\d\d(?:\.\d\d)?)[^\]]*\](?:\[[^\]]*\])?\]/g;

export const place = (literal) => literal.symbol?.vdex ?? {};
export const document = (literal) => literal.trait.DOCUMENT;

export const coordinateOf = (literal) => {
  const { area, category, id } = place(literal);
  return id ? Object.entries(id)[0].join(".") : (category ?? area ?? null);
};

export const numeric = (coordinate) => coordinate.split(/[.-]/).map(Number);

export const within = (area, category) => {
  const [low, high] = numeric(area);
  const held = Number(category);
  return held >= low && held <= high;
};

export const node = (symbol) => {
  const [, kind, ...rest] = symbol.slug.split(".");
  return {
    slug: symbol.slug,
    depth: DEPTH[kind],
    coordinate: rest.join("."),
    name: symbol.trait?.LABELED?.name ?? "",
    path: symbol.trait?.LABELED?.description ?? "",
  };
};

export const rank = (held) => {
  const [head, tail = 0] = numeric(held.coordinate);
  if (held.depth === 0) return [head, 0, 0, 0];
  return [Math.floor(head / 10) * 10, head, tail, held.depth];
};

export const before = (a, b) => {
  const [x, y] = [rank(a), rank(b)];
  return x[0] - y[0] || x[1] - y[1] || x[2] - y[2] || x[3] - y[3];
};

export const nodes = (symbols) => symbols.filter((symbol) => COORDINATE.test(symbol.slug)).map(node).sort(before);
export const areas = (held) => held.filter((one) => one.depth === 0);
export const categoriesOf = (held, area) => held.filter((one) => one.depth === 1 && within(area, one.coordinate));
export const idsOf = (held, category) => held.filter((one) => one.depth === 2 && one.coordinate.split(".")[0] === category);
export const areaOf = (held, category) => areas(held).find((area) => within(area.coordinate, category)) ?? null;
export const named = (held, coordinate) => held.find((one) => one.coordinate === coordinate) ?? null;

// documents under a category (any id, or loose); null = the unfiled — no category at all
export const filed = (documents, category) =>
  documents.filter((literal) => (category === null ? !place(literal).category : place(literal).category === category));

export const inArea = (documents, area) => documents.filter((literal) => place(literal).area === area);

export const under = (literal, held) => {
  const { area, category, id } = place(literal);
  if (held.depth === 0) return area === held.coordinate;
  if (held.depth === 1) return category === held.coordinate;
  const [head, tail] = held.coordinate.split(".");
  return id?.[head] === tail;
};

export const sorted = (documents, by) =>
  [...documents].sort(
    by === "title"
      ? (a, b) => document(a).title.localeCompare(document(b).title)
      : by === "mtime"
        ? (a, b) => (document(b).mtime ?? "").localeCompare(document(a).mtime ?? "")
        : (a, b) => byCoordinate(coordinateOf(a), coordinateOf(b)) || a.slug.localeCompare(b.slug),
  );

// the unfiled sort last; a loose file (its category alone) sorts before the ids under that category
const byCoordinate = (a, b) => (a === b ? 0 : a === null ? 1 : b === null ? -1 : a.localeCompare(b));

// the next free id under a category: one past the highest minted, NN.00 being the index
export const nextId = (held, category) => {
  const taken = idsOf(held, category).map((one) => Number(one.coordinate.split(".")[1]));
  return `${category}.${String((taken.length ? Math.max(...taken) : 0) + 1).padStart(2, "0")}`;
};

// id:22 · format:pdf · free text over slug + title
export const find = (documents, query) => {
  const trimmed = query.trim().toLowerCase();
  if (!trimmed) return [];
  const id = /id:(\S+)/.exec(trimmed)?.[1];
  const format = /format:(\S+)/.exec(trimmed)?.[1];
  const text = trimmed.replace(/(id|format):\S+/g, "").trim();
  return documents.filter(
    (literal) =>
      (!id || (coordinateOf(literal) ?? "").startsWith(id)) &&
      (!format || document(literal).format === format) &&
      (!text || literal.slug.includes(text) || document(literal).title.toLowerCase().includes(text)),
  );
};

// a document as sections: the lede, then one per heading at the shallowest level present. A leading markdown h1 is
// the title, not a section. A section's body excludes its own heading line; from/to are line ranges over the text.
export const sections = (text, format) => {
  const lines = (text ?? "").split("\n");
  const grammar = HEADING[format];
  const whole = [{ key: 0, head: null, level: 0, from: 0, to: lines.length, body: lines.join("\n") }];
  if (!grammar) return whole;
  const heads = lines.map((line, at) => {
    const match = line.match(grammar);
    return match && { at, level: match[1].length, head: match[2] };
  }).filter(Boolean);
  const title = format !== "org" && heads[0]?.level === 1 && lines.slice(0, heads[0].at).every((line) => !line.trim()) ? heads[0] : null;
  const rest = title ? heads.slice(1) : heads;
  if (!rest.length) return [{ ...whole[0], from: title ? title.at + 1 : 0, body: lines.slice(title ? title.at + 1 : 0).join("\n") }];
  const level = Math.min(...rest.map((one) => one.level));
  const cuts = rest.filter((one) => one.level === level);
  const start = title ? title.at + 1 : 0;
  const held = [];
  if (lines.slice(start, cuts[0].at).some((line) => line.trim())) held.push({ key: 0, head: null, level: 0, from: start, to: cuts[0].at, body: lines.slice(start, cuts[0].at).join("\n") });
  cuts.forEach((cut, index) => {
    const to = cuts[index + 1]?.at ?? lines.length;
    held.push({ key: held.length, head: cut.head, level, from: cut.at, to, body: lines.slice(cut.at + 1, to).join("\n") });
  });
  return held;
};

export const blocks = (body) => body.split(/\n\s*\n/).filter((chunk) => chunk.trim()).length;

// sections moved: the text before the first section stays, the sections follow the order given
export const rearranged = (text, format, order) => {
  const lines = (text ?? "").split("\n");
  const held = sections(text, format).filter((section) => section.head !== null);
  if (!held.length) return text;
  const prefix = lines.slice(0, held[0].from);
  const moved = order.map((key) => held.find((section) => section.key === key)).flatMap((section) => lines.slice(section.from, section.to));
  return [...prefix, ...moved].join("\n");
};

export const references = (text) => [...new Set([...(text ?? "").matchAll(REFERENCE)].map((match) => match[1] ?? match[2]))];

// the reader makes a reference a jump: → 22.04 and [[22.04 …]] become links to #vdex:22.04 in the renderer's own grammar
export const linked = (text, format) =>
  (text ?? "").replace(REFERENCE, (_, arrow, bracket) => {
    const coordinate = arrow ?? bracket;
    return format === "org" ? `[[#vdex:${coordinate}][→ ${coordinate}]]` : `[→ ${coordinate}](#vdex:${coordinate})`;
  });

export const size = (bytes) => (bytes < 1024 ? `${bytes} B` : bytes < 1048576 ? `${Math.round(bytes / 1024)} kB` : `${(bytes / 1048576).toFixed(1)} MB`);

export const age = (iso, now = Date.now()) => {
  if (!iso) return "";
  const days = Math.max(0, Math.round((now - new Date(iso)) / 86400000));
  return days < 1 ? "today" : days < 7 ? `${days}d` : days < 30 ? `${Math.round(days / 7)}w` : days < 365 ? `${Math.round(days / 30)}mo` : `${Math.round(days / 365)}y`;
};
