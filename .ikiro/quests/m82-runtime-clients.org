| target | a module receives | the model it builds into | the wire | the engine |

    if (record instanceof Bundle) return record;
    this.entries = record.entries ?? [];
    this.url = record.url ?? null;
  }

  static merge(...bundles) {
    const entries = new Map();
    for (const bundle of bundles) for (const entry of new Bundle(bundle).entries) entries.set(entry.mime, entry);
    return new Bundle({ entries: [...entries.values()] });
  }

  withUrl(url) { this.url = url; return this; }

  entry(mime) { return this.entries.find((entry) => entry.mime === mime); }

  negotiate(accept) { return accept.map((mime) => this.entry(mime)).find(Boolean) ?? null; }

  async load(accept) {
    const entry = this.negotiate(accept);
    if (!entry) throw new Error(`bundle offers none of ${accept}: ${this.entries.map((entry) => entry.mime)}`);
    if (verified.has(entry.integrity)) return verified.get(entry.integrity);
    const response = await fetch(this.url + entry.mount);
    if (!response.ok) throw new Error(`bundle fetch ${response.status}`);
    const text = await response.text();
    const digested = await digest(text);
    if (digested !== entry.integrity) throw new Error(`integrity sha256 ${digested.slice(0, 12)} is not ${entry.integrity.slice(0, 12)}`);
    const loaded = entry.mime.endsWith("+javascript") ? await imported(text) : { default: text, entry };
    verified.set(entry.integrity, loaded);
    return loaded;
  }

  get json() { return { entries: this.entries }; }
  toJSON() { return this.json; }
}

#+BEGIN_SRC js
// the ring — re-frame in 30 lines, the only kernel
export const ring = () => {
  let db = {};
  const handlers = new Map(), effects = new Map(), listeners = new Set();
  const dispatch = async ([name, ...args]) => {
    const handler = handlers.get(name);
    if (!handler) throw new Error(`no event ${name}`);
    const ctx = { coeffects: { db, event: [name, ...args] }, effects: {} };
    await handler.chain(ctx, async () => { ctx.effects = (await handler.body(ctx.coeffects, ...args)) ?? {}; });
    for (const [key, value] of Object.entries(ctx.effects)) await effects.get(key)(value, dispatch);
  };
  effects.set("db", (next) => { db = next; for (const fn of listeners) fn(db); });
  effects.set("dispatch", (event, dispatch) => dispatch(event));
  effects.set("dispatch-n", (events, dispatch) => Promise.all(events.map(dispatch)));
  const chain = (interceptors) => interceptors.reduceRight((next, step) => (ctx, done) => step(ctx, () => next(ctx, done)), (ctx, done) => done());
  return {
    dispatch,
    event: (name, body, interceptors = []) => handlers.set(name, { body, chain: chain(interceptors) }),
    fx: (name, run) => effects.set(name, run),
    sub: (deps, compute) => () => compute(...deps.map((dep) => dep())),
    watch: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
    read: () => db,
  };
};

** attempt 2 — the demand graph: the client as a live cache of the host graph — X

Unconstrained, no typology. // the graph — resources keyed by address, demand-counted; 30 lines, the only kernel
export const graph = () => {
  const held = new Map();
  const resource = (key, open) => {
    if (held.has(key)) return held.get(key);
    const node = { key, status: "pending", value: undefined, error: null, demand: 0, readers: new Set(), close: null };
    const notify = () => { for (const fn of node.readers) fn(node); };
    node.set = (value) => { node.value = value; node.status = "ready"; notify(); };
    node.fail = (error) => { node.error = error; node.status = "error"; notify(); };
    node.read = (fn) => {
      node.readers.add(fn);
      if (node.demand++ === 0) node.close = open(node);
      return () => { node.readers.delete(fn); if (--node.demand === 0) { node.close?.(); held.delete(key); } };
    };
    held.set(key, node);
    return node;
  };
  const derive = (key, deps, compute) =>
    resource(key, (node) => {
      const settle = () => deps.every((dep) => dep.status === "ready") && node.set(compute(...deps.map((dep) => dep.value)));
      const offs = deps.map((dep) => dep.read(settle));
      return () => offs.forEach((off) => off());
    });
  return { resource, derive };
};

// systems/ghost/prototypes/ring.js — the command loop: one db, handlers pure, effects data; the evaluator is a Vector
import { atom } from "nanostores";
import { contexts, Signal, shard, steer, Vector } from "@vivalence/typology";

export class Ring {
  db = atom({});
  loop = new Vector();
  effects = {};

  constructor(client) {
    this.loop.use(shard.context.spread(contexts.client, client)).use(async (ctx, next) => { ctx.db = this.db.get(); await next(); });
  }

  on(name, handler) { this.loop.open(name, handler); return this; }
  fx(name, run) { this.effects[name] = run; return () => delete this.effects[name]; }

  async dispatch(name, input) {
    const produced = await steer.dispatch.invoke(this.loop, new Signal(name), steer.strategy.direct)(input);
    for (const [key, value] of Object.entries(produced ?? {})) {
      if (!this.effects[key]) throw new Error(`ring: no effect ${key}`);
      await this.effects[key](value, this);
    }
  }
}
