prototype:

#+begin_src javascript
// mode
export const dataset = { note: "skills" };            // reader.lift → { walk: "skills" }
// systems/runtime/daemon/traits/primed.js — mirrors INTENTED
export const PRIMED = async (mode, daemon) => {
  const sources = new Dataset(mode.module.dataset ?? {}).sources.note ?? [];
  return async () => {
    for (const source of sources)
      for (const row of await pull(source, mode.module.mount))
        await daemon.entities.note.ensure({ ...row, mode: mode.entity.id });
    await daemon.entities.em.flush();
  };
};
#+end_src
