import { atom } from "nanostores";
import { KeySignal, Queue, Vector, shape, shard, steer } from "@vivalence/typology";

export const translation = new Vector()
  .open("*", (ctx) => ctx.legend.presses.enqueue(ctx.signal))
  .open("ctrl-v", (ctx) => ctx.legend.presses.enqueue(new KeySignal("leader")));

export class Legend {
  presses = new Queue();
  $chord = atom(null);
  translation = new Vector().use(shard.context.bind("legend", this)).graft(translation);

  constructor(client) {
    this.client = client;
    this.presses.$backlog.listen((presses) => this.traverse(presses));
  }

  get chord() {
    return this.$chord.get();
  }

  dispatch(event) {
    steer.dispatch.invoke(this.translation, new KeySignal(event), steer.strategy.echo)(event);
  }

  traverse(presses) {
    if (presses.length === 0) return this.$chord.set(null);
    const keys = presses.map((signal) => signal.nature);
    const [effect, carry, steps, node] = steer.dispatch.traverse(this.client.activeTerminal.keymap, new KeySignal(keys));
    if (effect !== null) return this.execute(effect, carry).catch((error) => this.client.controller.stdout.fault(error));
    if (steps.length < keys.length) return this.flush();
    this.$chord.set(shape.strip(node));
  }

  async execute(effect, carry) {
    this.flush();
    const ctx = { client: this.client, output: undefined };
    await carry(ctx, steer.strategy.resolve(effect));
    if (ctx.output !== undefined) await this.client.activeTerminal.repl.dispatch(ctx.output);
  }

  flush() {
    this.presses.flush();
  }
}

* the law

1. a surface forwards the raw event; =KeySignal= spells the nature once — W3C =ctrl alt shift meta= in that order, a printable its character; the DOM arm takes =event.key= as a character only when it is one grapheme.
2. =legend.dispatch(event)= runs the event through =legend.translation= (Emacs =key-translation-map=) on every press; its =*= edge enqueues the press, a surface's leader edge enqueues =leader=.
3. the Legend's constructor listens on =presses.$backlog= and walks the held keys over =client.activeTerminal.keymap= once per change: an effect → flush, then run it (its line → =activeTerminal.repl.dispatch=, a fault → =stdout.fault=); a dead end → flush; a prefix → =$chord = shape.strip(node)=.
4. the terminal's keymap is read at every call: =shard.keymap.fromRepl(repl.commands)=, then =client.keymap=, then the mode's — later wins, field by field (=graft= merges patterns).
5. =client.keymap= declares the leader's groups that have keys under them (=e a h d s=), each with its valence; a key is labelled from =effect.valence= · =meta.valence=; a valence without " · " is the label itself.
6. =scope= asks the literals first, the =*= only when none matched; a pattern never writes on the signal it is asked about.
