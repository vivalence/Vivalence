import { Collection, EntitySchema, types } from "@mikro-orm/core";
import { DataEntity, DataRepository, DataSchema, LiteralEntity, LiteralRepository, LiteralSchema, UserEntity } from "@vivalence/runtime";

export const manifest = { type: "domain", slug: "recall", version: "0.0.1", traits: [] };

export const LiteralTraits = { TRANSLATED: "TRANSLATED", RANKED: "RANKED" };
export const UNRANKED = 999999;
export const INTERVALS = [1, 2, 4, 8, 16, 32];

export const rankOf = (trait) => Number.isInteger(trait?.RANKED?.rank) && trait.RANKED.rank >= 1 ? trait.RANKED.rank : UNRANKED;

export class RecallLiteralEntity extends LiteralEntity {
  rank = UNRANKED;
  retentions = new Collection(this);

  get retention() {
    return this.retentions.isInitialized() ? this.retentions.getItems()[0] : undefined;
  }
}

export class RecallLiteralRepository extends LiteralRepository {
  search(query) {
    return {
      $or: [
        { slug: { $like: `%${query}%` } },
        { "trait.TRANSLATED.known": { $like: `%${query}%` } },
        { "trait.TRANSLATED.learning": { $like: `%${query}%` } },
      ],
    };
  }

  owner() {
    return this.getEntityManager().getFilterParams("user")?.user;
  }

  novel(options = {}) {
    return this.find({ retentions: { $none: { user: this.owner() } } }, { orderBy: { rank: "ASC" }, ...options });
  }

  due(options = {}) {
    return this.find(
      { retentions: { $some: { user: this.owner(), nextAt: { $lt: new Date() } } } },
      { populate: ["retentions"], ...options },
    );
  }

  async feed(limit = 10) {
    const due = await this.due({ limit });
    const novel = due.length < limit ? await this.novel({ limit: limit - due.length }) : [];
    return [...due, ...novel];
  }
}

export class RankSubscriber {
  getSubscribedEntities() {
    return [RecallLiteralEntity];
  }
  beforeCreate({ entity }) {
    entity.rank = rankOf(entity.trait);
  }
  beforeUpdate({ entity }) {
    entity.rank = rankOf(entity.trait);
  }
}

export const RecallLiteralSchema = new EntitySchema({
  class: RecallLiteralEntity,
  extends: LiteralSchema,
  name: "Literal",
  tableName: "Literal",
  repository: () => RecallLiteralRepository,
  properties: {
    traits: { type: types.json, enum: true, array: true, items: () => Object.values(LiteralTraits), defaultRaw: `'[]'` },
    rank: { type: types.integer, default: UNRANKED },
    retentions: { kind: "1:m", entity: () => RetentionEntity, mappedBy: (retention) => retention.literal, strategy: "select-in" },
  },
});

export class RetentionEntity extends DataEntity {
  user;
  literal;
  streak = 0;
  seen = 0;
  lastSignal = null;
  nextAt = new Date();

  review(remembered) {
    this.seen += 1;
    this.streak = remembered ? this.streak + 1 : 0;
    this.lastSignal = remembered ? "SUCCESS" : "FAILURE";
    const days = remembered ? INTERVALS[Math.min(this.streak, INTERVALS.length) - 1] : 0;
    this.nextAt = new Date(Date.now() + days * 86400000);
    return this;
  }
}

export const RetentionSchema = new EntitySchema({
  class: RetentionEntity,
  extends: DataSchema,
  name: "Retention",
  tableName: "Retention",
  uniques: [{ properties: ["user", "literal"] }],
  filters: { user: { cond: (args) => ({ user: args.user }), default: true } },
  properties: {
    user: { kind: "m:1", entity: () => UserEntity, updateRule: "cascade", deleteRule: "cascade" },
    literal: { kind: "m:1", entity: () => RecallLiteralEntity, updateRule: "cascade", deleteRule: "cascade" },
    streak: { type: types.integer },
    seen: { type: types.integer },
    lastSignal: { type: types.string, nullable: true },
    nextAt: { type: Date },
  },
});

export const entities = {
  literal: {
    type: "literal",
    entity: RecallLiteralEntity,
    schema: RecallLiteralSchema,
    repository: RecallLiteralRepository,
    subscriber: new RankSubscriber(),
  },
  retention: {
    type: "retention",
    entity: RetentionEntity,
    schema: RetentionSchema,
    repository: DataRepository,
  },
};
