* yap
Essentially nothing that is an /entity/ anymore.

Every entity under =typology/entities= is already a =RemoteRepository= over a daemon or lighthouse branch — mode, thread, buffer, turn, activity, intent, literal, symbol, daemon dossier. Terminal was the one locally-owned entity (its own id, its own persistence, its own lifecycle). Once it moves, Anima owns no entities; it owns /state/:

#+BEGIN_SRC md
Anima-owned (after the move)
├─ stores/lighthouse.js      $authority, $identity, $status      — session, cached from wire
├─ stores/terminals.js       $entities, $active                   — projection of lighthouse Terminal rows
├─ stores/bridge/            geometry, panes, gesture, dock math  — viewport/layout, device-bound
├─ stores/box/               device, drivers                      — hardware (mic, keys, screen)
├─ Stall / $settling          per-terminal runtime                — ephemeral, re-derived
└─ prototypes/dictation      in-flight input                      — never persisted
#+END_SRC

That is the right shape for what you said: Anima is a display. The division of ownership becomes clean:

#+BEGIN_SRC md
lighthouse  →  who you are, what you have open, how you like it   (Identity, Terminal, Authority)
daemon      →  what you're working on                              (Thread, Buffer, Mode, …)
anima       →  where it is on this screen, right now               (bridge, box, stall)
#+END_SRC

Two things to decide while you're in there:

- =dock= is split: the /preference/ goes to =Identity.configured.dock=, the /per-terminal/ value to =Terminal.docked=, but =bridge/geometry= (window size, pane split) stays local — it's a property of the glass, not the user. Keep it in localStorage, keyed by device, don't lift it.
- =$settling= and the serialized-reference dance in =app/terminals.js= exist because the terminal held a thread id it couldn't resolve yet. With =seated: { daemon, thread }= living on the lighthouse row the same race exists — the daemon may be unreachable — so the resolver survives, it just reads from the row instead of localStorage.

Net: Anima shrinks from "client with a database" to "client with a cache", and the only truly Anima-owned things are the ones that would be wrong to sync across devices.
#+END_QUOTE

export function authenticate(provider) {
  return async (ctx, next) => {
    if (ctx.lighthouse) return await next();
    const header = ctx.request.headers?.get("authorization");
    const param = ctx.request.url.searchParams?.get("token");
    const token = header?.startsWith("Bearer ") ? header.split(" ")[1] : param;

    try {
      ctx.lighthouse = await provider.authenticate(token);
    } catch (error) {

    await next();
  };
}

      .createQueryBuilder("i")

export function authenticate() {
  return async (ctx, next) => {
    const header = ctx.request.headers?.get("authorization");

    if (!ctx.identity) return refuse(ctx, "IDENTITY_NOT_FOUND", "Identity not found");
    await next();
  };
}

import * as jose from "jose";
import { JSONFilePreset } from "lowdb/node";

const ACCESS_TTL = "1d";
const REFRESH_TTL = "180d";
const REFRESH_MS = 180 * 24 * 60 * 60 * 1000;
const SWEEP_MS = 60 * 60 * 1000;

export async function tokens(secret, file) {
  const key = new TextEncoder().encode(secret);
  const db = await JSONFilePreset(file, { refresh: {} });
  await db.read();

  const sign = (payload, expiration) =>
    new jose.SignJWT(payload).setProtectedHeader({ alg: "HS256" }).setIssuedAt().setExpirationTime(expiration).sign(key);

  const sweep = async () => {
    const now = Date.now();
    let changed = false;
    for (const [token, entry] of Object.entries(db.data.refresh)) {
      if (entry.expiresAt && now > entry.expiresAt) {
        delete db.data.refresh[token];
        changed = true;
      }
    }
    if (changed) await db.write();
  };

  const jwt = {
    create: (payload, expiration = ACCESS_TTL) => sign(payload, expiration),
    verify: async (token) => (await jose.jwtVerify(token, key)).payload,
  };

  const rft = {
    create: async (payload) => {
      const token = await sign({ ...payload, type: "refresh" }, REFRESH_TTL);
      db.data.refresh[token] = { ...payload, createdAt: Date.now(), expiresAt: Date.now() + REFRESH_MS };
      await db.write();
      return token;
    },
    verify: (token) => {
      const entry = db.data.refresh[token];
      if (!entry) return false;
      if (entry.expiresAt && Date.now() > entry.expiresAt) {
        delete db.data.refresh[token];
        return false;
      }
      return true;
    },
    revoke: async (token) => {
      delete db.data.refresh[token];
      await db.write();
    },
    sweep,
  };

  await sweep();
  const interval = setInterval(sweep, SWEEP_MS);

  return { jwt, rft, release: () => clearInterval(interval) };
}

const ABSENT = Symbol("absent");
const UNREACHABLE = Symbol("unreachable");
const PENDING = Symbol("pending");
const DOCK_DEBOUNCE_MS = 250;

const named = (outcome) => (outcome === ABSENT ? "absent" : outcome === UNREACHABLE ? "unreachable" : "resolved");

const same = (a, b) => JSON.stringify(a) === JSON.stringify(b);

export function sync({ terminals }) {
  const write = (terminal, data) =>
    terminals.repository
      .updateOne({ id: terminal.id }, data)
      .catch((error) => logger.entry(`terminals/sync/${terminal.id}`).fault(error));

  const dock = (terminal) => {
    let timer = null;
    return terminal.$dock.subscribe((value) => {
      if (same(value, terminal.trait.DOCKED)) return;
      clearTimeout(timer);
      timer = setTimeout(() => {
        const traits = terminal.traits.includes("DOCKED") ? terminal.traits : [...terminal.traits, "DOCKED"];
        write(terminal, { traits, trait: { ...terminal.trait, DOCKED: value } });
      }, DOCK_DEBOUNCE_MS);
    });
  };

  const seat = (terminal) =>
    terminal.$thread.subscribe((thread) => {
      const next = { daemon: thread?.daemon?.id ?? null, thread: thread?.id ?? null };
      const current = { daemon: terminal.daemon?.id ?? terminal.daemon ?? null, thread: terminal.thread ?? null };
      if (same(next, current)) return;
      write(terminal, next);
    });

  let inner = [];
  const off = terminals.$entities.subscribe((entities) => {
    inner.forEach((unsubscribe) => unsubscribe());
    inner = entities.flatMap((terminal) => [dock(terminal), seat(terminal)]);
  });
  return () => {
    off();
    inner.forEach((unsubscribe) => unsubscribe());
  };
}

export function settle({ terminals, lighthouse }) {
  const restore = async (terminal) => {
    const daemon = terminal.daemon;
    if (!daemon?.status?.is("healthy") || !daemon.entities?.thread) return PENDING;
    try {
      const entity = await daemon.entities.thread.findOne({ id: terminal.thread }, { populate: ["mode", "intent"] });
      if (entity) {
        daemon.entities.thread.resolve?.(entity);
        return entity;
      }
    } catch (error) {
      logger.entry("settle/thread").note({ id: terminal.thread, daemon: daemon.slug }).fault(error);
      return UNREACHABLE;
    }
    return ABSENT;
  };

  const pass = async () => {
    let settling = null;
    const record = (data) => (settling ??= logger.entry("settle").open()).note(data);
    for (const terminal of terminals.entities) {
      if (!terminal.thread || terminal.$thread.get()) {
        terminal.$settling.set(null);
        continue;
      }
      const outcome = await restore(terminal);
      if (outcome === PENDING) {
        terminal.$settling.set({ thread: terminal.thread });
        continue;
      }
      record({ message: `thread ${named(outcome)}`, terminal: terminal.id, thread: terminal.thread });
      if (outcome === UNREACHABLE) continue;
      if (outcome === ABSENT) {
        await terminals.repository
          .updateOne({ id: terminal.id }, { daemon: null, thread: null })
          .catch((error) => logger.entry(`terminals/unseat/${terminal.id}`).fault(error));
      } else {
        terminal.seat(outcome);
      }
      terminal.$settling.set(null);
    }
    settling?.close();
  };

  let inflight = null;
  let queued = false;
  const attempt = () => {
    if (inflight) {
      queued = true;
      return inflight;
    }
    inflight = (async () => {
      do {
        queued = false;
        await pass();
      } while (queued);
      inflight = null;
    })();
    return inflight;
  };

  let statuses = [];
  const offDaemons = lighthouse.$daemons.subscribe((daemons) => {
    statuses.forEach((unsubscribe) => unsubscribe());
    statuses = daemons.map((daemon) => daemon.status.$transient.subscribe(() => attempt()));
    attempt();
  });
  const offTerminals = terminals.$entities.subscribe(() => attempt());
  return () => {
    offDaemons();
    offTerminals();
    statuses.forEach((unsubscribe) => unsubscribe());
  };
}
