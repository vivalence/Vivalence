// systems/runtime/gestalten/belt/turns.js
import { soma } from "@vivalence/typology";
import { TurnEntity } from "@vivalence/runtime";

export async function* record(entities, turn, thread, mode, packets) {
  const em = entities.em.fork();
  let folded = null;
  let parent = em.getReference(TurnEntity, turn.id);
  let persisted = 0;
  try {
    for await (const packet of packets) {
      folded = soma.transcript(folded, packet);
      for (const row of folded.turns.slice(persisted)) {
        parent = em.create(TurnEntity, { role: row.role, parts: row.parts, meta: row.meta, parent, thread, mode });
      }
      persisted = folded.turns.length;
      yield packet;
    }
    parent = verdict(em, folded, parent, thread, mode);
    await em.flush();
  } catch (error) {
    em.clear();
    throw error;
  }
}

export const verdict = (em, folded, parent, thread, mode) => {
  const close = folded?.meta;
  if (!close || close.state === "complete") return parent;
  const last = folded.turns.at(-1);
  if (last?.role === "assistant" && last.parts.length) return parent;
  if (last?.role === "assistant") {
    parent.meta = { ...last.meta, ...close };
    return parent;
  }
  return em.create(TurnEntity, { role: "assistant", parts: [], meta: close, parent, thread, mode });
};

export const append = async (entities, turn, thread, mode, rows) => {
  let parent = turn;
  for (const row of rows) {
    parent = await entities.turn.chain({ role: row.role, parts: row.parts, meta: row.meta, parent, thread, mode });
  }
};
