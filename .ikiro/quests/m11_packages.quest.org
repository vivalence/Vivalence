#+TITLE: M11 — Packages (the registry dissolves into a package supply)
#+AUTHOR: Vivalence
#+STATUS: DESIGN
#+DATE: 2026-07-01

* Intent

Turn registry loading into a functional fold over a declared list of *packages*,
and flatten the on-disk tree. Two moves, one ontology:

- =a= — registry loading becomes =for (source of variant.packages) vip.mount(source)=.
  Multiple registries, not one hardcoded quad of tier-branches.
- =b= — drop the two intermediary layers (=<tier>/@vivalence/=). A module's
  path stops carrying identity; the manifest already does.

The unifying idea: *=registry= is not a thing, =package= is.* A package is a
self-declaring discovery scope (=type: package=). =owner= IS the package
identity — already the top axis of the pensieve (=owner→type→slug→version=). The
variant *declares* which packages to supply (authored, portable); the pensieve
is filled from them; the runtime stays package-blind (=accio= never knows a
package exists).

This quest resolves =a= + =b= for the *in-tree* case. The *out-of-tree* apparatus
(ledger, clone/materialize, =viva install package=) is deferred — in-tree
tautologises it (source = mount, version = working copy).

* The locked design (the decision trail)

** owner = package identity
=pensieve.register= keys =owner→type→slug→version=. Reinterpret =owner= as *the
package*, not "the org". =@vivalence= is the core package, not a namespace. Two
packages shipping =game/flashcard= land under different owners → different
subtrees → no collision. The package partition already exists; we only renamed
what =owner= means. The one ripple: =owner= must be *stamped by the package at
mount*, not defaulted to =@vivalence= in =pensieve.register=.

** supply / demand — system vs variant
- *system = supply* — materialises packages, fills the pensieve. Host-global:
  the pensieve lives on the =paladin= singleton, and =registry()= runs at
  runtime-boot *before* variant composition. Packages are pre-variant supply.
- *variant = demand* — composes module *identities* (=daemon.modes =
  [@vivalence/game/write]=), resolved from the filled pensieve. Package-blind at
  the *runtime* level.

** variant declares, ledger pins (flake.nix / flake.lock)
"Variant or ledger?" is a false binary — they hold different objects.
- *DECLARATION* — authored, portable, in-repo. =variant.packages=. = flake.nix inputs.
- *LEDGER* — generated, per-machine, pins resolved rev+path. = flake.lock.

The ledger is *generated* (the system dir is 100% derived, never authored) — so
it *cannot* hold authored intent. The declaration must live in an authored file;
the variant is the natural home (one-variant-per-host, already declares its
world: runtime/clients/daemons/services). *Runtime* stays package-blind; the
*variant* (the author's world-spec) is package-aware. Author ≠ runtime.

** the sanity cut — in-tree needs no ledger
For the in-tree self-package: =source = mount= (path as-is), =rev = working copy=.
Recording that in a ledger is a no-op tautology. So: NO ledger, NO
=materialize()=, NO =viva install=, NO system-dir redesign in this quest. Build
them when the first *out-of-tree* package forces it.

** deno static-import resolution is already solved (no S0)
=@vivalence/*= resolves by *workspace member name* (=deno.jsonc workspace[]= →
each member's =deno.jsonc "name"=), NOT via =import_map.json= (which holds only
=@std=/=@cliffy=/=npm=). Name-resolution is global within a config and
*location-independent*: a dynamically-imported out-of-tree =.viva.js= resolves
its =@vivalence/typology= against the running process's config, by name. And
spawns already pass =--config <root>= (=ghost/trajectories/instance/target.js:28=,
=ghost/ghost.sh:6=). Registry modules statically import ONLY core barrels
(=typology 161 · shared 63 · drapes 29 · paladin 17 · vector 15 · entities 12=);
ZERO registry→registry static imports (those go through =accio=). So the
import-map generator I first proposed is redundant with the workspace — killed.

** the boot order is already correct
=systems/runtime/run.js:6= =await paladin.variant.mount()= runs BEFORE
=systems/runtime/die.js:17= =lifecycle.population.registry(this)=. So
=variant.packages= is populated when the loader folds. No reorder.

** =subsystems/paladin/prototypes/vip.js= — mount reads the package, stamps owner (M11.1)
=mount(root)= now reads =root/package.viva.js= for the owner and stamps it onto
every module it walks. The package assigns identity; module files stay

#+BEGIN_SRC javascript :tangle ../../subsystems/paladin/prototypes/vip.js :mkdirp yes
import { is, cast, Path } from "@vivalence/typology";
import { Pensieve } from "./pensieve.js";

export class Vip {
  constructor(paladin) {
    this.paladin = paladin;
    this.pensieve = new Pensieve();
  }

  async mount(root) {
    // the package self-declares — its manifest owner stamps every module it holds.
    const pkg = await this.paladin.read.viva(root.branch("package.viva.js"));
    const owner = pkg.manifest.owner;
    if (!owner) throw new Error(`[VIP] package at ${root.absolute ?? root} declares no owner`);

    const paths = await this.paladin.find.viva(root);
    for await (const path of paths) {
      const cake = await this.paladin.read.viva(path);
      // stamp owner on a COPY — read.viva returns the live module namespace; don't mutate the import.
      // the package assigns identity; path does not. package owner WINS over any module-declared owner.
      this.pensieve.register({ ...cake, manifest: { ...cake.manifest, owner }, mount: new Path(path) });
    }

    return this;
  }

  async list(query = {}) {
    return this.pensieve.byType(query.type);
  }

  async accio(query) {
    const lookup = cast.lookup(query);
    const module = await this.pensieve.revelio(lookup);
    if (module) return module;
    throw new Error(`[VIP] Module 404: ${JSON.stringify({ lookup })}`);
  }

  async accioMany(many) {
    return await Promise.all(many.map(async (query) => {
      if (typeof query === "object" && query !== null && typeof query.module === "string") {
        return { service: await this.accio(query.module), mask: query };
      }
      return await this.accio(query);
    }));
  }

  async accioMap(many) {
    const accioedModules = await Promise.all(
      Object.entries(many).map(async ([slug, query]) => {
        if (!query) return null;
        if (typeof query === "string") {
          return [slug, await this.accio(query)];
        } else if (typeof query.module === "string") {
          return [slug, await this.accio(query.module)];
        } else if (is.array(query)) {
          return [slug, await this.accioMany(query)];
        } else if (is.object(query)) {
          return [slug, await this.accioMap(query)];
        }
      }),
    );
    const filteredModule = Object.fromEntries(
      accioedModules.filter((entry) => entry !== null),
    );
    return filteredModule;
  }

  toJSON() {
    return { pensive: this.pensieve };
  }
}
#+END_SRC

** =subsystems/paladin/prototypes/pensieve.js= — drop the hardcoded owner default (M11.1)
=owner= is set by =vip.mount= from the package. The old
=owner ??= "@vivalence"= line goes; a missing owner is now an error (mount must

#+BEGIN_SRC javascript :tangle ../../subsystems/paladin/prototypes/pensieve.js :mkdirp yes
import * as semver from "@std/semver";
import { Manifest } from "./manifest.js";

export class Pensieve extends Map {
  register(cake) {
    if (!cake || !cake.manifest) console.log("no manifest", cake);
    cake.manifest = new Manifest(cake.manifest);
    const { owner, type, slug, version } = cake.manifest;
    // owner is stamped by vip.mount from the package manifest — never defaulted here.
    if (!owner) throw new Error(`[Pensieve] register: no owner (mount must stamp) — ${type}/${slug}`);

    if (!this.has(owner)) this.set(owner, new Map());
    const ownerMap = this.get(owner);

    if (!ownerMap.has(type)) ownerMap.set(type, new Map());
    const typeMap = ownerMap.get(type);

    if (!typeMap.has(slug)) typeMap.set(slug, new Map());
    const slugMap = typeMap.get(slug);

    slugMap.set(version, cake);
    return this;
  }

  async revelio({ owner, type, slug, version }) {
    const ownerMap = this.get(owner);
    if (!ownerMap) return null;

    const typeMap = ownerMap.get(type);
    if (!typeMap) return null;

    const slugMap = typeMap.get(slug);
    if (!slugMap) return null;

    if (!version) return this.latest(slugMap);

    const versions = Array.from(slugMap.keys());
    const matchingVersion = versions.find((v) => semver.satisfies(v, version));
    return matchingVersion ? slugMap.get(matchingVersion) : null;
  }

  latest(slugMap) {
    const [top] = Array.from(slugMap.keys())
      .sort((a, b) => semver.compare(semver.parse(a), semver.parse(b)))
      .reverse();
    return slugMap.get(top);
  }

  byType(type) {
    const out = [];
    for (const ownerMap of this.values())
      for (const slugMap of ownerMap.get(type)?.values() ?? []) out.push(this.latest(slugMap));
    return out;
  }
}

import { describe, it } from "@std/testing/bdd";
import { expect } from "@std/expect";
import { Pensieve } from "../prototypes/pensieve.js";

const cake = (owner, type, slug, version = "0.0.1") => ({ manifest: { owner, type, slug, version } });

describe("Pensieve", () => {
  it("keys by manifest owner→type→slug→version", async () => {
    const pensieve = new Pensieve();
    pensieve.register(cake("@vivalence", "game", "write"));
    const found = await pensieve.revelio({ owner: "@vivalence", type: "game", slug: "write" });
    expect(found.manifest.slug).toBe("write");

  it("does NOT default a missing owner — mount must stamp", () => {
    const pensieve = new Pensieve();
    expect(() => pensieve.register({ manifest: { type: "game", slug: "write", version: "0.0.1" } }))
      .toThrow();

  it("two packages (owners) shipping the same type/slug do not collide", async () => {
    const pensieve = new Pensieve();
    pensieve.register(cake("@vivalence", "game", "flashcard"));
    pensieve.register(cake("@finn", "game", "flashcard"));
    const mine = await pensieve.revelio({ owner: "@vivalence", type: "game", slug: "flashcard" });
    const theirs = await pensieve.revelio({ owner: "@finn", type: "game", slug: "flashcard" });
    expect(mine.manifest.owner).toBe("@vivalence");
    expect(theirs.manifest.owner).toBe("@finn");
  });

  it("latest() resolves the highest semver within a slug", async () => {
    const pensieve = new Pensieve();
    pensieve.register(cake("@vivalence", "game", "write", "0.1.0"));
    pensieve.register(cake("@vivalence", "game", "write", "0.2.0"));
    const found = await pensieve.revelio({ owner: "@vivalence", type: "game", slug: "write" });
    expect(found.manifest.version).toBe("0.2.0");
  });
});
