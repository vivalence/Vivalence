* the law

- the world is dumb and visual; the target language is the only input. Every misunderstanding is visible in the world, never in a score.
- staggered: one linguistic axis per tier, over the same board. A tier opens from retention, never from a menu.
- learnable with nothing: every tier has two phases. OBEY — the world commands, the learner reacts (comprehension, the mechanics learned). COMMAND — the learner speaks, the world moves. A word enters the parser only after it has entered through the world's speech.
- feedback is the recast: a partial parse executes what was understood and answers in whole, correct Spanish. An empty parse asks /¿cómo?/ and the clock keeps running.
- two dials, independent: TIER is language, TEMPO is pressure. Tempo rises on success and falls on failure, per session; tier never resets.
- continuous: no levels, no menus, one world that never stops. Tasks spawn; a new tier does not interrupt.
- every command is a rep on known slots. Judging is the parser, deterministic, offline; the grade lands on `literal.review` exactly as dojo's does. No LLM in the loop. The LLM, later, writes varied recasts and composes boards.
- three is raw and mode-local, imported from node_modules, as `~/.viva/registry/droneaid/modes/assembly/pt10/buffer/scene.js` does today. The roadmap fork (`quests/roadmap/threlte-rapier-drapes.org`, `* the fork`) is NOT ruled here; `mando/scene.js` has the shape that quest already specs — `scene(container) → { scene, camera, renderer, loop, dispose }` — so the port is a move when beef rules.

** the loop

1. the emitter mints a buffer: `tier` from the learner's retention over the lexicon, `tempo` 1, `phase` OBEY.
2. OBEY: the world composes a command from the open tier, shows it, the learner presses. Right → `/heard` grades SUCCESS on the slots. Wrong → MISTAKE. Three clean reactions open COMMAND for that tier.
3. COMMAND: the learner types. `/command` parses with the tier's grammar, grades every slot, reviews, returns the command and the recast. The client applies the order to the sim and shows the recast.
4. the sim steps on `requestAnimationFrame`; goals spawn at `tempo`; tempo moves on the last ten.
5. a new tier opens when 80 % of the previous tier's lexicon is KNOWN or better; its words first appear in OBEY.

const strip = (text) =>
  text
    .toLowerCase()
    .normalize("NFC")
    .replace(/[¿?¡!.,;:]/g, " ")
    .trim();

export const tokens = (text) => strip(text).split(/\s+/).filter(Boolean);

const distance = (a, b) => {
  if (a === b) return 0;
  const rows = Array.from({ length: b.length + 1 }, (_, i) => i);
  for (let i = 1; i <= a.length; i++) {
    let previous = rows[0]++;
    for (let j = 1; j <= b.length; j++) {
      const current = rows[j];
      rows[j] = Math.min(rows[j] + 1, rows[j - 1] + 1, previous + (a[i - 1] === b[j - 1] ? 0 : 1));
      previous = current;
    }
  }
  return rows[b.length];
};

const near = (word, surface) => distance(word, surface) <= 1 || (word.length >= 4 && surface.startsWith(word.slice(0, 4)));

const grams = (rest) => [3, 2, 1].filter((n) => n <= rest.length).map((n) => rest.slice(0, n).join(" "));

const match = (lexemes, rest) => {
  for (const gram of grams(rest)) {
    for (const lexeme of lexemes) {
      if (lexeme.surfaces.includes(gram)) return { lexeme, surface: gram, hit: "HIT", used: gram.split(" ").length };
    }
  }
  const [word] = rest;
  for (const lexeme of lexemes) {
    const surface = lexeme.surfaces.find((s) => !s.includes(" ") && near(word, s));
    if (surface) return { lexeme, surface: word, hit: "NEAR", used: 1 };
  }
  return null;
};

export const slot = (name, lexemes) => (rest) => {
  const found = match(lexemes, rest);
  if (!found) return { ok: false, slots: [{ slot: name, hit: "MISS", word: rest[0] ?? null }], rest };
  return {
    ok: true,
    slots: [{ slot: name, slug: found.lexeme.slug, canonical: found.hit === "HIT" ? found.surface : found.lexeme.surfaces[0], surface: found.surface, hit: found.hit }],
    rest: rest.slice(found.used),
  };
};

export const seq = (...parts) => (rest) =>
  parts.reduce(
    (acc, part) => {
      if (!acc.ok) return acc;
      const next = part(acc.rest);
      return { ok: next.ok, slots: [...acc.slots, ...next.slots], rest: next.rest };
    },
    { ok: true, slots: [], rest },
  );

export const alt = (...parts) => (rest) => {
  const attempts = parts.map((part) => part(rest));
  return attempts.find((a) => a.ok) ?? attempts.sort((a, b) => b.slots.length - a.slots.length)[0];
};

export const opt = (part) => (rest) => {
  const next = part(rest);
  return next.ok ? next : { ok: true, slots: [], rest };
};

export const parse = (grammar, text) => {
  const result = grammar(tokens(text));
  const complete = result.ok && result.rest.length === 0;
  return { ...result, ok: complete, partial: result.ok && !complete };
};

export const recast = (template, slots) =>
  (typeof template === "function" ? template(slots) : template)
    .replace(/\{(\w+)\}/g, (_, name) => slots.find((s) => s.slot === name && s.slug)?.canonical ?? "")
    .replace(/\s+/g, " ")
    .replace(/\s+,/g, ",")
    .replace(/^[,\s]+/, "")
    .trim();

#+BEGIN_SRC js :tangle modes/games/mando/dial.js
const STRONG = new Set(["KNOWN", "GRADUATED"]);

const OPEN = 0.8;

export const tier = (tiers, frontier) =>
  tiers.reduce((open, slugs, index) => {
    if (index === 0) return 0;
    const previous = tiers[index - 1];
    const strong = previous.filter((slug) => STRONG.has(frontier[slug])).length;
    return open === index - 1 && strong / previous.length >= OPEN ? index : open;
  }, 0);

const clamp = (lo, hi, n) => Math.min(hi, Math.max(lo, n));

export const tempo = (previous, history) => {
  const recent = history.slice(-10);
  if (recent.length < 5) return previous;
  const ratio = recent.filter(Boolean).length / recent.length;
  return clamp(1, 5, previous + (ratio > 0.8 ? 0.5 : ratio < 0.5 ? -0.5 : 0));
};

export const phase = (reactions) => (reactions.slice(-3).every(Boolean) && reactions.length >= 3 ? "COMMAND" : "OBEY");

import { Vector, v } from "@vivalence/typology";
import { parse, recast } from "./grammar.js";

const GRADE = { HIT: "SUCCESS", NEAR: "MISTAKE", MISS: "FAILURE" };

export const grades = (slots) =>
  slots.filter((slot) => slot.slug || slot.word).map((slot) => ({ literal: slot.slug ?? null, grade: GRADE[slot.hit] }));

const review = async (ctx, literal, grade) => {
  if (!literal) return null;
  const row = await ctx.daemon.entities.literal.findOne({ slug: literal });
  if (!row) return null;
  const retention = await row.review(grade, ctx);
  return { literal, grade, status: retention?.status ?? null };
};

const COMMAND_INPUT = v.object({
  buffer: v.string().desc("The game buffer the command is issued in."),
  tier: v.integer({ minimum: 0 }).desc("The open tier; picks the grammar."),
  text: v.string().desc("What the learner typed or said."),
});

const HEARD_INPUT = v.object({
  buffer: v.string().desc("The game buffer."),
  slugs: v.array(v.string()).desc("Literal slugs the world's command carried."),
  reacted: v.boolean().desc("Whether the learner reacted correctly."),
});

export const apertures = (grammars) =>
  new Vector()
    .open({ nature: "/command", input: COMMAND_INPUT }, async (ctx) => {
      const { grammar, template, command } = grammars[Math.min(ctx.input.tier, grammars.length - 1)];
      const result = parse(grammar, ctx.input.text);
      const reviews = await Promise.all(grades(result.slots).map((g) => review(ctx, g.literal, g.grade)));
      const understood = result.ok || result.partial;
      return {
        ok: result.ok,
        partial: result.partial,
        command: understood ? command(result.slots) : null,
        recast: understood ? recast(template, result.slots) : "¿cómo?",
        slots: result.slots,
        reviews: reviews.filter(Boolean),
      };
    })
    .open({ nature: "/heard", input: HEARD_INPUT }, async (ctx) => {
      const grade = ctx.input.reacted ? "SUCCESS" : "MISTAKE";
      const reviews = await Promise.all(ctx.input.slugs.map((slug) => review(ctx, slug, grade)));
      return { reviews: reviews.filter(Boolean) };
    });

import * as THREE from "three";

export const scene = (container, { span = 12, background = 0x0b0d10 } = {}) => {
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.setClearColor(background, 1);
  container.appendChild(renderer.domElement);

  const graph = new THREE.Scene();
  const camera = new THREE.OrthographicCamera(-span, span, span, -span, 0.1, 100);
  camera.position.set(span, span, span);
  camera.lookAt(0, 0, 0);

  graph.add(new THREE.AmbientLight(0xffffff, 0.7));
  const sun = new THREE.DirectionalLight(0xffffff, 0.8);
  sun.position.set(5, 10, 3);
  graph.add(sun);

  const resize = () => {
    const { clientWidth: w, clientHeight: h } = container;
    const aspect = w / Math.max(h, 1);
    camera.left = -span * aspect;
    camera.right = span * aspect;
    camera.updateProjectionMatrix();
    renderer.setSize(w, h, false);
  };
  const observer = new ResizeObserver(resize);
  observer.observe(container);
  resize();

  let frame = null;
  let last = performance.now();
  const loop = (tick) => {
    const run = (now) => {
      tick((now - last) / 1000);
      last = now;
      renderer.render(graph, camera);
      frame = requestAnimationFrame(run);
    };
    frame = requestAnimationFrame(run);
  };

  const dispose = () => {
    if (frame) cancelAnimationFrame(frame);
    observer.disconnect();
    graph.traverse((o) => {
      o.geometry?.dispose?.();
      [].concat(o.material ?? []).forEach((m) => m.dispose?.());
    });
    renderer.dispose();
    renderer.domElement.remove();
  };

  return { scene: graph, camera, renderer, loop, dispose, THREE };
};

export const box = (color, size = 0.8) => new THREE.Mesh(new THREE.BoxGeometry(size, size, size), new THREE.MeshStandardMaterial({ color }));

export const tile = (color, size = 1) => {
  const mesh = new THREE.Mesh(new THREE.PlaneGeometry(size, size), new THREE.MeshStandardMaterial({ color }));
  mesh.rotation.x = -Math.PI / 2;
  return mesh;
};

const { describe, it, expect } = specimen;

describe("mando grammar — tier 0 commands parse, near-misses grade MISTAKE, the recast is whole", () => {
  const { grammar, template } = grammars[0];

  it("parses a bare direction", () => {
    const r = parse(grammar, "a la izquierda");
    expect(r.ok).toBe(true);
    expect(r.slots.find((s) => s.slot === "direction").slug).toBe("izquierda.noun");

    expect(recast(template, r.slots)).toBe("sube arriba");

  lost: 0,
  units: Array.from({ length: units }, (_, i) => ({
    id: i,
    unit: ["soldado.noun", "camión.noun", "explorador.noun"][i % 3],
    color: Object.keys(COLOR)[i % 3],
    x: rand(w),
    y: rand(h),
    orders: [],
    clock: 0,
  })),
  goals: [],
  hazards: Array.from({ length: 6 }, () => ({ x: rand(w), y: rand(h) })),
});

const addressee = (world, command) =>
  world.units.find((u) => (!command.unit || u.unit === command.unit) && (!command.color || u.color === command.color)) ?? world.units[0];

export const order = (world, command) => {
  const target = addressee(world, command);
  if (!target) return world;
  const units = world.units.map((u) =>
    u !== target
      ? u
      : command.kind === "stop"
        ? { ...u, orders: [] }
        : { ...u, orders: [...u.orders, ...Array.from({ length: command.count }, () => DELTA[command.direction])] },
  );
  return { ...world, units };
};

const inside = (world, x, y) => x >= 0 && y >= 0 && x < world.w && y < world.h;

const stepUnit = (world, u, dt, tempo) => {
  const period = 0.6 / tempo;
  if (!u.orders.length) return u;
  const clock = u.clock + dt;
  if (clock < period) return { ...u, clock };
  const [[dx, dy], ...orders] = u.orders;
  const x = u.x + dx;
  const y = u.y + dy;
  return inside(world, x, y) ? { ...u, x, y, orders, clock: 0 } : { ...u, orders: [], clock: 0 };
};

export const spawn = (world, tempo) =>
  world.goals.length >= Math.ceil(tempo)
    ? world
    : { ...world, goals: [...world.goals, { id: crypto.randomUUID(), x: rand(world.w), y: rand(world.h), due: world.t + 30 / tempo }] };

export const step = (world, dt, tempo) => {
  const t = world.t + dt;
  let units = world.units.map((u) => stepUnit(world, u, dt, tempo));
  let score = world.score;
  let lost = world.lost;
  const goals = world.goals.filter((g) => {
    const reached = units.some((u) => u.x === g.x && u.y === g.y);
    if (reached) score += 1;
    const overdue = t > g.due;
    if (overdue) lost += 1;
    return !reached && !overdue;
  });
  units = units.map((u) => (world.hazards.some((h) => h.x === u.x && h.y === u.y) ? { ...u, orders: [], x: rand(world.w), y: rand(world.h) } : u));
  return spawn({ ...world, t, units, goals, score, lost }, tempo);
};

export const speak = (world, tier) => {
  const directions = Object.keys(DELTA);
  const direction = directions[rand(directions.length)];
  const count = tier >= 1 ? 1 + rand(3) : 1;
  const u = world.units[rand(world.units.length)];
  const command = { kind: "move", direction, count, unit: tier >= 2 ? u.unit : null, color: tier >= 2 ? u.color : null };
  const slugs = [direction, ...(tier >= 1 ? [["uno.numeral", "dos.numeral", "tres.numeral"][count - 1]] : []), ...(tier >= 2 ? [u.unit, u.color] : [])];
  return { command, slugs, expect: DELTA[direction], unit: u.id };
};

export const colour = (slug) => COLOR[slug] ?? 0xcccccc;
