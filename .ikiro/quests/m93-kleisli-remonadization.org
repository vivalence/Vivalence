export const kleisli = (arrows) => (value) => {
  try {
    return Promise.resolve(arrows.reduce((carried, arrow) => (carried instanceof Promise ? carried.then(arrow) : arrow(carried)), value));

    return Promise.reject(error);
  }
};

}
