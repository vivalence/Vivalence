// attempt 1 — the event ring: re-frame in 30 lines (m82:641-663)
export const ring = () => {
  let db = {};
  const handlers = new Map(), effects = new Map(), listeners = new Set();
  const dispatch = async ([name, ...args]) => {
    const handler = handlers.get(name);
    if (!handler) throw new Error(`no event ${name}`);
    const ctx = { coeffects: { db, event: [name, ...args] }, effects: {} };
    await handler.chain(ctx, async () => { ctx.effects = (await handler.body(ctx.coeffects, ...args)) ?? {}; });
    for (const [key, value] of Object.entries(ctx.effects)) await effects.get(key)(value, dispatch);
  };
  effects.set("db", (next) => { db = next; for (const fn of listeners) fn(db); });
  effects.set("dispatch", (event, dispatch) => dispatch(event));
  const chain = (interceptors) => interceptors.reduceRight((next, step) => (ctx, done) => step(ctx, () => next(ctx, done)), (ctx, done) => done());
  return { dispatch, event: (name, body, interceptors = []) => handlers.set(name, { body, chain: chain(interceptors) }), fx: (name, run) => effects.set(name, run), sub: (deps, compute) => () => compute(...deps.map((dep) => dep())), watch: (fn) => (listeners.add(fn), () => listeners.delete(fn)), read: () => db };
};
#+END_SRC

// attempt 2 — the demand graph: a resource per ADDRESS, demand-counted (m82:739-762)
export const graph = () => {
  const held = new Map();
  const resource = (key, open) => {
    if (held.has(key)) return held.get(key);
    const node = { key, status: "pending", value: undefined, error: null, demand: 0, readers: new Set(), close: null };
    const notify = () => { for (const fn of node.readers) fn(node); };
    node.set = (value) => { node.value = value; node.status = "ready"; notify(); };
    node.fail = (error) => { node.error = error; node.status = "error"; notify(); };
    node.read = (fn) => {
      node.readers.add(fn);
      if (node.demand++ === 0) node.close = open(node);
      return () => { node.readers.delete(fn); if (--node.demand === 0) { node.close?.(); held.delete(key); } };
    };
    held.set(key, node);
    return node;
  };
  const derive = (key, deps, compute) => resource(key, (node) => { const settle = () => deps.every((dep) => dep.status === "ready") && node.set(compute(...deps.map((dep) => dep.value))); const offs = deps.map((dep) => dep.read(settle)); return () => offs.forEach((off) => off()); });
  return { resource, derive };
};
#+END_SRC
