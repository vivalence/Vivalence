import { traits, v, Vector } from "@vivalence/typology";
import { ANNOTATED } from "../traits/literal/annotated.js";
import { TRANSLATED } from "../traits/literal/claims.js";

export const literal = new Vector()
  .open(
    {
      nature: "/literal/update",
      valence:
        "Rewrite a literal's text and its tokens. Find it first with lookup. When you change " +
        "TRANSLATED.learning of a sentence, send ANNOTATED.tokens for the new text in the same " +
        "call: one token per word, each with the slug of its word when lookup finds one. " +
        "A literal that wears VOCALIZED keeps its old recording: after a text change it " +
        "probably no longer speaks the text, so say that to the user.",
      input: v
        .object({
          slug: v.string().desc("its address inside the daemon").example("hola-como-estas"),
          TRANSLATED: TRANSLATED.schematic.optional(),
          ANNOTATED: ANNOTATED.schematic.optional(),
        })
        .example({
          slug: "hola-como-estas",
          TRANSLATED: { known: "Hi, how are you?", learning: "¡Hola! ¿Qué tal estás?" },
          ANNOTATED: {
            tokens: [
              { index: 0, form: "Hola", gloss: "hi", deprel: "discourse", literal: "hola.interjection" },
              { index: 1, form: "Qué", gloss: "what", deprel: "advmod", literal: "qué.pronoun" },
              { index: 2, form: "tal", gloss: "such", deprel: "fixed" },
              { index: 3, form: "estás", gloss: "you are", deprel: "root" },
            ],
          },
        }),
    },
    async (ctx) => {
      const { slug, ...trait } = ctx.input;
      const held = await ctx.entities.literal.findOneOrFail({ slug });
      const updated = await ctx.entities.literal.updateOne({ id: held.id }, traits.patch(held, trait));
      return { message: `updated literal ${updated.slug}`, literal: [updated] };
    },
  )
  .open(
    {
      nature: "/literal/rename",
      valence:
        "Give a literal a new slug. Every sentence, conjugation and retention note that names " +
        "the old slug is moved to the new one shortly after this call returns.",
      input: v
        .object({
          slug: v.string().desc("its address inside the daemon").example("hola-como-estas"),
          to: v.string().desc("its new address").example("hola-que-tal-estas"),
        })
        .example({ slug: "hola-como-estas", to: "hola-que-tal-estas" }),
    },
    async (ctx) => {
      const held = await ctx.entities.literal.findOneOrFail({ slug: ctx.input.slug });
      const updated = await ctx.entities.literal.updateOne({ id: held.id }, { slug: ctx.input.to });
      return { message: `renamed ${ctx.input.slug} → ${updated.slug}`, literal: [updated] };
    },
  );

const REFERENCES = [
  [
    "ANNOTATED",
    (payload, from, to) => ({
      ...payload,
      tokens: payload.tokens.map((token) => (token.literal === from ? { ...token, literal: to } : token)),
    }),
  ],
  [
    "CONJUGATED",
    (payload, from, to) => ({
      ...payload,
      infinitive: payload.infinitive === from ? to : payload.infinitive,
      paradigm: Object.fromEntries(
        Object.entries(payload.paradigm ?? {}).map(([form, slug]) => [form, slug === from ? to : slug]),
      ),
    }),
  ],
];

const checked = (ctx) => {
  const { entity } = ctx.input;
  for (const name of entity.traits) {
    const errors = [...ctx.entities.literal.schematics[name].errors(entity.trait[name])];
    if (!errors.length) continue;
    const faults = errors.map((error) => `${error.instancePath} ${error.message}`).join(" · ");
    throw new Error(`[${name}] ${entity.slug}: ${faults}`);
  }
};

const spelled = (ctx) => {
  const { entity } = ctx.input;
  if (!entity.traits.includes("ANNOTATED")) return;
  const text = entity.trait.TRANSLATED.learning;
  const strays = entity.trait.ANNOTATED.tokens.filter(
    (token) => !text.toLowerCase().includes(token.form.toLowerCase()),
  );
  if (!strays.length) return;
  const named = strays.map((token) => `"${token.form}"`).join(" · ");
  throw new Error(
    `[ANNOTATED] ${entity.slug}: token ${named} is not in the text "${text}" — ` +
      "send ANNOTATED.tokens for the new text in the same call",
  );
};

const referenced = async (ctx) => {
  const slugs = [...new Set([...annotated(ctx.input.entity), ...conjugated(ctx.input.entity)])];
  if (!slugs.length) return;
  const used = await ctx.entities.literal.find({ slug: { $in: slugs } });
  const unknown = slugs.filter((slug) => !used.some((word) => word.slug === slug));
  if (!unknown.length) return;
  throw new Error(`[ANNOTATED] ${ctx.input.entity.slug}: ${unknown.join(" · ")} is not in the corpus`);
};

const linked = async (ctx) => {
  const slugs = [...new Set([...annotated(ctx.input.entity), ...conjugated(ctx.input.entity)])];
  const { literal } = ctx.entities;
  const entity = await literal.findOneOrFail({ id: ctx.input.entity.id }, { populate: ["uses"] });
  const used = await literal.find({ slug: { $in: slugs } });
  entity.uses.set(used);
  await literal.em.flush();
};

const renamed = async (ctx) => {
  const from = ctx.input.changeSet.originalEntity.slug;
  const to = ctx.input.entity.slug;
  if (from === to) return;
  const { literal, note } = ctx.entities;
  const holders = await literal.find({ uses: ctx.input.entity.id });
  for (const holder of holders) {
    for (const [name, respell] of REFERENCES) {
      if (!holder.traits.includes(name)) continue;
      holder.trait = { ...holder.trait, [name]: respell(holder.trait[name], from, to) };
    }
  }
  const retentions = await note.find({ literal: ctx.input.entity.id, slug: `retention.${from}` });
  for (const retention of retentions) retention.slug = `retention.${to}`;
  await literal.em.flush();
};

export const twitch = new Twitch();

twitch.open("/before/literal/create", ranked);
