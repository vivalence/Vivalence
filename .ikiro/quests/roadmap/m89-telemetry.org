// subsystems/typology/prototypes/span.js
const CONFIG = {
  journal: 1000,
};

export class Span extends Signature {
  static tally = 0;

  constructor(signature = null, trace = null) {
    super(signature, trace);
    this.id ??= Span.tally++;
    if (!this.trace) {
      this.journal ??= [];
      this.channel ??= new Pipe();
    }
  }
  from(trace) { return super.from(trace, true); }
  get records() { return this.root.journal; }
  get pipe() { return this.root.channel; }
  to(...sinks) { this.pipe.to(...sinks); return this; }

  mark(verb, data) {
    const record = { span: this.id, trace: this.trace?.id ?? null, path: this.absolute, verb, at: performance.now() };
    if (data !== undefined) record.data = data;
    const journal = this.records;
    journal.push(record);
    if (journal.length > CONFIG.journal) journal.splice(0, journal.length - CONFIG.journal);
    this.pipe.send(record);
    return this;
  }
  open() / close() / note(data) → mark("open"|"close"|"note", data)
  fault(error) → mark("fault", { message: error?.message ?? String(error), code: error?.code ?? null })
}

export function span(name, pipe) {
  return async (ctx, next) => {
    const nature = typeof name === "function" ? name(ctx) : name;
    const parent = ctx.span;
    ctx.span = parent ? parent.branch(nature) : new Span(nature).to(pipe);
    ctx.span.open();
    try {
      await next();
    } finally {
      ctx.span.close();
      ctx.span = parent ?? ctx.span;
    }
  };
}

// subsystems/typology/specimen/snapshot.js
export function snapshot(subject, options = {}) {
  const { parse, locate, meta = {}, depth = 2, pick, omit, base, write = true, dry = false } = options;

  const selected = parse
    ? parse(subject)
    : pick
      ? Object.fromEntries(pick.map((key) => [key, fold(subject[key], depth)]))
      : fold(subject, depth);

  const pojo = omit
    ? Object.fromEntries(Object.entries(selected).filter(([key]) => !omit.includes(key)))
    : selected;

  const located = typeof locate === "function" ? locate(meta, pojo) : locate;
  const path =
    typeof located === "string" && located && base && !isAbsolute(located)
      ? join(base, located)
      : located;

  if (write && !dry) {
    if (typeof path !== "string" || !path) throw new Error("snapshot: locate must resolve to a path");
    Deno.writeTextFileSync(path, JSON.stringify(pojo, null, 2));
  }
  return { pojo, path };
}

#+BEGIN_SRC js :tangle subsystems/typology/schematics/primitives/telemetry.js
import { v } from "../v.js";

export const RANK = ["off", "fault", "control", "note", "wire"];

export const LEVEL = {
  fault: "fault",
  open: "control", close: "control", stop: "control", pause: "control", resume: "control", abort: "control",
  note: "note", subject: "note",
  request: "wire", response: "wire",
};

export const Level = v.enum(RANK).desc('A floor over the record verbs: off admits nothing, fault only faults, control adds the machine verbs, note adds note and subject, wire adds request and response. Example: "control"');

export const Filters = v.array(v.string({ pattern: "^/" })).desc('Path prefixes a surface admits; empty admits every path. Example: ["/runtime/runtime/daemon/chess"]');

export const Tail = v.object(
  {
    keep: v.enum(["faulty", "slower"]).desc('Tail sampling once the root seals: faulty writes a run only when a node faulted, slower only past threshold. Example: "faulty"'),
    threshold: v.number({ minimum: 0 }).default(500).optional().desc("Milliseconds a root must exceed for slower. Example: 500"),
  },
  { additionalProperties: false },
);

export const Surface = v.object(
  { level: Level.optional(), filters: Filters.optional(), tail: Tail.optional() },
  { additionalProperties: false },
).desc('One consumer of a controller stdout, narrowing the telemetry word key by key. Example: { "level": "fault" }');

export const Surfaces = v.object(
  {
    console: Surface.default({ level: "fault" }).optional().desc('A process stderr or the browser console. Example: { "level": "control" }'),
    ledger: Surface.default({}).optional().desc('<ledger>/logs/<instance>/spans.jsonl. Example: { "level": "control" }'),
    inspector: Surface.default({ level: "off" }).optional().desc('The story pane on a client surface. Example: { "level": "note" }'),
  },
  { additionalProperties: false },
);

export const Caps = v.object(
  {
    journal: v.integer({ minimum: 1 }).default(1000).optional().desc("Records a root span keeps in memory. Example: 1000"),
    story: v.integer({ minimum: 1 }).default(200).optional().desc("Children a root keeps in an inspector story, and roots it keeps. Example: 200"),
  },
  { additionalProperties: false },
);

export const Telemetry = v.object(
  { level: Level.default("note").optional(), filters: Filters.default([]).optional(), caps: Caps.default({}).optional(), surfaces: Surfaces.default({}).optional() },
  { additionalProperties: false },
).desc('The telemetry word of a ledger or an instance: a floor, prefixes, caps, three surfaces narrowing them. Example: { "level": "note", "surfaces": { "console": { "level": "control" } } }');
