export const dispatch = (client, terminal, line) => {
  const signal = new ReplSignal(line);
  const replDie = {
    signal,
    client,
    runtime: client.runtime,
    terminal,
    controller: (terminal === null ? client.controller : terminal.controller).branch(signal.pathname),
    repl: compose(client, terminal),
    output: undefined,
  };
  return steer.dispatch.execute(execution, replDie).then(() => replDie.output);
};

  .use(shard.controller.open)
  .do((replDie) => steer.dispatch.execute(replDie.repl.commands, replDie, replDie.signal));

export class Repl {
  commands = new Vector();

  constructor(terminal) {
    this.terminal = terminal;
  }

  dispatch(line) {
    const signal = new ReplSignal(line);
    const replDie = {
      signal,
      client: this.terminal.client,
      runtime: this.terminal.client.runtime,
      terminal: this.terminal,
      controller: this.terminal.controller.branch(signal.pathname),
      repl: this,
      output: undefined,
    };
    return steer.dispatch.execute(execution, replDie).then(() => replDie.output);
  }
}

import { atom } from "nanostores";
import { specimen } from "@vivalence/typology";
import * as ghost from "@vivalence/ghost";

const { rows } = ghost.lifecycle.client;

const settle = () => new Promise((resolve) => setTimeout(resolve, 0));

const NAMES = ["thread", "buffer", "turn", "activity", "terminal"];

const OWNED = {
  beef: { thread: [{ id: "th-beef" }], buffer: [{ id: "bu-beef" }], turn: [{ id: "tu-beef" }], activity: [{ id: "ac-beef" }], terminal: [{ id: "t-beef-1" }, { id: "t-beef-2" }] },
  lena: { thread: [{ id: "th-lena" }], buffer: [], turn: [], activity: [], terminal: [{ id: "t-lena-1" }] },
};

const world = () => {
  const $user = atom({ slug: "beef" });
  const streams = { opened: 0, closed: 0 };
  const reads = { count: 0 };
  const entities = { em: { drop: (name, id) => entities[name].$entities.set(entities[name].$entities.get().filter((entity) => entity.id !== id)) } };
  for (const name of NAMES) {
    entities[name] = {
      $entities: atom([]),
      find: async () => {
        reads.count += 1;
        const known = entities[name].$entities.get().map((entity) => entity.id);
        entities[name].$entities.set([...entities[name].$entities.get(), ...OWNED[$user.get().slug][name].filter((entity) => !known.includes(entity.id))]);
      },
      subscribe: () => {
        streams.opened += 1;
        return () => (streams.closed += 1);
      },
    };
  }
  const lighthouse = { $user, get user() { return $user.get(); } };
  const clientDie = { runtime: { lighthouse, entities } };
  const held = Promise.withResolvers();
  const running = rows(clientDie, () => held.promise);
  const ids = (name) => entities[name].$entities.get().map((entity) => entity.id);
  return { $user, streams, reads, ids, held, running };
};

* the law

1. a terminal holds exactly one =Repl=; the client holds none; a mode keeps its own =mode.repl= Vector.
2. =terminal.repl.dispatch(line)= is the only way a line runs; =client.dispatch= · =terminal.dispatch= do not exist.
3. =terminal.repl.commands= is the whole truth: the client's · =/shell= on a tty · the hosted modes' · the terminal's (=*= last) · every runtime mode's own =repl=, by reference, under =/mode/<type>/<slug>= and =/mode/<slug>=; rebuilt whole on every change of the runtime's mode list.
4. a command reads =ctx.repl.commands=; =/help= · the palette · =/metadata/repl= · the keymap's chord harvest read the same Vector.
5. the login screen is the whole screen until the lighthouse authorizes; it calls the lighthouse, never a line; nothing boots before it.
6. the terminal's keymap: the harvest of its commands, then the client's keymap, then the thread mode's — later wins.
7. ghost exports no surface; the opentui host is sheets', =host= on sheets' root barrel.
