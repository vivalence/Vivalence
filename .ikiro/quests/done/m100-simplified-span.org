export const seed = () => ({ nodes: new Map(), roots: [] });

export const chronicle = (records) => records.reduce(step, seed());

export const key = (path) => JSON.stringify(path);

const sprout = (story, record) => {
  const node = { path: record.path, nature: record.path.at(-1), timing: null, fault: null, entries: [], children: [] };
  story.nodes.set(key(record.path), node);
  (story.nodes.get(key(record.path.slice(0, -1)))?.children ?? story.roots).push(node);
  return node;
};

const MARKS = [
  [(record) => record.track === "open", (node, record) => (node.timing = { ...node.timing, begun: record.at })],
  [(record) => record.track === "close", (node, record) => (node.timing = { ...node.timing, sealed: record.at })],
  [(record) => record.track === "fault", (node, record) => (node.fault = { at: record.at, ...record.data })],
  [() => true, (node, record) => node.entries.push({ track: record.track, at: record.at, data: record.data })],
];

export const step = (story, record) => {
  const node = story.nodes.get(key(record.path)) ?? sprout(story, record);
  MARKS.find(([test]) => test(record))[1](node, record);
  return { ...story };
};

export const dictate = (story) => {
  const records = [];
  const walk = (node) => {
    if (node.timing?.begun != null) records.push({ at: node.timing.begun, path: node.path, track: "open" });
    for (const entry of node.entries) records.push({ at: entry.at, path: node.path, track: entry.track, data: entry.data });
    if (node.fault) {
      const { at, ...data } = node.fault;
      records.push({ at, path: node.path, track: "fault", data });
    }
    node.children.forEach(walk);
    if (node.timing?.sealed != null) records.push({ at: node.timing.sealed, path: node.path, track: "close" });
  };
  story.roots.forEach(walk);
  return records;
};

export const live = (span) => span.pipe.fold(chronicle(span.history), step);

export const duration = (node) =>
  node.timing?.begun != null && node.timing?.sealed != null ? node.timing.sealed - node.timing.begun : null;

export const timing = (story) => {
  const entries = [];
  const walk = (node) => {
    const elapsed = duration(node);
    if (elapsed != null) entries.push(`${node.nature};dur=${elapsed.toFixed(1)}`);
    node.children.forEach(walk);
  };
  story.roots.forEach(walk);
  return entries.join(", ");
};

// subsystems/typology/prototypes/pipe.js   ← whole file
import * as fromm from "../gestalten/fromm/index.js";
import { atom } from "nanostores";
import { Queue } from "./queue.js";

export class Pipe {
  listeners = new Set();

  send(value) {
    for (const listener of this.listeners) listener(value);
  }

  tap(target) {
    const listener = fromm.sink(target).write;
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

  from(...sources) {
    for (const origin of sources)
      (async () => {
        for await (const value of fromm.source(origin).read) this.send(value);
      })();
    return this;
  }

  fold(seed, step) {
    const store = atom(seed);
    this.tap((value) => store.set(step(store.get(), value)));
    return store;
  }

  async *drain(signal) {
    const queue = new Queue();
    const untap = this.tap(queue);
    try {
      yield* queue.drain(signal);
    } finally {
      untap();
    }
  }

  [Symbol.asyncIterator]() {
    return this.drain();
  }
}

// subsystems/typology/prototypes/span.js   ← whole file
import { onSet } from "nanostores";
import { Signature } from "./signature.js";
import { Pipe } from "./pipe.js";

export class Span extends Signature {
  static limit = 1000;

  constructor(signature = null, trace = null) {
    super(signature, trace);
    if (this.pipe) return this;
    this.pipe = new Pipe();
    this.$history = this.pipe.fold([], (held, record) => [...held, record]);
    if (this.trace) this.pipe.tap(this.trace.pipe);
    onSet(this.$history, ({ newValue }) => newValue.splice(0, newValue.length - Span.limit));
  }

  from(trace) {
    return super.from(trace, true);
  }

  get history() {
    return this.$history.get();
  }

  track(name, data) {
    this.pipe.send({ at: performance.now(), path: this.absolute, track: name, data });
    return this;
  }

  fault(error) {
    return this.track("fault", { message: error?.message ?? String(error), code: error?.code ?? null });
  }
}
