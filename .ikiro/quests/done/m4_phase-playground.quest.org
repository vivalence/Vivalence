* Architecture (grounded — the landed contract)
The stall, post-M3, is a pure reactive function over a thread:
#+BEGIN_SRC
  Stall({ source, active, phase, pull, depth })
    source = thread.$buffers          active = terminal.$buffer (the one render pointer)
    phase  = thread.$phase            pull   = thread.pull (AIMED) ; depth = thread.trait.QUEUEING.depth

  phases (read live from thread.$phase):
    inert       no subscription; deliberate null active respected (homepage empty-state)
    manual      settle (auto-focus + advance-on-release); the app drives pulls
    continuous  manual + pull() to keep depth() queued; release refills
    escort      manual; walks a pre-loaded queue, returns to `home` on drain (NO fetch)

  roles:
    DRIVER   writes thread.phase (+ optional .only filter)     c-panel e/f · buffer-apps · (M5: the face)
    TARGET   calls buffer.release()                            13 game modes + spawned
#+END_SRC

Trait shapes (verified, historical):
- =AIMED: { mount: "/emit/<type>/<nature>" }= → installs =thread.pull(args)= =
  =mode.connection.call(mount, merge({thread}, MASKED, args))=.
- =MASKED: { where?, limit?, ... }= → the emit body merged into the pull call.
- =QUEUEING: { depth: N }= → the stall's =depth()=.
The Thread =beforeCreate= hook copies =intent.traits= + deep-merges =intent.trait=
(thread wins per key), so a mode's =dataset.intent= seeds all of this for free.

<script>
  import { atom } from "@vivalence/typology";

  const { terminal, buffer } = $props();
  const $phase = atom.chain(terminal, "$thread", "$phase");
  const $buffers = atom.chain(terminal, "$thread", "$buffers");
  const isSpawned = (b) => b.mode?.slug === "spawned";

  // only the two SELF-MANAGED phases — no continuous/escort here.
  const PHASES = [
    { key: "inert", label: "free", hint: "stall does nothing; each spawned self-evicts" },
    { key: "manual", label: "walk", hint: "stall advances the cursor on done; the app spawns" },
  ];

  async function spawn() {
    await buffer.mode.emit.spawn({ thread: terminal.thread.id });
  }
  function setPhase(key) {
    terminal.stall?.only(isSpawned);
    terminal.thread.phase = key;
    terminal.thread.daemon.entities.thread.updateOne({ id: terminal.thread.id }, { phase: key });
    if (key === "manual")
      terminal.buffer = terminal.thread.$buffers.get().find(isSpawned) ?? buffer;
  }
</script>

<script>
  import { atom } from "@vivalence/typology";

  const { buffer, terminal } = $props();
  const $data = atom.chain(buffer, "$data");

  // report done. under MANUAL the stall advances + the terminal evicts; under INERT the
  // app owns it, so evict here (the self-managed path).
  function done() {
    buffer.release();

    if (terminal.buffer?.id === buffer.id) terminal.buffer = null;
  }
</script>

import { App, Vector, v } from "@vivalence/typology";

export const manifest = {
  type: "playground",
  slug: "dealer",
  name: "Dealer",
  description: "Driver hub: deals card buffers, drives the thread phase, talks.",
  version: "0.1.0",
  traits: ["APPLICATION", "STANDALONE", "EMITTER", "HARNESSED", "CONVERSATIONAL"],
};

// the hub is a control surface — its buffer carries nothing.
export const app = new App("buffer/Dealer.svelte", v.buffer({ data: {} }));

// the finite deck the dealer deals from. running off the end = EXHAUSTED.
const DECK = ["A♠", "K♥", "Q♦", "J♣", "10♠", "9♥", "8♦", "7♣"];

// ── harness · the dealer's persona — powers /oracle AND the conversation ─────────
export const harness = new Vector();
harness.use(async (ctx, next) => {
  ctx.hallucination.add([
    "You are the Dealer in a card-game playground inside vivalence.",
    "You deal cards (render buffers) onto the table (the moat) for the player.",
    "Keep replies short and plain — two or three sentences, no markdown.",
  ]);
  await next();
});

// ── emitter · two ways to deal, both yielding `card` buffers (cross-mode) ─────────
export const emitter = new Vector()
  // deterministic: deal the next N off the finite deck. thread.counter = how many already
  // dealt, so the deck empties → Pool drains EXHAUSTED → continuous stops cleanly.
  .open(
    { nature: "/deal", input: v.object({ count: v.integer({ minimum: 1 }).default(2) }) },
    async (ctx) => {
      const card = ctx.daemon.modes.playground.card;
      const start = ctx.thread?.counter ?? 0;
      for (const face of DECK.slice(start, start + ctx.input.count))
        ctx.pool.add(card.buffer({ data: { face } }));
    },
  )
  // agentic: the dealer picks a themed hand via its harness (mirrors aprende /coach).
  .open(
    {
      nature: "/oracle",
      input: v.object({
        count: v.integer({ minimum: 1 }).default(3),
        focus: v.string().optional(),
        thread: v.string().optional(),
      }),
    },
    async (ctx) => {
      const card = ctx.daemon.modes.playground.card;
      const context = [
        `Pick ${ctx.input.count} cards for a themed hand from this deck: ${DECK.join(" ")}.`,
        ctx.input.focus ? `Theme: ${ctx.input.focus}.` : null,

#+BEGIN_SRC html :tangle ../../registry/playground/switchboard/buffer/Switchboard.svelte :mkdirp yes
<script>
  import { atom } from "@vivalence/typology";

  const { terminal, buffer } = $props();
  const $phase = atom.chain(terminal, "$thread", "$phase");
  const $active = atom.chain(terminal, "$buffer");
  const isCard = (b) => b.mode?.slug === "card";
  const cards = () => (terminal.thread?.$buffers.get() ?? []).filter(isCard);

  const ORDER = ["inert", "manual", "continuous", "escort"];
  let trace = $state([]);

  function setPhase(key) {
    terminal.stall?.only(isCard);
    terminal.thread.phase = key; // live → the stall re-engages on the SAME buffers
    terminal.thread.daemon.entities.thread.updateOne({ id: terminal.thread.id }, { phase: key });
    trace = [`phase → ${key}`, ...trace].slice(0, 6);
  }
  function cyclePhase() {
    setPhase(ORDER[(ORDER.indexOf(terminal.thread.phase) + 1) % ORDER.length]);
  }
  // switch the render cursor independently of the phase.
  function step(delta) {
    const list = cards();
    if (!list.length) return;
    const at = list.findIndex((b) => b.id === terminal.buffer?.id);
    terminal.buffer = list[(at + delta + list.length) % list.length];
    trace = [`cursor → ${terminal.buffer?.data?.face ?? "?"}`, ...trace].slice(0, 6);
  }
  async function feed() {
    await buffer.mode.emit.feed({ thread: terminal.thread.id });
  }
  const cursorAt = () => cards().findIndex((b) => b.id === $active.get?.()?.id) + 1;
</script>
