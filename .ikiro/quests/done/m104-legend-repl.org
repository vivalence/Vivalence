* the law

1. a terminal holds one =Repl= and one =Legend=; the client holds neither; a mode keeps its own =mode.repl= · =mode.keymap= Vectors.
2. both trees are getters built at read, the same four layers in one order: the client's · the hosted modes' · the terminal's · the attached mode's (=terminal.mode=) — later wins. =repl.commands= = =client/commands= (=/shell= · =/quit= only when =paladin.is.tty=) · hosted =mode.repl= · =terminal/commands= · the attached =mode.repl=. =legend.keymap= = =shard.keymap.fromRepl(repl.commands)= first (m99's kept order), then =client/keymap.js= =keymap= · hosted =mode.keymap= · the attached =mode.keymap=. Nothing is stored; nothing is rebuilt on a list change; the runtime's mode list is not read.
3. =terminal.repl.dispatch(line)= is the only way a line runs; =terminal.legend.dispatch(event)= the only way a press enters. A surface forwards to =client.terminal= at the moment of the line or the press; there is always a terminal once a surface paints regions.
4. =legend.dispatch(event)= runs the event through =legend.translation= — a getter, inline: =*= enqueues, the leader key (=meta-v= in a browser off Apple, else =ctrl-v=, by =paladin.is.browser= · =paladin.is.apple=) prevents the default and enqueues =leader= — on every press; the queue's listener walks the held presses over =legend.keymap= once per change: an effect → flush, run it with =ctx = { client, terminal }=, its line → =terminal.repl.dispatch=; a dead end → flush; a prefix → =$chord = shape.strip(node)=.
5. =repl.$catalog= is the palette's listing of =repl.commands=, beside =legend.$chord= — per terminal; a painter follows the active one by =belt.atom.chain(client.$terminal, "repl", "$catalog")= · =(…, "legend", "$chord")=; a switch lands on the other terminal's own (empty) chord and catalog.
6. a runtime harness tool names its terminal: =/terminal/:terminal/repl= · =/metadata/repl= on the client's aperture call THAT terminal's Repl.
7. =KeySignal= spells the nature once (W3C =ctrl alt shift meta=, a printable its character); =scope= asks the literals first, the =*= only when none matched; a pattern never writes on the signal it is asked about (m99).
8. the login screen is the whole screen until the lighthouse authorizes; a logout keeps the username and forgets the password; another user's sign-in flushes the cache and re-seats the active terminal (m102).

import paladin from "@vivalence/paladin";
import { atom } from "nanostores";
import { KeySignal, Queue, Vector, fromm, is, shape, shard, steer } from "@vivalence/typology";
import * as client from "../client/keymap.js";
import { keymap } from "../../gestalten/shard/index.js";

export class Legend {
  presses = new Queue();
  $chord = atom(null);

  constructor(terminal) {
    this.terminal = terminal;
    this.presses.$backlog.listen((presses) => this.traverse(presses));
  }

  get chord() {
    return this.$chord.get();
  }

  get translation() {
    return new Vector()
      .use(shard.context.bind("legend", this))
      .open("*", (ctx) => ctx.legend.presses.enqueue(ctx.signal))
      .open(paladin.is.browser && !paladin.is.apple ? "meta-v" : "ctrl-v", (ctx) => {
        ctx.input.preventDefault();
        ctx.legend.presses.enqueue(new KeySignal("leader"));
      });
  }

  get keymap() {
    return new Vector().graft(
      keymap.fromRepl(this.terminal.repl.commands),
      client.keymap,
      ...fromm.modes(this.terminal.client.modes).flat.map((mode) => mode.keymap),
      ...(is.object(this.terminal.mode) ? [this.terminal.mode.keymap] : []),
    );
  }

  dispatch(event) {
    steer.dispatch.invoke(this.translation, new KeySignal(event), steer.strategy.echo)(event);
  }

  traverse(presses) {
    if (presses.length === 0) return this.$chord.set(null);
    const keys = presses.map((signal) => signal.nature);
    const [effect, carry, steps, node] = steer.dispatch.traverse(this.keymap, new KeySignal(keys));
    if (effect !== null) return this.execute(effect, carry).catch((error) => this.terminal.client.controller.stdout.fault(error));
    if (steps.length < keys.length) return this.flush();
    this.$chord.set(shape.strip(node));
  }

  async execute(effect, carry) {
    this.flush();
    const ctx = { client: this.terminal.client, terminal: this.terminal, output: undefined };
    await carry(ctx, steer.strategy.resolve(effect));
    if (ctx.output !== undefined) await this.terminal.repl.dispatch(ctx.output);
  }

  flush() {
    this.presses.flush();
  }
}
