* The design

** faculty-Vector encoding — =/type/slug= leaves

=faculties = new Vector()=. Each faculty a leaf: =faculties.branch(type).open({nature: slug}, faculty)=.
Yields =/dialogue/opus=, =/dialogue/sonnet=, =/dialogue/haiku=, =/speech/luiza=,
=/verbatim/pt-BR=… — beef's own =harnessed.js:14= spelling (=/type/slug/…=) made literal.
The faculty object is the effect value; =tune= stays on the faculty (read by =nearest=).

Pre-naming tolerance: =open({nature: faculty.slug ?? faculty.type}, faculty)= — before M2,
same-type faculties share nature but are distinct effect entries (Map keys by pattern
IDENTITY, one =new Pattern= per =open=), so =stored(type)= still returns all of them in
insertion order. Naming (M2) only makes the trie legible; the core works without it.

** =stored(type)= — the non-mutating descent (why not =branch()=)

=Vector.branch(type)= is get-OR-CREATE: calling it in =resolve/has= would pollute the tree
with empty branches on every miss. So =stored= does a read-only find (the =revelio= descent):

: stored(type) {
:   const branch = [...this.faculties.trajectories.entries()]
:     .find(([pattern]) => pattern.nature === type)?.[1];
:   return branch ? [...branch.effects.values()] : [];
: }

Returns the per-type faculty array (or =[]=) without mutating. =resolve/has/shard= all fold
through this. (A non-mutating =Vector.find(nature)= could later absorb this, but that
touches typology-holy Vector — kept inline in cortex for a contained recast.)

** =resolve= — descend + nearest (structure unchanged, source swapped)

Body is IDENTICAL to today except =this.table.get(type) || []= becomes =this.stored(type)=.
The derivation fallback, via-filter, and =nearest(eligible, tune ?? [.5×4])= are untouched.
=has/via/derivable= likewise: same logic, =stored()= instead of =table.get/has=.

** =shard.faculties= — data-driven; =FACULTY_TYPES= dies

Today it loops a hardcoded =["dialogue","object","speech","verbatim"]=. The recast drives it
from the Vector's ACTUAL type-branches ∪ =DERIVATIONS= keys (so derived =object= still
opens):

Same output set (only =has()=-true types get branches), the =FACULTY_TYPES= constant gone —

#+begin_src javascript :tangle ../../subsystems/typology/prototypes/cortex.js :mkdirp no
import { array, is, soma, Vector } from "@vivalence/typology";

const DERIVATIONS = {
  object: (cortex, tune) => ({
    type: "object",
    slug: "object",
    channels: { in: ["text"], out: ["object"] },
    tune: tune ?? [0.5, 0.5, 0.5, 0.5],
    via: { render: deriveObjectFallback(cortex, tune) },
  }),
};

export const tiers = {
  frugal: [0.1, 0.3, 0.9, 1.0],
  balanced: [0.4, 0.6, 0.6, 0.6],
  capable: [0.6, 0.8, 0.4, 0.4],
  unleashed: [0.9, 1.0, 0.2, 0.2],
  eager: [0.3, 0.5, 0.5, 0.1],
};

export class Cortex {
  faculties = new Vector();

  extend(supplied) {
    for (const faculty of supplied) {
      if (faculty.tune.length === 3) faculty.tune = [...faculty.tune, 0.5];
      this.faculties.branch(faculty.type).open({ nature: faculty.slug ?? faculty.type }, faculty);
    }
    return this;
  }

  // the revelio descent: read-only lookup of a /type branch's faculty leaves, in
  // insertion order. NOT `branch(type)` — that is get-or-create and would pollute the
  // tree with empty branches on every resolve/has miss.
  stored(type) {
    const branch = [...this.faculties.trajectories.entries()] //
      .find(([pattern]) => pattern.nature === type)?.[1];
    return branch ? [...branch.effects.values()] : [];
  }

  has(type) {
    return this.stored(type).length > 0 || (type in DERIVATIONS && this.derivable(type));
  }

  derivable(type) {
    return type === "object" ? !!this.resolve("dialogue", { via: "render" }) : false;
  }

  resolve(type, { tune, via } = {}) {
    const eligible = this.stored(type).filter((faculty) => !via || faculty.via[via]);
    const native = nearest(eligible, tune ?? [0.5, 0.5, 0.5, 0.5]);
    if (native) return native;

    const derive = DERIVATIONS[type];
    if (!derive || !this.derivable(type)) return native;
    const derived = derive(this, tune);
    return !via || derived.via[via] ? derived : undefined;
  }

  via(type, name, tune) {
    return this.resolve(type, { tune, via: name })?.via[name];
  }

  get shard() {
    const cortex = this;
    return {
      faculties: (vector) => {
        const types = new Set([
          ...[...cortex.faculties.trajectories.keys()].map((pattern) => pattern.nature),
          ...Object.keys(DERIVATIONS),
        ]);

        for (const type of types) {
          if (!cortex.has(type)) continue;

          const branch = vector.branch(type);

          if (cortex.resolve(type, { via: "stream" }))
            branch.open("stream", streamLeaf(cortex, type));

          if (cortex.resolve(type, { via: "render" }))
            branch.open("render", renderLeaf(cortex, type));
        }
      },
    };
  }
}

async function executeTools(tools, parts, thread) {
  const results = [];
  for (const part of parts) {
    if (part.type !== "tool_use") continue;
    const tool = tools[part.name];
    const handler = typeof tool === "function" ? tool : tool?.execute;
    if (!handler) {
      results.push({
        type: "tool_result",
        id: part.id,
        output: { error: `unknown tool: ${part.name}` },
      });
      console.log("@cortex: unknown tool called", { results });
      continue;
    }
    const input =
      typeof part.input === "string"
        ? part.input
          ? JSON.parse(part.input)
          : {}
        : (part.input ?? {});
    const returned = await handler(thread ? { ...input, thread } : input);
    const native =
      is.object(returned) &&
      ("message" in returned || "entities" in returned || "object" in returned);
    const { message, entities, object } = native ? returned : { message: returned };
    results.push({ type: "tool_result", id: part.id, output: message, entities, object });
  }
  return results;
}

function streamLeaf(cortex, type) {
  return async (ctx) => {
    const { tuning, turns, tools, config, thread } = ctx.hallucination;
    ctx.output = (async function* () {
      const stream = cortex.via(type, "stream", tuning);
      const callConfig = { ...config };
      if (Object.keys(tools).length) callConfig.tools = tools;
      let currentTurns = [...turns];
      for (let round = 0; round < 10; round++) {
        let turn = null;
        for await (const packet of await stream(currentTurns, callConfig)) {
          turn = soma.pour(turn, packet);
          yield packet;
        }
        if (turn.meta?.stop !== "tool_use") return;
        const results = await executeTools(tools, turn.parts, thread);
        yield* soma.drain({ role: "user", parts: results });
        currentTurns = [...currentTurns, turn, { role: "user", parts: results }];
      }
    })();
  };
}

function renderLeaf(cortex, type) {
  return async (ctx) => {
    const { tuning, turns, tools, config, thread } = ctx.hallucination;
    const render = cortex.via(type, "render", tuning);
    const callConfig = { ...config, thread };
    if (Object.keys(tools).length) callConfig.tools = tools;
    let currentTurns = [...turns];
    for (let round = 0; round < 10; round++) {
      const turn = await render(currentTurns, callConfig);
      if (turn.meta?.stop !== "tool_use") {
        ctx.output = turn;
        return;
      }
      const results = await executeTools(tools, turn.parts, thread);
      currentTurns = [...currentTurns, turn, { role: "user", parts: results }];
    }
  };
}

function deriveObjectFallback(cortex, tune) {
  return async (turns, config) => {
    const tools = config?.tools ?? {};
    const respond = {
      valence: "Return the final result as structured data.",
      input: config?.schema,
    };
    const render = cortex.via("dialogue", "render", tune);
    let currentTurns = [...turns];
    for (let round = 0; round < 10; round++) {
      const turn = await render(currentTurns, {
        ...config,
        tools: { ...tools, respond },
        tool_choice: { type: "any" },
      });
      const done = turn.parts.find((part) => part.type === "tool_use" && part.name === "respond");
      if (done) {
        const data =
          typeof done.input === "string" ? (done.input ? JSON.parse(done.input) : {}) : done.input;
        return {
          role: "assistant",
          parts: [{ type: "object", data }],
          meta: { stop: "end_turn" },
          object: data,
        };
      }
      const results = await executeTools(tools, turn.parts, config.thread);
      currentTurns = [...currentTurns, turn, { role: "user", parts: results }];
    }
  };
}

export function nearest(faculties, target) {
  if (typeof target === "string") target = tiers[target] ?? [0.5, 0.5, 0.5, 0.5];
  return array.nearest(faculties, target, (faculty) => faculty.tune);
}
