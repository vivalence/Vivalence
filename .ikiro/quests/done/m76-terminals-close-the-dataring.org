* yap
Essentially nothing that is an /entity/ anymore.

Every entity under =typology/entities= is already a =RemoteRepository= over a daemon or lighthouse branch — mode, thread, buffer, turn, activity, intent, literal, symbol, daemon dossier. Terminal was the one locally-owned entity (its own id, its own persistence, its own lifecycle). Once it moves, Anima owns no entities; it owns /state/:

#+BEGIN_SRC md
Anima-owned (after the move)
├─ stores/lighthouse.js      $authority, $identity, $status      — session, cached from wire
├─ stores/terminals.js       $entities, $active                   — projection of lighthouse Terminal rows
├─ stores/bridge/            geometry, panes, gesture, dock math  — viewport/layout, device-bound
├─ stores/box/               device, drivers                      — hardware (mic, keys, screen)
├─ Stall / $settling          per-terminal runtime                — ephemeral, re-derived
└─ prototypes/dictation      in-flight input                      — never persisted
#+END_SRC

That is the right shape for what you said: Anima is a display. The division of ownership becomes clean:

#+BEGIN_SRC md
lighthouse  →  who you are, what you have open, how you like it   (Identity, Terminal, Authority)
daemon      →  what you're working on                              (Thread, Buffer, Mode, …)
anima       →  where it is on this screen, right now               (bridge, box, stall)
#+END_SRC

Two things to decide while you're in there:

- =dock= is split: the /preference/ goes to =Identity.configured.dock=, the /per-terminal/ value to =Terminal.docked=, but =bridge/geometry= (window size, pane split) stays local — it's a property of the glass, not the user. Keep it in localStorage, keyed by device, don't lift it.
- =$settling= and the serialized-reference dance in =app/terminals.js= exist because the terminal held a thread id it couldn't resolve yet. With =seated: { daemon, thread }= living on the lighthouse row the same race exists — the daemon may be unreachable — so the resolver survives, it just reads from the row instead of localStorage.

Net: Anima shrinks from "client with a database" to "client with a cache", and the only truly Anima-owned things are the ones that would be wrong to sync across devices.

const refuse = (ctx, status, code, message) => {

const bearer = (ctx) => {
  const header = ctx.request.headers?.get("authorization");
  const param = ctx.request.url.searchParams?.get("token");
  return header?.startsWith("Bearer ") ? header.split(" ")[1] : param;
};

export class Lighthouse {
  constructor(connection, { users, status, daemon, remote }) {
    this.connection = connection;
    this.users = users;
    this.status = status;
    this.daemon = daemon;
    this.remote = remote;
    this.announced = null;
    this.shard = {
      authenticate: (ctx, next) => this.identify(ctx, next),
      authorize: (ctx, next) => this.identify(ctx, () => this.admit(ctx, next)),
    };
    status.$transient.subscribe(() => this.beacon().catch(() => {}));
  }

  beacon() {
    const outcome = Promise.resolve(this.announced).then(() => this.announce());
    this.announced = outcome.catch((refused) => refused);
    return outcome;
  }

  announce() {
    return this.connection.call("/daemons/announce", { slug: this.daemon, remote: this.remote, reported: this.status.reflection.code });
  }

  async identify(ctx, next) {
    if (ctx.lighthouse) return await next();
    const token = bearer(ctx);
    if (!token) return refuse(ctx, 401, "MISSING_TOKEN", "Authorization header or token param required");
    try {
      ctx.lighthouse = await this.authenticate(token);
    } catch (error) {
      return refuse(ctx, 401, "INVALID_TOKEN", error.message);
    }
    await next();
  }

  async admit(ctx, next) {
    try {
      ctx.user = await ctx.lighthouse.getUser();
    } catch {
      return refuse(ctx, 500, "USER_LOOKUP_FAILED", "Failed to retrieve user");
    }
    if (!ctx.user) return refuse(ctx, 401, "USER_NOT_FOUND", "no userspace in this daemon — /userspace/handshake first");
    await next();
  }

  async verify(token) {
    const response = await this.connection.fetch("/auth/verify", { access: token }, { method: "POST" });
    if (response.ok) return response.body;
    return response.body?.status === "ERROR"
      ? response.body
      : { status: "ERROR", error: { code: "VERIFY_FAILED", message: "Verification failed" } };
  }

  async authenticate(token) {
    if (!token) throw new Error("Token required");
    const result = await this.verify(token);
    if (result.status === "ERROR") {
      const error = new Error(result.error.message);
      error.code = result.error.code;
      throw error;
    }
    const { identity } = result;
    if (!identity?.id) throw new Error("Identity not found in token");
    return {
      getUser: () => this.users.findOne({ id: identity.id }),
      enroll: async () => {
        const granted = await this.connection.fetch("/auth/enroll", { access: token, daemon: this.daemon }, { method: "POST" });
        if (granted.body?.status !== "SUCCESS") throw new Error(granted.body?.error?.message ?? "Enrollment refused by the lighthouse");
        await this.users.upsert({ id: identity.id, roles: ["USER"], config: {} }, { onConflictAction: "ignore" });
        return this.users.findOne({ id: identity.id });
      },
    };
  }
}

import * as argon2 from "argon2";
import * as jose from "jose";

const ACCESS_TTL = "1d";
const REFRESH_TTL = "180d";

const refuse = (ctx, code, message) => {
  ctx.response.status = 401;
  ctx.response.body = { status: "ERROR", error: { code, message } };
};

export function signer(ctx) {
  const key = new TextEncoder().encode(ctx.mask.secrets.jwt);
  const sign = (payload, expiration) =>
    new jose.SignJWT(payload).setProtectedHeader({ alg: "HS256" }).setIssuedAt().setExpirationTime(expiration).sign(key);
  return {
    access: (identity) => sign({ id: identity.id, type: "access" }, ACCESS_TTL),
    refresh: (identity) => sign({ id: identity.id, type: "refresh" }, REFRESH_TTL),
    verify: async (token, type) => {
      const payload = await jose.jwtVerify(token, key).then((result) => result.payload).catch(() => null);
      return payload?.type === type ? payload : null;
    },
  };
}

export async function identify(ctx, { username, password }) {
  if (!username || !password) return null;
  const identity = await ctx.entities.identity
    .createQueryBuilder("i")
    .where(`json_extract(i.authentication, '$.credentials.username') = ?`, [username])
    .getSingleResult();
  if (!identity?.authentication?.credentials?.password) return null;
  const valid = await argon2.verify(identity.authentication.credentials.password, password);
  return valid ? identity : null;
}

export async function bearer(ctx, refresh) {
  const payload = await signer(ctx).verify(refresh, "refresh");
  if (!payload) return null;
  const identity = await ctx.entities.identity.findOne({ id: payload.id });
  return identity?.authentication?.tokens?.refresh === refresh ? identity : null;
}

export function authenticate() {
  return async (ctx, next) => {
    const header = ctx.request.headers?.get("authorization");
    const token = header?.startsWith("Bearer ") ? header.split(" ")[1] : ctx.request.url.searchParams?.get("token");
    if (!token) return refuse(ctx, "MISSING_TOKEN", "Authorization header or token param required");
    const payload = await signer(ctx).verify(token, "access");
    if (!payload) return refuse(ctx, "INVALID_TOKEN", "Invalid or expired token");
    ctx.identity = await ctx.entities.identity.findOne({ id: payload.id });
    if (!ctx.identity) return refuse(ctx, "IDENTITY_NOT_FOUND", "Identity not found");
    await next();
  };
}

import { Aperture, Connection, Datamap, Url, Vector, shape, shard } from "@vivalence/typology";
import { lighthouse } from "./fixtures.js";
import { accio } from "./registry.js";

export async function create() {
  const { orm, em, repos, fixtures } = await lighthouse.seed();
  const { attachment, manifest } = await accio("@commons/lighthouse/multiplayer");
  const datamap = new Datamap(orm);
  const mask = { secrets: { jwt: "scenario-secret-scenario-secret-scenario" }, statics: {}, attachment };
  const service = {
    manifest,
    attachment: new Aperture(),
    controller: { status: "RUNNING" },
    datamap,
    entities: datamap.entities,
  };

  service.twitch = new Vector();
  service.twitch.branch("/after").use(shard.datamap.detached(datamap));
  datamap.registerSubscriber(shape.subscriber(service.twitch));
  service.attachment.use(shard.datamap.inject(datamap));

  service.attachment
    .open("/status", () => service.controller.status)
    .open("/manifest", () => service.manifest)
    .use(shard.context.bind("service", service))
    .use(shard.context.bind("mask", mask));
  service.attachment.graft(mask.attachment);

  const handler = shape.http(service.attachment);
  const connect = (token) => {
    const connection = new Connection(new Url("http://lighthouse"), shard.transmitter.inline(handler));
    if (token) {
      connection.use(async (ctx, next) => {
        ctx.request.headers.set("authorization", `Bearer ${token}`);
        await next();
      });
    }
    return connection;
  };
  const conn = connect();

  const signup = async (username) => {
    const body = await conn.call("/auth/signup", { username, password: `pw-${username}` });
    return { identity: body.identity, access: body.authority.access, refresh: body.authority.refresh };
  };

  const seedDaemon = async (slug, identities) => {
    const daemon = datamap.entities.daemon.create({ slug, remote: `http://${slug}:2501`, reported: "RUNNING" });
    for (const id of identities) {
      const identity = await datamap.entities.identity.findOneOrFail({ id });
      identity.daemons.add(daemon);
    }
    await datamap.entities.em.flush();
    return daemon;
  };

  const close = () => datamap.close();

  return { conn, orm, em, repos, fixtures, datamap, service, mask, connect, signup, seedDaemon, close };
}

  index = 0;

  trait = {};

  $settling = atom(null);

  }

  }
  get settling() {

  }

  toJSON() {
    const kept = (key) => !key.startsWith("$") && !["stall", "repository", "docking", "derivedDock"].includes(key);
    return Object.fromEntries(Object.entries(super.toJSON()).filter(([key]) => kept(key)));
  }

  write(data) {

      .updateOne({ id: this.id }, data)
      .catch((error) => logger.entry(`terminals/write/${this.id}`).fault(error));
  }

  seat(thread) {
    return this.write({ daemon: thread?.daemon?.id ?? null, thread: thread?.id ?? null });
  }

    this.docking = setTimeout(() => {
      const value = this.$dock.get();
      if (same(value, this.trait.DOCKED) || same(value, this.derivedDock)) return;
      const traits = this.traits.includes("DOCKED") ? this.traits : [...this.traits, "DOCKED"];
      this.write({ traits, trait: { ...this.trait, DOCKED: value } });
    }, DOCK_DEBOUNCE_MS);
  }

}

export function mountSeat(terminal, identity) {
  let bound = null;
  let unbind = [];
  terminal.$dock.listen(() => terminal.persistDock());

  const bind = (daemon) => {
    if (daemon === bound) return;
    unbind.forEach((off) => off());
    bound = daemon;
    unbind = daemon
      ? [daemon.entities?.thread?.$entities.listen(derive), daemon.status?.$transient?.listen(derive)].filter(Boolean)
      : [];
  };

  const derive = () => {
    const docked = terminal.trait.DOCKED ?? identity?.trait?.DOCKED;
    terminal.derivedDock = docked ?? null;
    if (docked && !same(docked, terminal.$dock.get())) terminal.$dock.set({ ...docked });

    const daemon = terminal.daemon;
    bind(daemon);
    const id = terminal.thread;
    if (!id) {
      terminal.$settling.set(null);
      if (terminal.$thread.get()) terminal.$thread.set(null);
      return;
    }
    const store = daemon?.entities?.thread;
    if (!daemon?.status?.is?.("healthy") || !store) {
      terminal.$settling.set({ thread: id });
      if (terminal.$thread.get()) terminal.$thread.set(null);
      return;
    }
    const found = store.findOneLocal({ id });
    terminal.$settling.set(null);
    if (found) {
      if (terminal.$thread.get() !== found) terminal.$thread.set(found);
      return;
    }
    if (terminal.$thread.get()) terminal.$thread.set(null);
    logger.entry("terminal/unseat").note({ terminal: terminal.id, thread: id, daemon: daemon.slug });
    terminal.write({ daemon: null, thread: null });
  };

  const offRows = terminal.repository.$entities.listen(derive);
  derive();
  return () => {
    offRows();
    unbind.forEach((off) => off());
  };
}

export function mountStall(terminal) {
  let threadVigil = null;
  let bufferVigil = null;

  const vigil = (entity, repository, clear) =>
    repository?.$entities?.listen((entities) => {
      if (!entities.some((candidate) => candidate.id === entity.id)) clear();
    }) ?? null;

  terminal.$buffer.subscribe((buffer) => {
    bufferVigil?.();
    bufferVigil = buffer ? vigil(buffer, terminal.$thread.get()?.daemon?.entities?.buffer, () => terminal.$buffer.set(null)) : null;
  });

  let seated = null;
  terminal.$thread.subscribe((thread) => {
    if ((thread?.id ?? null) !== seated) terminal.$buffer.set(null);
    seated = thread?.id ?? null;
    threadVigil?.();
    threadVigil = thread ? vigil(thread, thread.daemon?.entities?.thread, () => terminal.$thread.set(null)) : null;
    terminal.stall?.deactivate();
    terminal.stall =
      thread &&
      Stall({
        source: thread.$buffers,
        active: terminal.$buffer,
        phase: thread.$phase,
        pull: () => pull(thread, { blacklist: new Blacklist().absorb(thread.$buffers.get()) }),
        depth: () => depth(thread),
      });
    terminal.stall?.on.release((buffer) => {
      thread.daemon.entities.buffer.drop(buffer.id);
      thread.daemon.entities.buffer
        .removeOne({ id: buffer.id })
        .catch((error) => logger.entry(`buffers/release/${buffer.id}`).fault(error));
    });
  });
}

class DaemonDouble {
  status = { code: "mounting", is: (code) => this.status.code === code, $transient: atom(0) };
  entities = { thread: { $entities: atom([]), findOneLocal: ({ id }) => this.entities.thread.$entities.get().find((t) => t.id === id) ?? null } };
  healthy(code = "healthy") {
    this.status.code = code;
    this.status.$transient.set(this.status.$transient.get() + 1);
  }
}
const DaemonDossierDouble = {
  name: "daemon",
  kind: () => DaemonDouble,
  repository: (schema, dataspace) => new RemoteRepository(schema.kind()).connect(dataspace.connection.branch("/entities/daemon")),
};

const threadDouble = (id, daemon) => ({ id, daemon, $buffers: atom([]), $phase: atom("manual"), trait: { QUEUEING: { depth: 1 } } });
const tick = (ms = 0) => new Promise((resolve) => setTimeout(resolve, ms));
const settled = async () => { for (let i = 0; i < 12; i++) await tick(0); };
const until = async (predicate, ms = 1500) => {
  const start = Date.now();
  while (!predicate()) {
    if (Date.now() - start > ms) throw new Error("until: timed out");
    await tick(5);
  }
};

async function client(house, username, access = null) {
  let identity;
  if (!access) ({ identity, access } = await house.signup(username));
  else identity = (await house.connect(access).call("/auth/verify", { access })).identity;
  const lighthouse = { $identity: map(identity), $daemons: atom([]), get identity() { return this.$identity.get(); } };
  const connection = house.connect(access);
  const dataspace = new Dataspace({
    entities: [DaemonDossierDouble, TerminalDossier],
    connection,
    seed: (vector) => vector.use(shard.context.bind("lighthouse", lighthouse)),
  });
  await dataspace.init();
  await dataspace.populate(["daemon", "terminal"]);
  const terminals = new Terminals(dataspace.terminal);
  return { identity, access, lighthouse, connection, dataspace, terminals };
}
