** The shape (after beef's pass)

A STANDALONE, turn-free game. Three structural decisions his comments forced:

1. *Two pools, not one* — the riddle SUBJECT (symbol-filtered words a riddle is
   built FROM) is distinct from the learner LEVEL (a broad sample of the learner's
   memory, to pitch difficulty). Separate feeds, separate persona blocks.
2. *History replay, not recompile* — the emitter seeds =buffer.data.history[0]= with
   the per-riddle system context (answer + hint + reply rules) and =[1]= with the
   riddle. The aperture REPLAYS history and appends the new message; it never
   reloads vocabulary/riddle/answer.
3. *Reply ≠ evaluation* — =/assistant/message= produces the snappy in-character
   reply; grading (the memory =GRADE= signal) is a SEPARATE, accurate, async job
   fired only when a riddle is newly resolved.

  import { Vector } from "@vivalence/typology";
  import { identity, subject, learnerStatus, composing, opening } from "./persona.js";
  import { EMITTER_INPUT, EMITTER_OUTPUT } from "./types.js";

  const CANDIDATE_POOL_SIZE_MULTIPLIER = 3;

  export const emitter = new Vector().open(
    { nature: "/riddle/fromSymbol", input: EMITTER_INPUT },
    async (ctx) => {
        const hallucination = ctx.daemon.cortex.hallucination({tune:"eager"});
      // @beef "NEVER the language() one-off — just declare using ctx.deep.nested.lookups." @claude: inlined.
      const language = ctx.daemon.statics.language;
      const count = ctx.input.numberOfRiddles;

      // subject pool — the symbol-filtered words the riddles are built FROM
      const pool = await ctx.daemon.entities.literal.feed(
        { ontology: "word", symbols: ctx.input.symbols },
        { limit: count * CANDIDATE_POOL_SIZE_MULTIPLIER, blacklist: ctx.input.blacklist, populate: ["memories", "memories.strength", "symbols"] },
      );

      // learner level — a broad sample of the learner's memory, NOT symbol-filtered, to pitch difficulty
      const level = await ctx.daemon.entities.literal.feed(
        { ontology: "word" },
        { limit: ctx.input.levelPoolSize, blacklist: ctx.input.blacklist, populate: ["memories", "memories.strength"] },
      );

      // GENERATION context — identity + what to build from + who we build for + the build order.
      // instructions defaults to "" (EMITTER_INPUT) so .add drops it when absent — no inline ternary.
        // @beef this is the old interface.
      hallucination
        .add(identity(language), subject(pool), learnerStatus(level), composing({ learning: language.learning, count }), ctx.input.instructions)
        // .tune("capable") // @beef "a bit more pep on initial creation" — @claude: capable (was frugal); the reply path stays snappy.
        .configure({ schema: EMITTER_OUTPUT }); // @beef rename schema→output stack-wide — @claude: cross-cutting typology task, see Forks; kept `schema` (current API).

      // @beef sample-data harvest for the level-reactive examples (W2)
      console.log("[riddler/cast]", { input: ctx.input, pool, level, hallucination: ctx.hallucination });
      const render = await hallucination.object.render();
      console.log("[riddler/render]", { input: ctx.input, render });
      const { object } = render;

      for (const cast of (object?.riddles ?? []).slice(0, count)) {
        const literals = pool.filter((literal) => cast.literals?.includes(literal.slug));
        ctx.pool.add(
          ctx.mode.app.buffer({
            data: {
              riddle: cast.riddle,
              answer: cast.answer,
              hint: cast.hint, // @beef missing hint — added here + on the app + in EMITTER_OUTPUT
  	      // @beef maybe expectation?i:cast.expectation = [{role:"xy",text:"bc"},....]
              // @beef "seed history so the aperture doesn't recompile." @claude: history[0]=per-riddle system
              //   seed (private answer/hint + reply rules), history[1]=the riddle. NOTE: reconstructed per
              //   riddle — the emitter batches N riddles in one render, so this is the per-buffer equivalent,
              //   not the verbatim batched i/o. That is what makes the aperture's no-reload claim hold.
              history: [
                { index: 0, role: "system", text: opening(language, cast) },
                { index: 1, role: "riddler", text: cast.riddle },
              ],
            },
            symbols: ctx.input.symbols, // @beef "symbols literals" — both on the buffer now
            literals: literals.length ? literals : pool,
          }),
        );
      }
    },
  );

  import { Vector } from "@vivalence/typology";
  import { judging } from "./persona.js";
  import { ASSISTANT_INPUT, ASSISTANT_OUTPUT, RESOLVE_INPUT, RESOLVE_OUTPUT, GRADE_OUTPUT } from "./types.js";

  const toTurn = (entry) => ({
    role: entry.role === "challenger" ? "user" : entry.role === "riddler" ? "assistant" : "system",
    parts: [{ type: "text", text: entry.text }],
  });

  export const aperture = new Vector()
    .open(
      { nature: "/assistant/message", input: ASSISTANT_INPUT, output: ASSISTANT_OUTPUT },
      async (ctx) => {
       const hallucination = ctx.daemon.hallucination({tune:"eager"})
        // @beef trailing fork: auto-transport ctx.buffer + ctx.thread into every ctx (aperture/emitter/tool). Parked — backend.
        const buffer = await ctx.daemon.entities.buffer.findOne({ id: ctx.input.buffer });

        // @beef "no need to reload vocabulary — the history contains the first hallucination i/o. validate!"
        // @claude VALIDATED (per-buffer): history[0] is the seeded system context (identity + this riddle's
        // answer/hint + reply rules), history[1] the riddle. We REPLAY history as turns and append the new
        // message — riddle, answer, and disposition are all present, nothing is recompiled.
        // @beef i dont like this .add().tune().configure() interface — @claude: a one-call
        //   `render({ turns, tune, output })` nicer-interface is the cleaner shape; it's a hallucination-
        //   prototype change (post-quest). Parked in Forks; builder used here since it's what's provided.
        const history = buffer.data.history;

        hallucination
          .add(...history.map(toTurn), { role: "user", parts: [{ type: "text", text: ctx.input.message }] })
          // .tune("eager") // snappy — the reply is on the critical path
          .configure({ schema: ASSISTANT_OUTPUT });

        const { object: reply = {} } = await hallucination.object.render();

        const base = history.length;
        buffer.data = {
          ...buffer.data,
          history: [
            ...history,
            { index: base, role: "challenger", text: ctx.input.message },
            { index: base + 1, role: "riddler", text: reply.message },
          ],
        };

        // @beef "judging is a separate hallucinations job than evaluation. trigger async on newly-resolved."
        const wasResolved = buffer.status === "DONE";
        if (reply.resolved && !wasResolved) {
          buffer.status = "DONE";
          evaluate(ctx, buffer); // fire-and-continue — the accurate grade must not delay the reply
        }
        await ctx.daemon.entities.em.flush();

        return {
          message: reply.message,
          taunt: reply.taunt ?? "",
          hint: reply.hint ?? "",
          resolvable: !!reply.resolvable && !wasResolved,
          resolved: !!reply.resolved,
        };
      },
    )
    .open(
      { nature: "/assistant/resolve", input: RESOLVE_INPUT, output: RESOLVE_OUTPUT },
      async (ctx) => {
        const buffer = await ctx.daemon.entities.buffer.findOne({ id: ctx.input.buffer });
        buffer.status = "DONE";
        await ctx.daemon.entities.em.flush();
        await ctx.daemon.entities.buffer.removeOne({ id: ctx.input.buffer });
        return { resolved: true };
      },
    );

  // EVALUATION — the async, accurate grade. Separate hallucination (its OWN request), judging() rubric,
  // GRADE enum → writes the memory signal for the riddle's literals.
  // @claude seams (backend, flagged): (a) it needs a FRESH hallucination, not ctx's (which just replied);
  //   (b) the fire-and-forget trigger (queued job vs detached task) is the daemon's call; (c) the SRS
  //   write (grade → literal memory) reuses the same signal path aprende grades through.
  async function evaluate(ctx, buffer) {
    const language = ctx.daemon.statics.language;
    const hallucination = ctx.daemon.hallucinate({tune:"capable"}); // FRESH request — seam
    hallucination // also uses the old interface! 
      .add(...buffer.data.history.map(toTurn), judging(language))
      .tune("capable")
      .configure({ schema: GRADE_OUTPUT }); // @beef .schema should be .output!
    const render = await hallucination.object.render();
    // grade.grade → memory signal on buffer.literals (SRS update) — seam
      // @beef log render. snapshot to paladin.scribe.log() // or scribe.snapshot or scribe.telemetry or or or
      const { object: grade = {} } = render
    return grade;
  }

- *v-API surface* (=schematics/v.js:88=, =enhance= proxy L8): fluent methods are ONLY
  =desc=/=descr= (aliases, set =description=), =optional=, =default=, =$id=, plus the
  runtime ops (=check/create/clean/errors/compile/fill/cast=). There is NO =.min=/=.max=.
  =minimum/maximum/pattern/format= etc. are typebox constructor opts → pass the options
  object. Uniform rule adopted in types.js: *every constraint in the options object*.
- *v.rel* — beef wants both scalars named. Full proposal in [[*v.rel — explicit id + slug scalars]].
