** the shape

One factory, written against the Wafer interface, no transport in it:

#+BEGIN_SRC js
import { Vector } from "@vivalence/typology";

const reflect = (die) => ({
  status: die.status.reflection,
  processes: (die.good?.processes ?? []).map((process) => ({
    process: process.slug,
    pid: process.child?.pid ?? null,
    status: process.status.reflection,
  })),
});

export function control(die) {
  return new Vector()
    .open("/status", () => reflect(die))
    .open("/up", async () => {
      await die.integrate();
      return reflect(die);
    })
    .open("/down", async () => {
      await die.disintegrate();
      return reflect(die);
    })
    .open("/restart", async () => {
      await die.disintegrate();
      await die.populate();
      await die.resolve();
      await die.integrate();
      return reflect(die);
    })
    .open("/logs", async (input) => die.ledger.log(die.instance).tail(input?.process, input?.lines ?? 50));
}
#+END_SRC
