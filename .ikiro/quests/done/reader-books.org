import { unzipSync } from "npm:fflate@0.8.3";

export const epub = (bytes) => {
  const archive = unzipSync(bytes);
  const read = (path) => parseHTML(new TextDecoder().decode(archive[path])).document;
  const opf = opened(archive);
  const titles = contents(archive, opf);
  return opf.spine.map((href) => {
    const document = read(join(opf.base, href));
    const heading = document.querySelector("h1, h2, h3")?.textContent.trim() ?? href;
    return { title: titles[href] ?? heading, root: document.body };
  });
};

#+BEGIN_SRC js
async function* crawled(ctx, contents) {
  const kept = contents.filter((entry) => entry.kept);
  const chapters = [];
  for (const [index, entry] of kept.entries()) {
    const response = await fetch(entry.href, { redirect: "follow" }).catch((error) => error);
    if (!(response instanceof Response) || !response.ok) {
      yield { step: "fetch", value: `page ${index + 1}/${kept.length} · ${entry.title} · skipped` };
      continue;
    }
    chapters.push(...page(await response.bytes(), response.headers.get("content-type"), response.url));
    yield { step: "fetch", value: `page ${index + 1}/${kept.length} · ${entry.title}` };
  }
  return chapters;
}
