// systems/ghost/client/keymap.js
import { Dropped, Press, ReplSignal, Vector, fromm, is, shape, steer } from "@vivalence/typology";
import * as shard from "../gestalten/shard/index.js";

const leader = new Vector()
  .open({ nature: "leader", valence: "the catalog — every repl verb, narrowed as you type. Example: ctrl-v ctrl-v" }, (ctx) => {
    ctx.client.$catalog.set(shape.strip(shard.repl.compose(ctx.client, ctx.client.active)));
  })
  .open({ nature: "`", valence: "toggle the renderer console. Example: ctrl-v `" }, (ctx) => ctx.client.surface.console());

export const keymap = new Vector().open({ nature: "esc", valence: "drop the chord and the catalog. Example: esc" }, (ctx) => {
  ctx.client.$chord.set(null);
  ctx.client.$catalog.set(null);
});
keymap.branch({ nature: "leader", valence: "the leader — ctrl-v on a tty" }, leader);

const line = (steps) => "/" + steps.map((step) => step.nature).join("/");

export const harvest = (composed) =>
  steer.trie.rollup(composed, (carry, effect, steps) => (is.string(steps.at(-1).chord) ? [steps.at(-1).chord, line(steps)] : null))
    .filter((entry) => entry.fn !== null)
    .reduce((keymap, { fn: [chord, target] }) => keymap.open({ nature: chord, valence: target }, target), new Vector());

export const compose = (client, terminal) =>
  new Vector().graft(
    harvest(shard.repl.compose(client, terminal)),
    ...fromm.modes(client.modes).flat.map((mode) => mode.keymap),
    client.keymap,
    ...(terminal === null ? [] : [terminal.keymap, ...(is.object(terminal.mode) ? [terminal.mode.keymap] : [])]),
  );

const FAULTS = [
  [(error) => error instanceof Dropped, () => undefined],
  [() => true, (error, client) => client.controller.stdout.fault(error)],
];

export const navigate = async (terminal, first, more) => {
  const client = terminal.client;
  const composed = terminal.keymap;
  if (!composed.trie.has(first.key)) return false;
  let pending = first;
  const next = async (patterns) => {
    client.$chord.set(patterns.map((pattern) => ({ key: pattern.nature, valence: pattern.valence })));
    const press = pending ?? (await more());
    pending = null;
    if (press.key === "esc" && patterns !== composed.patterns) throw new Dropped(patterns);
    return new Press(press.key);
  };
  const ctx = { client, dispatch: (text) => client.dispatch(text), output: undefined };
  try {
    const [effect, carry] = await steer.dispatch.walk(composed, next);
    await carry(ctx, steer.strategy.resolve(effect));
    if (ctx.output !== undefined) await client.dispatch(new ReplSignal(ctx.output));
  } catch (error) {
    FAULTS.find(([test]) => test(error))[1](error, client);
  } finally {
    client.$chord.set(null);
  }
  return true;
};

// systems/anima/src/surface/keys.js
import { Queue } from "@vivalence/typology";

const MAC = /Mac|iPhone|iPad/.test(globalThis.navigator.platform);
export const LEADER = MAC ? "ctrl-v" : "meta-v";

const NAMES = { Escape: "esc", Enter: "enter", ArrowLeft: "left", ArrowRight: "right", ArrowUp: "up", ArrowDown: "down" };
const named = (event) => (Object.hasOwn(NAMES, event.key) ? NAMES[event.key] : event.key.toLowerCase());

const CHORDS = [
  [(event) => (MAC ? event.ctrlKey : event.metaKey) && !event.shiftKey && event.key === "v", () => "leader"],
  [(event) => event.ctrlKey, (event) => `ctrl-${named(event)}`],
  [(event) => event.metaKey, (event) => `meta-${named(event)}`],
  [() => true, named],
];

export const press = (event) => ({ key: CHORDS.find(([test]) => test(event))[1](event) });

export const keypresses = (window) => {
  const queue = new Queue();
  const listener = (event) => queue.enqueue(press(event));
  window.addEventListener("keydown", listener);
  return Object.assign(queue, { close: () => { window.removeEventListener("keydown", listener); Queue.prototype.close.call(queue); } });
};
