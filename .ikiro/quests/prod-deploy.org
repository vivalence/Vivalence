#+BEGIN_SRC sh
# on R000 — the hash from deno (argon2 · uuid in the import map), the row from the sqlite CLI (apk); single-quoted SQL — sqlite reads no backslash
C=$(docker ps --format '{{.Names}}' | grep '^runtime-')
docker exec -w /viva/repository $C deno eval '
import * as argon2 from "argon2";
import { v7 } from "uuid";
const [username, password] = Deno.args;
const authentication = { provider: "password", credentials: { username, password: await argon2.hash(password) } };
const json = JSON.stringify(authentication).replaceAll("'"'"'", "'"''"'");
const user = v7();
console.log(`insert into User (id, created_at, updated_at, slug, authentication, traits, trait) values ('"'"'${user}'"'"', ${Date.now()}, ${Date.now()}, '"'"'${username}'"'"', '"'"'${json}'"'"', '"'"'[]'"'"', '"'"'{}'"'"');`);
console.log(`insert into Terminal (id, user_id) values ('"'"'${v7()}'"'"', '"'"'${user}'"'"');`);
' -- beef '<pw>' > /tmp/user.sql
docker cp /tmp/user.sql $C:/tmp/user.sql
docker exec $C sh -c 'sqlite3 /viva/mountpoint/prod.viva.db < /tmp/user.sql'
docker exec $C sqlite3 /viva/mountpoint/prod.viva.db "select id, slug, json_extract(authentication, '$.credentials.username') from User"
curl -s https://runtime.s4y.vivalence.com/lighthouse/login -H 'content-type: application/json' -d '{"username":"beef","password":"<pw>"}'
