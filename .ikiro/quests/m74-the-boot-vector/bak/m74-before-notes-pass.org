What every process does, whatever its type.

#+BEGIN_SRC js
// systems/runtime/lifecycle/process.js
import { control, steer } from "@vivalence/typology";

export const seal = async (die, next) => {
  try {
    await next();
  } catch (error) {
    die.controller.stdout.fault(error);
  } finally {
    die.controller.stdout.close();
  }
};

export const exit = async (die, next) => {
  await next();
  Deno.exit(die.controller.status.is("STOPPED") ? 0 : 1);
};

export const children = ({ run, masks, repository, scope }) => async (die, next) => { // @beef no i dont like this abstraction. wrong place, wrong percpective
  const processes = await scope(die)(() =>
    Promise.all(
      masks(die).map((mask) => {
        const child = { controller: die.controller.branch(mask.mount.absolute), mask };
        const execution = steer.dispatch.execute(run, child);
        return repository(die).controlled({ id: mask.manifest.slug, mask, controller: child.controller, execution });
      }),
    ),
  );

  try {
    await Promise.all(processes.map((process) => control.controlled(process.controller)));
    await next();
  } finally {
    await Promise.all(processes.map((process) => process.controller.kill(process.controller.status.is("IDLE") ? "SIGKILL" : "SIGTERM")));
    await Promise.all(processes.map((process) => process.execution));
  }
};
#+END_SRC

- =seal= first :: fault and close run inside the carry, so =exit= reads the settled status: 0 on a clean stop, 1 on a fault.
- the die is uniform :: =controller · mask=, minted here and once in =run.js=.
- the controller branches along =mask.mount= :: the seat paladin computes (=/daemon/chess= · =/mode/game/board= · =/attached/process/service/lighthouse/multiplayer=).
- SIGKILL to a child still IDLE :: the machine has no =stop= from IDLE; SIGTERM to a booting child is dropped and the parent waits forever.

import { types } from "@mikro-orm/core";
import { VirtualEntity, VirtualRepository, VirtualSchema } from "@vivalence/typology/entities";

export class ProcessEntity extends VirtualEntity {
  mask = {};
  #controller;
  #execution;

  set controller(controller) {
    if (this.#controller) throw new Error("a process is controlled once");
    this.#controller = controller;
  }
  get controller() {
    return this.#controller;
  }
  set execution(execution) {
    if (this.#execution) throw new Error("a process is executed once");
    this.#execution = execution;
  }
  get execution() {
    return this.#execution;
  }
}

export const ProcessSchema = new VirtualSchema({
  class: ProcessEntity,
  repository: () => ProcessRepository,
  properties: { mask: { type: types.json } },
});

export class ProcessRepository extends VirtualRepository {
  async controlled({ controller, execution, ...data }) {
    const process = (await this.findOne({ id: data.id })) ?? (await this.create(data));
    process.controller = controller; // @Beef any reason were keeping this out of mikro?? mirko offer features for this. lets use 'just mikro'!
    process.execution = execution;
    execution.then(() => this.remove({ id: process.id }));
    return process;
  }
}
