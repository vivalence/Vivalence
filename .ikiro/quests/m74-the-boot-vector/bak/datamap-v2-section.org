#+BEGIN_SRC js
// commons/datamaps/libsql/libsql.viva.js
import { RequestContext } from "@mikro-orm/core";
import { defineConfig, MikroORM } from "@mikro-orm/sqlite";
import { Migrator } from "@mikro-orm/migrations";

export const manifest = { type: "datamap", slug: "libsql", name: "libsql" };

export const config = (datamap, options = {}) => {
  const seat = datamap.mountpoint?.branch(datamap.statics.db.file).absolute;
  return defineConfig({
    dbName: seat ?? ":memory:",
    loadStrategy: "balanced",
    ...(datamap.statics?.db?.file && { contextName: datamap.statics.db.file }),
    ...(seat && {
      extensions: [Migrator],
      migrations: { tableName: "_mikro_migrations", path: datamap.mountpoint.branch("migrations").absolute, transactional: false },
    }),
    ...options,
  });
};

export const provider = async (datamap, options) => {
  const orm = await MikroORM.init(config(datamap, options));
  if (orm.config.get("dbName") === ":memory:") await orm.schema.createSchema();
  else {
    const migrator = orm.getMigrator();
    if (await migrator.checkMigrationNeeded()) await migrator.createMigration();
    if ((await migrator.getPendingMigrations()).length) await migrator.up();
  }
  const entities = { em: orm.em };
  for (const meta of Object.values(orm.getMetadata().getAll())) {
    if (meta.abstract || meta.pivotTable || meta.embeddable) continue;
    entities[meta.className.toLowerCase().replace("entity", "")] = orm.em.getRepository(meta.class);
  }
  return {
    orm,
    entities,
    scope: (task) => RequestContext.create(orm.em, task),
    bind: (name, resolve) => async (ctx, next) => {
      RequestContext.getEntityManager(orm.em.name)?.setFilterParams(name, resolve(ctx));
      await next();
    },
    carry: () => {
      const context = RequestContext.currentRequestContext();
      return (task) => (context ? RequestContext.storage.run(context, task) : task());
    },
  };
};
#+END_SRC

- subscribers do not care where the rows live :: mikro's =EventManager= runs above the driver; a virtual repository dispatches the same events by hand (=daemon/entities/base/VirtualEntity.ts:31=). The runtime passes none it does not need; the twitch on its process rows is the seat a differ, a watchdog or =/processes= subscribe hangs on.
- =disintegrate= is cut :: beef: /"seems odd. not sure. your call. maybe cut?"/ Mikro's word: =datamap.orm.close()=.
- the repositories are derived :: from the ORM's metadata, by the rule =shard.datamap.strip= already uses (=subsystems/typology/gestalten/shard/datamap.js:277=). The provider takes no descriptors.

- the scenario provider is cut :: =tests/scenarios/datamap.js:50= becomes a call to the service; with it goes the only =allowGlobalContext: true=.
- =citizen= is not a word here :: beef: /"not like 'citizen'"/. The accio'd module is named by what it is: =datamap=.
