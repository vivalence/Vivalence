no test writes into a runtime it did not boot.

    mask: instance.runtime,

#+BEGIN_SRC js
// systems/runtime/tests/scenarios/world.js
import { Connection, Url, shape, shard } from "@vivalence/typology";

const routes = (strip, path = "") => [
  ...(strip.effect ? [path || "/"] : []),
  ...Object.entries(strip.branches ?? {}).flatMap(([name, branch]) => routes(branch, `${path}/${name}`)),
];

const rows = async (repository) => Object.fromEntries((await repository.find({})).map((held) => [`${held.type} ${held.slug}`, held.status]));

export const world = (die, execution) => {
  const connection = new Connection(new Url("http://test"), shard.transmitter.inline(shape.http(die.runtime.aperture)));
  const held = (slug) => die.runtime.entities.process.findOneOrFail({ slug });
  return {
    die,
    execution,
    connection,
    daemon: (slug) => die.runtime.daemons[slug],
    as: (token) =>
      connection.clone().use(async (ctx, next) => {
        ctx.request.headers.set("authorization", `Bearer ${token}`);
        await next();
      }),
    census: () => routes(shape.strip(die.runtime.aperture)).sort(),
    ledger: async () =>
      Object.assign(
        await rows(die.runtime.entities.process),
        ...(await Promise.all(Object.values(die.runtime.daemons).map((daemon) => rows(daemon.entities.process)))),
      ),
    chronicle: () => {
      const told = {};
      for (const record of die.controller.stdout.records) (told[record.path] ??= []).push(record.verb);
      return told;
    },
    faults: () => die.controller.stdout.records.filter((record) => record.verb === "fault").map((record) => `${record.path} ${record.data.message}`),
    kill: async (slug, signal) => (await held(slug)).controller.kill(signal),
    stop: async () => {
      await die.controller.kill("SIGTERM");
      await execution;
    },
  };
};

// systems/runtime/tests/scenarios/spawn.js
import { Connection, Pipe, Url, shard } from "@vivalence/typology";
import { TextLineStream } from "@std/streams";

export const spawn = async ({ instance, mountpoint, port, deadline = 60_000 }) => {
  const child = new Deno.Command(Deno.execPath(), {
    args: ["run", "-A", new URL("../../run.js", import.meta.url).pathname],
    cwd: instance,
    clearEnv: true,
    env: {
      PATH: Deno.env.get("PATH"),
      HOME: Deno.env.get("HOME"),
      VIVA_SYSTEM_MODE: "DEVELOPMENT",
      VIVA_SYSTEM_ROLE: "RUNTIME",
      VIVA_LEDGER_MOUNT: `${mountpoint}/ledger`,
      VIVA_REGISTRY_MOUNT: `${Deno.env.get("HOME")}/.viva/registry`,
      VIVA_INSTANCE_MOUNT: instance,
      VIVA_MOUNTPOINT_MOUNT: `${mountpoint}/mountpoint`,
      VIVA_RUNTIME_ORIGIN: `http://localhost:${port}`,
    },
    stdout: "piped",
    stderr: "piped",
  }).spawn();
  const lines = [];
  const out = new Pipe().from(child.stdout.pipeThrough(new TextDecoderStream()).pipeThrough(new TextLineStream()));
  out.tap((line) => lines.push(line));
  let timer;
  await Promise.race([
    new Promise((resolve) => out.tap((line) => /^Status:ALIVE$/.test(line.trim()) && resolve())),
    child.status.then((exit) => Promise.reject(new Error(`runtime exited ${exit.code}\n${lines.join("\n")}`))),
    new Promise((_, reject) => (timer = setTimeout(() => reject(new Error(`runtime not ready after ${deadline} ms`)), deadline))),
  ]).finally(() => clearTimeout(timer));
  return {
    lines,
    connection: new Connection(new Url(`http://localhost:${port}`), shard.transmitter.fetcher),
    signal: (name) => child.kill(name),
    exit: () => child.status,
  };
};

// systems/runtime/tests/cycles/lifecycle.test.js
import { specimen } from "@vivalence/typology";
import { boot, citizens, instances } from "@vivalence/runtime/scenarios";
import frozen from "./frozen/lifecycle.json" with { type: "json" };

specimen.describe("the life of a runtime — campus: two daemons, five modes, one service", () => {
  specimen.it("gestalt: what stands once it runs", async () => {
    const world = await boot(await instances.campus());
    specimen.expect(await world.ledger()).toEqual(frozen.booted.ledger);
    specimen.expect(world.census()).toEqual(frozen.booted.census);
    await world.stop();
  });

  specimen.it("cycles: a day, told from its spans", async () => {
    const world = await boot(await instances.campus());
    const told = {};
    await world.kill("north", "SIGSTOP");
    told.paused = await world.ledger();
    told.served = (await world.connection.call("/daemon/south/status")).code;
    await world.kill("north", "SIGCONT");
    await world.kill("south", "SIGTERM");
    await (await world.die.runtime.entities.process.findOne({ slug: "south" })).execution;
    told.shrunk = { ledger: await world.ledger(), census: world.census().length };
    await world.kill("north", "SIGTERM");
    await world.kill("north", "SIGTERM");
    await world.stop();
    specimen.expect({ ...told, chronicle: world.chronicle(), faults: world.faults() }).toEqual(frozen.day);
  });

  for (const [name, fault] of Object.entries(frozen.faults))
    specimen.it(`breaks: ${name}`, async () => {
      const refused = await boot(await instances.campus({ faults: citizens.faults[fault.citizen](...fault.arguments) })).then(
        (world) => world.stop().then(() => ({ chronicle: world.chronicle(), faults: world.faults(), settled: world.die.controller.status.reflection.code })),
        (error) => ({ refused: error.message }),
      );
      specimen.expect(refused).toEqual(fault.stands);
    });
});
