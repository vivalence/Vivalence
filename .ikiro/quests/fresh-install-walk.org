* the walk, off the published images

A push to GitLab builds four images and publishes them, tag =alpine= and the commit's short hash: =registry.vivalence.org/beef/viva/vivalence/{viva,runtime,anima,documentation}=. They pull without a login. They are =arm64= only.

No shell work inside a container. The instance folder of the droneaid package is the compose project: =~/.viva/registry/droneaid/instances/droneaid/= holds =instance.viva.json=, =docker-compose.yml=, =Dockerfile=, =ledger.json=, =.env.example=. It is the prod compose of runtime and anima, both services on one host, without the proxy labels.

** 0 · the folder, on the server

#+begin_src sh
git clone https://github.com/vivalence/registry-droneaid.git
cd registry-droneaid/instances/droneaid
cp .env.example .env
openssl rand -base64 24
vim .env
#+end_src

In =.env=: =SECRET_VIVA_JWT= is the line =openssl= printed, =SECRET_VIVA_OPENROUTER_API_KEY= the key, =PUBLIC_VIVA_RUNTIME_REMOTE= and =VIVA_CLIENT_ANIMA_ORIGIN= the two addresses a browser reaches, =VIVA_TAG= a commit's short hash to pin the images or =alpine= for the newest. The compose files reach GitHub only after his push of the package.

** 1 · what the folder holds

- =Dockerfile=: =FROM …/runtime:${VIVA_TAG}=, then =viva registry/tap= of =registry-assembly= and =registry-droneaid= from GitHub. The packages are baked into the image at build.
- =ledger.json=: the flat ledger recipe (not named =*.viva.json=: the registry would read it as a module and fault it), mounted over the image's =/viva/ledger/ledger.viva.json=: lighthouse =AUTOSIGNUP=, one OpenRouter =dialogue= faculty.
- =instance.viva.json=: the recipe with =HOME= on anima. The folder is mounted read-only at =/viva/instance= in both services; its =.env= is read there as the instance's.
- volume =mountpoint=: the database, =/viva/mountpoint/droneaid.viva.db=.

#+begin_src js
const shown = (ctx, answer) =>
  ctx.signal.flags.json === true ? answer : { status: answer.status, user: answer.user };

const PASSWORD = {
  title: "sign up",
  pages: [{
    title: "password",
    fields: [{ name: "password", label: "password", kind: "password", initial: "" }],
  }],
  actions: ["commit", "abort"],
};

const signup = async (ctx) => {
  const [username, password] = ctx.signal.body;
  if (!username) throw new Error("usage: viva instance/lighthouse/signup <username> [password]");
  await mount(paladin);
  const runtime = new Connection(paladin.instance.runtime.statics.remote);
  const traits = paladin.instance.lighthouse.traits;
  const named = !traits.includes("JOINABLE") && !traits.includes("INVITATIONAL");
  if (password && named) {
    await runtime.call("/lighthouse/autosignup", { username });
    const proof = await operator(paladin.secret.get("SECRET_VIVA_JWT"));
    const reset = { operator: proof, username, password };
    return shown(ctx, await runtime.call("/lighthouse/password", reset));
  }
  if (password) return shown(ctx, await runtime.call("/lighthouse/signup", { username, password }));
  if (traits.includes("AUTOSIGNUP")) {
    return shown(ctx, await runtime.call("/lighthouse/autosignup", { username }));
  }
  if (!ctx.hold) {
    throw new Error("instance/lighthouse/signup: this lighthouse needs <username> <password>");
  }
  const asked = await ctx.hold(views.form, PASSWORD);
  if (asked.aborted) return { aborted: true };
  const typed = { username, password: asked.values.password };
  return shown(ctx, await runtime.call("/lighthouse/signup", typed));
};

const signin = async (ctx) => {
  const [username, password] = ctx.signal.body;
  if (!username) throw new Error("usage: viva instance/lighthouse/signin <username> [password]");
  await mount(paladin);
  const runtime = new Connection(paladin.instance.runtime.statics.remote);
  if (password) return shown(ctx, await runtime.call("/lighthouse/login", { username, password }));
  if (!paladin.instance.lighthouse.traits.includes("AUTOSIGNUP")) {
    throw new Error("instance/lighthouse/signin: this lighthouse needs <username> <password>");
  }
  return shown(ctx, await runtime.call("/lighthouse/autosignup", { username }));
};
#+end_src
