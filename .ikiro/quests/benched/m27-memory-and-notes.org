// systems/runtime/daemon/entities/userspace/Note.ts
export enum NoteTraitsEnum {
  CONFIG = "CONFIG",         // data-bearing; code reads by (mode, slug); never in the prompt
  ADVISORY = "ADVISORY",     // mode learned it beyond one user; own system.notes
  ADVERTISED = "ADVERTISED", // enters OTHER modes' system.notes (MCP-resource semantics)
  PEDAGOGIC = "PEDAGOGIC",   // learner-type guidance; tutors read
}

export class NoteEntity extends DataEntity {
  mode?: Rel<ModeEntity>;                 // AUTHOR / provenance; null = daemon-wide
  slug: string & Opt = "";                // unique (mode, slug) → deterministic load
  traits: NoteTraitsEnum[] & Opt = [];    // claims (house idiom)
  trait: any & Opt = {};                  // per-trait payload
  gist!: string;                          // level-1 index line
  body?: string;                          // level-2 prose
  data: any & Opt = {};                   // level-2 structured ({count, axes})
  tags: string[] & Opt = [];
  salience: number & Opt = 0;
  audience: any & Opt = null;             // null=all | {modes:[slug]} | {traits:[ModeTrait]}
}
// uniques: [{ properties: ["mode", "slug"] }]  — Intent / Mode precedent
#+end_src
