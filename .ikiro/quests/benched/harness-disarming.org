import { Vector, traits, v } from "@vivalence/typology";

const Named = v.object({ tool: v.string().desc('A tool name as armed. Example: "fs_delete"') });

const off = (thread) => (thread.traits.includes("ARMED") ? thread.trait.ARMED.off : []);

const setting = (on) => async (ctx) => {
  const { tool } = belt.input.cast(Named, ctx.signal);
  const held = off(ctx.terminal.thread).filter((name) => name !== tool);
  const ARMED = { off: on ? held : [...held, tool] };
  await ctx.client.runtime.entities.thread.updateOne({ id: ctx.terminal.thread.id }, traits.patch(ctx.terminal.thread, { ARMED }));
  return { ARMED };
};

  .open({ nature: "/arm", valence: "arm a tool on the seated thread — ARMED.off loses it. Example: /arm fs_delete", input: Named }, setting(true))
  .open({ nature: "/disarm", valence: "disarm a tool on the seated thread — ARMED.off gains it. Example: /disarm fs_delete", input: Named }, setting(false));

#+BEGIN_SRC js
import { shape, specimen, Vector } from "@vivalence/typology";
import { create } from "./scenarios/cortex.js";

let scenario;
let captured;

specimen.describe("harness — ARMED.off prunes the armory per thread", () => {
  specimen.beforeAll(async () => {
    const harness = new Vector().use(async (ctx, next) => { await next(); captured = ctx; });
    scenario = await create({ harness });
  });
  specimen.afterAll(async () => { await scenario.orm.close(); });

  specimen.it("a thread without ARMED arms everything; off: [fs_delete] loses fs/delete from catalog and dispatch", async () => {
    const { dewey, createThread } = scenario;
    const bare = await createThread();
    await dewey.harness.dialogue.render({ thread: bare.id, parts: [{ type: "text", text: "hi" }] });
    specimen.expect(Object.keys(shape.strip(captured.hallucination.tools).branches.fs.branches)).toContain("delete");
    const armed = await createThread({ traits: ["ARMED"], trait: { ARMED: { off: ["fs_delete"] } } });
    await dewey.harness.dialogue.render({ thread: armed.id, parts: [{ type: "text", text: "hi" }] });
    specimen.expect(Object.keys(shape.strip(captured.hallucination.tools).branches.fs.branches)).not.toContain("delete");
  });

  specimen.it("an unknown name in off is loud: NotFound names the signal", async () => {
    const { dewey, createThread } = scenario;
    const thread = await createThread({ traits: ["ARMED"], trait: { ARMED: { off: ["fs_nope"] } } });
    await specimen.expect(dewey.harness.dialogue.render({ thread: thread.id, parts: [{ type: "text", text: "hi" }] })).rejects.toThrow("Not found");
  });
});
