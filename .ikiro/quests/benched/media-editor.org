* the decision trail

Every fork below was put with a recommendation in the design turn; /"yes go build"/ took all of them.

- *borrow the engines, build the ontology.* ffmpeg and ffprobe do the pixels on the daemon; the
  browser previews with its own =<video>= / =<audio>= elements over proxies and cached peaks. No
  wasm, no borrowed timeline: OpenShot · Shotcut · Olive are C++/Qt, the web timelines are React.
- *UI paradigm* — Resolve's Edit page: bin · viewer · timeline · inspector. Descript's transcript
  rail is the later language-learning seam (annotations), not built.
- *schema* — OpenTimelineIO's nouns. A composition is tracks; a track is a SEQUENCE of clips and
  gaps; an item's position is the sum of what precedes it. Ripple edits are free and overlap is
  impossible by construction; layering comes from tracks. Spans are integer frames at the
  composition's rate, never float seconds. Rejected: absolute offsets (Premiere) — free placement,
  but every ripple is a recompute and overlap has to be policed.
- *ontologies* — three on Literal, storage on =trait.*=, =symbol= derived:
  =media= (one row per file) · =composition= (ONE row holding the whole timeline; clips are not
  rows, as paragraphs are not rows — undo is a version of one document, export is one serialize) ·
  =annotation= (a span on a media or composition, with text and language). Render is not an
  ontology: it is a streamed process whose output IS a media row tagged =media.origin.rendered=.
- *names on disk* — =media/<sha256:16>.<ext>=; the row keeps the name it arrived with. Dedupe by
  content; a file dropped by hand under =media/= is renamed on sync.
- *Range for the monitor* — the design called it a missing prerequisite in =shard/serve.js=. WRONG:
  the cargo route (=systems/runtime/lifecycle/resolve.js= =attachCargo=) already answers 206. Its
  real defect is reading the whole file per request — known issue
  =cargo-range-reads-the-whole-file-per-request=.
- *home* — =~/.viva/registry/media=, package =@media=, mode =editor/media=, instance template
  =instances/media=; beef creates the instance.
- *the upload door* (found building) — =Socket.feed= decodes upstream as text, so a file cannot
  cross the anima multiplex. =buffer/upload.js= opens a second plain-http =Connection= over
  =shard.transmitter.fetcher= + =shard.connection.authorize=, File body, name in =x-viva-name=.

* the law

- =Socket.feed= (=subsystems/typology/prototypes/socket.js:271=) sends text frames: binary never
  crosses the multiplex.
- The literal =symbol= fold sets slugs by depth: =[media, media.kind.video, media.rendered]= →
  ={media:{kind:"video"}}=. A marker sharing a root with a deeper sibling needs the same depth.
- A JSON column is updated by assigning the property. =known.assign(object.patch(known, data))=
  left a written composition with 0 of 3 items; which layer ate them is unisolated.
- A rename-on-settle bin consumes its per-path marks only after a SUCCESSFUL upsert — the watcher
  probes a half-written render first and throws. Renders write to =renders/.incoming-<uuid>= and
  rename; a failed file is skipped with a warn, never aborts the pass.
- FFmpeg 9.0.1 rejects =-filter_complex_script=; =-/filter_complex <file>= is the form.
- =path.extname(".mp4")= is =""=: =settled(hash, name)= takes a filename.

const at = (ctx, path) => paladin.skills.fs.resolve(ctx.mode.mountpoint.absolute, path);

const one = (ctx, ontology, slug) => ctx.daemon.entities.literal.findOne({ ontology, mode: ctx.mode.id, slug });

const all = (ctx, ontology, where = {}) => ctx.daemon.entities.literal.find({ ontology, mode: ctx.mode.id, ...where });

const served = (catalog, literal) => {
  const media = literal.trait.MEDIA;
  const url = (path) => (path ? (catalog[path]?.url ?? null) : null);
  return {
    literal,
    url: url(media.asset.path),
    derived: {
      peaks: url(media.derived?.peaks),
      proxy: url(media.derived?.proxy),
      thumbs: media.derived?.thumbs
        ? { ...media.derived.thumbs, frames: Array.from({ length: media.derived.thumbs.count }, (_, i) => url(`${media.derived.thumbs.directory}/${String(i + 1).padStart(5, "0")}.jpg`)) }
        : null,
    },
  };
};

const binOf = async (ctx) => Object.fromEntries((await all(ctx, ONTOLOGY.media)).map((row) => [row.slug, row.trait.MEDIA]));

const missing = (what, slug) => new Error(`${slug} is not a ${what} in this bin`);

const compositionOf = async (ctx, slug) => {
  const literal = await one(ctx, ONTOLOGY.composition, slug);
  if (!literal) throw missing("composition", slug);
  return literal;
};

const saveComposition = async (ctx, slug, document) => {
  if (!v.check(Composition, document)) throw new Error(`not a composition: ${[...v.errors(Composition, document)].map((error) => `${error.path} ${error.message}`).join("; ")}`);
  const path = compositionPath(slug);
  await paladin.state.json(at(ctx, path), document);
  const { literal: [literal] } = await sync(ctx.daemon, ctx.mode, [path]);
  return literal;
};

const freeSlug = async (ctx, ontology, wanted) => {
  const taken = new Set((await all(ctx, ontology)).map((row) => row.slug));
  let slug = wanted;
  for (let i = 2; taken.has(slug); i++) slug = `${wanted}-${i}`;
  return slug;
};

async function* admitting(ctx, steps) {
  let step = await steps.next();
  while (!step.done) {
    yield step.value;
    step = await steps.next();
  }
  return step.value;
}

const remember = async (ctx, slug, derived) => {
  const store = fork(ctx.daemon);
  const literal = await store.literal.findOne({ ontology: ONTOLOGY.media, mode: ctx.mode.id, slug });
  literal.trait = { MEDIA: { ...literal.trait.MEDIA, derived } };
  await store.em.flush();
  for (const path of [derived.peaks, derived.proxy].filter(Boolean)) ctx.mode.freight.admit(path);
  if (derived.thumbs) for (const file of await paladin.find.walk(/\.jpg$/)(at(ctx, derived.thumbs.directory))) ctx.mode.freight.admit(file.absolute.slice(ctx.mode.mountpoint.absolute.length + 1));
  return literal;
};

export const aperture = new Aperture()
  .open("/media/list", async (input, ctx) => {
    const catalog = ctx.mode.freight.catalog;
    return (await all(ctx, ONTOLOGY.media)).map((row) => served(catalog, row));
  })
  .open("/media/read", async (input, ctx) => {
    const literal = await one(ctx, ONTOLOGY.media, input.slug);
    if (!literal) throw missing("media", input.slug);
    return served(ctx.mode.freight.catalog, literal);
  })
  .open("/media/import", async (input, ctx) => {
    const source = paladin.skills.fs.absolute(input.path);
    const name = input.name ?? source.split("/").pop();
    const { path } = await ingest(ctx.mode.mountpoint.absolute, { source, name, move: !!input.move });
    stamp(path, { name });
    const { literal: [literal] } = await sync(ctx.daemon, ctx.mode, [path]);
    return served(ctx.mode.freight.catalog, literal);
  })
  .open({ nature: "/media/ingest", yields: Progress }, async function* (input, ctx) {
    const body = ctx.request.raw?.body;
    if (!body) throw new Error("[media.ingest] no file in the request body");
    const header = ctx.request.headers.get("x-viva-name");
    const name = (header && decodeURIComponent(header)) || ctx.request.url.searchParams?.get("name") || "upload";
    yield { stage: "ingest", fraction: 0 };

  for (const symbol of symbols) if (!symbol.mode && !literal.symbols.contains(symbol)) literal.symbols.add(symbol);

const write = async (store, mode, fresh, known) => {
  const data = object.omit(fresh, ["symbols"]);
  if (known) {
    known.traits = data.traits;
    known.trait = data.trait;
    known.ontology = data.ontology;
    return known;

  const literal = store.literal.create({ ...data, mode: mode.id });

export const describe = async (root, path) => {
  const stat = await Deno.stat(`${root}/${path}`).catch(() => null);
  if (!stat) return null;
  if (stat.isDirectory) return { path, directory: true };
  return { path, bytes: stat.size, mtime: stat.mtime?.toISOString() ?? null };
};

export const stamp = (path, mark) => pending.set(path, { ...(pending.get(path) ?? {}), ...mark });

const canonical = async (root, file) => {
  const hash = await digest(`${root}/${file.path}`);
  const path = settled(hash, file.path);
  if (file.path === path || !file.path.startsWith("media/")) return { ...file, hash };
  const held = await Deno.stat(`${root}/${path}`).catch(() => null);
  if (held) await Deno.remove(`${root}/${file.path}`);
  else await Deno.rename(`${root}/${file.path}`, `${root}/${path}`);
  return { ...file, hash, path };

const upsertMedia = async (store, mode, file, held) => {
  const root = mode.mountpoint.absolute;
  const mark = pending.get(file.path) ?? {};
  const same = held.find((row) => row.trait.MEDIA.asset.path === file.path);
  if (same && same.trait.MEDIA.bytes === file.bytes && same.trait.MEDIA.mtime === file.mtime && !mark.name && !mark.rendered) return { literal: same, fresh: false };
  const name = mark.name ?? same?.trait.MEDIA.name ?? file.path.split("/").pop();
  const settledFile = await canonical(root, file);
  const known = same ?? held.find((row) => row.slug === settledFile.hash);
  const fresh = mediaRow({
    path: settledFile.path,
    hash: settledFile.hash,
    name,
    bytes: file.bytes,
    mtime: file.mtime,

    rendered: mark.rendered ?? known?.trait.MEDIA.rendered,

  const literal = await write(store, mode, fresh, known);

  const root = mode.mountpoint.absolute;

const reconcile = async (daemon, mode, paths) => {
  const root = mode.mountpoint.absolute;
  const store = fork(daemon);
  const held = [...(await rows(store, mode, ONTOLOGY.media)), ...(await rows(store, mode, ONTOLOGY.composition))];
  const under = (path) => held.filter((row) => pathOf(row) === path || pathOf(row).startsWith(`${path}/`));
  const gone = new Set();
  const files = [];
  for (const path of paths) {
    if (skipped(path)) continue;
    const file = await describe(root, path);
    if (!file) for (const row of under(path)) gone.add(row);
    else if (file.directory) {
      const found = (await paladin.find.index(`${root}/${path}`, mode.statics.ignore)).map((inner) => ({ ...inner, path: `${path}/${inner.path}` }));
      const present = new Set(found.map((inner) => inner.path));
      for (const row of under(path)) if (!present.has(pathOf(row))) gone.add(row);
      files.push(...found.filter((inner) => belongs(inner.path) && !skipped(inner.path)));
    } else if (belongs(path)) files.push(file);
  }
  for (const row of gone) store.em.remove(row);
  await store.em.flush();
  const kept = held.filter((row) => !gone.has(row));
  const literal = [];
  const created = [];
  for (const file of files) {
    try {
      const { literal: row, fresh } = file.path.startsWith("compositions/")
        ? await upsertComposition(store, mode, file, ofOntology(kept, ONTOLOGY.composition))
        : await upsertMedia(store, mode, file, ofOntology(kept, ONTOLOGY.media));
      if (fresh) created.push(row);
      literal.push(row);
    } catch (error) {
      console.warn(`[MEDIA] ${file.path} skipped: ${error.message.split("\n")[0]}`);
    }
  }
  await store.em.flush();
  for (const row of created) row.assign({ updatedAt: new Date() });
  await store.em.flush();
  return { literal, removed: gone.size };
};

export const watch = async (daemon, mode) => {
  const root = await Deno.realPath(mode.mountpoint.absolute);
  const ignored = paladin.find.skip(mode.statics.ignore);
  const watcher = Deno.watchFs(root, { recursive: true });
  const pending = new Set();
  let timer = null;
  const settle = () => {
    const paths = [...pending];
    pending.clear();
    sync(daemon, mode, paths).catch((error) => console.error("[MEDIA:watch]", error));
  };
  (async () => {
    try {
      for await (const event of watcher) {
        for (const absolute of event.paths) {
          const path = absolute.slice(root.length + 1);
          if (path && !path.split("/").some(ignored) && !path.startsWith("derived/")) pending.add(path);
        }
        if (!pending.size) continue;
        clearTimeout(timer);
        timer = setTimeout(settle, SETTLE);
      }
    } catch (error) {
      if (!(error instanceof Deno.errors.BadResource)) console.error("[MEDIA:watch]", error);
    }
  })();
  return () => {
    clearTimeout(timer);
    watcher.close();
  };
};

  for (const directory of ["media", "derived", "renders", "compositions"]) await paladin.state.dir(`${root}/${directory}`);

      `${await store.literal.count({ ontology: ONTOLOGY.composition, mode: mode.id })} compositions · synced ${literal.length} · pruned ${removed}`,
  );
  return watch(daemon, mode);
}

import { duration, locate, offsets, seconds } from "../composition.js";

export { duration, locate, offsets, seconds };

const pad = (value, width = 2) => String(value).padStart(width, "0");

export const timecode = (frame, rate) => {
  const whole = Math.max(0, Math.floor(frame));
  const total = Math.floor(whole / rate);
  return `${pad(Math.floor(total / 3600))}:${pad(Math.floor((total % 3600) / 60))}:${pad(total % 60)}:${pad(whole % rate)}`;
};

export const clock = (value) => {
  const total = Math.max(0, Math.round(value));
  return `${pad(Math.floor(total / 60))}:${pad(total % 60)}`;
};

export const bytes = (value) => {
  if (value < 1024) return `${value} B`;
  if (value < 1024 ** 2) return `${(value / 1024).toFixed(0)} KB`;
  if (value < 1024 ** 3) return `${(value / 1024 ** 2).toFixed(1)} MB`;
  return `${(value / 1024 ** 3).toFixed(2)} GB`;
};

export const under = (composition, frame) =>
  composition.tracks.map((track) => {
    const found = locate(track, frame);
    return found && found.item.type === "clip" ? { clip: found.item, local: found.local, at: found.at } : null;
  });

export const visible = (composition, frame) => {
  const rows = under(composition, frame);
  for (let i = composition.tracks.length - 1; i >= 0; i--) if (composition.tracks[i].kind === "video" && rows[i] && !composition.tracks[i].mute) return { trackIndex: i, ...rows[i] };
  return null;
};

export const source = (media) => media.derived?.proxy ?? media.url;

export const peaksFor = (peaks, item, rate) => {
  if (!peaks) return [];
  const from = Math.floor(seconds(item.source.start, rate) * peaks.perSecond);
  const to = Math.ceil(seconds(item.source.start + item.source.duration, rate) * peaks.perSecond);
  return peaks.peaks.slice(from, to);
};

export const drawPeaks = (canvas, bins, color) => {
  const context = canvas.getContext("2d");
  const { width, height } = canvas;
  context.clearRect(0, 0, width, height);
  if (!bins.length) return;
  context.fillStyle = color;
  const step = width / bins.length;
  const middle = height / 2;
  for (let i = 0; i < bins.length; i++) {
    const magnitude = Math.max(1, bins[i] * middle);
    context.fillRect(i * step, middle - magnitude, Math.max(1, step - 0.5), magnitude * 2);
  }
};

export const thumbsFor = (thumbs, item, rate, count) => {
  if (!thumbs?.frames?.length) return [];
  const start = seconds(item.source.start, rate);
  const span = seconds(item.source.duration, rate);
  return Array.from({ length: count }, (_, i) => {
    const at = start + (span * (i + 0.5)) / count;
    const index = Math.min(thumbs.frames.length - 1, Math.floor(at / thumbs.every));
    return thumbs.frames[index];
  });
};

export const snapshot = (composition) => JSON.stringify(composition);

export const history = () => {
  const past = [];
  const future = [];
  return {
    push(held) {
      past.push(snapshot(held));
      if (past.length > 100) past.shift();
      future.length = 0;
    },
    undo(current) {
      const held = past.pop();
      if (!held) return null;
      future.push(snapshot(current));
      return JSON.parse(held);
    },
    redo(current) {
      const held = future.pop();
      if (!held) return null;
      past.push(snapshot(current));
      return JSON.parse(held);
    },
    get depth() {
      return past.length;
    },
  };
};

    else out.push(item);

  const bounded = Math.max(0, Math.min(to > from ? to - 1 : to, items.length));

  return replaceTrack(composition, trackIndex, merged(items));
};

export const between = (composition, trackIndex, index, targetTrack, targetIndex) => {
  const item = trackAt(composition, trackIndex).items[index];
  if (!item) return composition;
  return insert(remove(composition, trackIndex, index), targetTrack, targetIndex, item);
};

};

  composition.tracks
    .map((track, trackIndex) => ({ track, trackIndex }))
    .filter(({ track }) => !kind || track.kind === kind)
    .flatMap(({ track, trackIndex }) => {
      const starts = offsets(track);
      return track.items.map((item, index) => ({ item, index, trackIndex, at: starts[index] })).filter(({ item }) => item.type === "clip");

export const validate = (composition, mediaBySlug) => {
  const issues = [];
  for (const { item, trackIndex, index } of clipsOf(composition)) {
    const media = mediaBySlug[item.media];
    if (!media) {
      issues.push({ trackIndex, index, fault: `media ${item.media} is not in the bin` });

      issues.push({ trackIndex, index, fault: `clip reads past the end of ${media.name}: ${item.source.start + item.source.duration} > ${available} frames` });
  }

    yield { stage: "peaks", fraction: 0 };
    derived.peaks = await peaks(root, media);
    yield { stage: "peaks", fraction: 1, done: true };

    while (!step.done) {
      yield step.value;
      step = await steps.next();
    }
    derived.proxy = step.value;
  }

const TAIL = 2000;

export const binary = { ffmpeg: "ffmpeg", ffprobe: "ffprobe" };

    return false;
  }

export const capture = async (bin, args) => {
  const { code, stdout, stderr } = await new Deno.Command(bin, { args, stdout: "piped", stderr: "piped" }).output();
  if (code !== 0) throw new Error(`[media] ${bin} exited ${code}: ${decoder.decode(stderr).trim().slice(-TAIL)}`);
  return stdout;

const lines = async function* (readable) {
  let held = "";
  for await (const chunk of readable) {
    held += decoder.decode(chunk, { stream: true });
    const parts = held.split("\n");
    held = parts.pop();
    yield* parts;
  }
  if (held) yield held;
};

export async function* run(args, { duration, stage = "ffmpeg", signal } = {}) {
  const child = new Deno.Command(binary.ffmpeg, {
    args: ["-hide_banner", "-nostats", "-y", "-progress", "pipe:1", ...args],
    stdout: "piped",
    stderr: "piped",
    signal,
  }).spawn();
  const tail = (async () => {
    let text = "";
    for await (const chunk of child.stderr) text = (text + decoder.decode(chunk)).slice(-TAIL);
    return text;
  })();
  let last = 0;
  yield { stage, fraction: 0 };
  for await (const line of lines(child.stdout)) {
    const [key, value] = line.split("=");
    if (key === "out_time_us" && duration > 0) {
      const fraction = Math.min(1, Number(value) / 1_000_000 / duration);
      if (fraction - last >= 0.01) {
        last = fraction;
        yield { stage, fraction };
      }
    }
  }
  const status = await child.status;
  const stderr = await tail;
  if (!status.success) throw new Error(`[media] ffmpeg ${stage} exited ${status.code}: ${stderr.trim()}`);
  yield { stage, fraction: 1, done: true };
}

export const settle = async (progress) => {
  let last = null;
  for await (const step of progress) last = step;
  return last;
};

  const inputs = [];

    const held = inputs.indexOf(slug);

    return inputs.length - 1;

  const laid = { video: [], audio: [] };
  composition.tracks.forEach((track, trackIndex) => {
    if (track.mute || !track.items.length) return;

      const label = `t${trackIndex}i${index}`;

      } else {
        const input = inputOf(item.media);

      }
      labels.push(`[${label}]`);

    chains.push(`${labels.join("")}concat=n=${labels.length}:v=${track.kind === "video" ? 1 : 0}:a=${track.kind === "audio" ? 1 : 0}[${out}]`);

  await Deno.remove(script).catch(() => {});
