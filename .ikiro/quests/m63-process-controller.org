import { Span, Status, Vector, Signal, Failure, shard, steer, promise, v } from "@vivalence/typology";

export function Controller({
  stdout = new Span(),
  status = new Status("IDLE"),
  stdin = new Vector(),
  abort = new AbortController(),
  parent = null,
  children = new Set(),
} = {}) {
  const { MACHINE } = v.prototypes.controller;
  const gate = promise.waiter();
  const controller = { stdout, status, stdin, abort, parent, children };

  controller.settled = new Promise((fulfil, reject) => {
    const untap = stdout.pipe.tap((record) => {
      if (record.span !== stdout.id) return;
      const from = status.reflection.code;
      const code = MACHINE.transitions[record.verb]?.[from] ?? from;
      if (code === from) return;
      const { error } = MACHINE.verbs[record.verb];
      status.set({ code, ...(error && { error: new Failure(error, record) }) });
      gate.wake();
      const { settled } = MACHINE.states[code];
      if (!settled) return;
      untap();
      parent?.children.delete(controller);
      for (const child of children) child.abort.abort(`parent ${code}`);
      settled === "reject" ? reject(status.reflection.error) : fulfil(controller.toJSON());
    });
  });
  controller.settled.catch(() => {});

  const aborted = () => {
    stdout.mark("abort", { reason: abort.signal.reason });
    for (const child of children) child.abort.abort(abort.signal.reason);
  };
  abort.signal.aborted ? aborted() : abort.signal.addEventListener("abort", aborted, { once: true });

  stdin.use(shard.controller.signalling(controller));
  for (const name of Object.keys(MACHINE.signals)) if (!stdin.trie.has(name)) stdin.open(name, () => {});

  parent?.children.add(controller);
  return Object.assign(controller, {
    kill: (name, input) => steer.dispatch.invoke(stdin, new Signal(name), steer.strategy.echo)(input),
    proceed: async () => {
      while (status.is("PAUSED")) await gate.wait(abort.signal);
      return status.is("RUNNING");
    },
    toJSON: () => ({
      path: stdout.absolute,
      span: stdout.id,
      ...status.reflection,
      children: [...children].map((child) => child.toJSON()),
    }),
    [Symbol.for("nodejs.util.inspect.custom")]: () => `Controller(${stdout.absolute}):${status.reflection.code}`,
  });
}

import { specimen, Controller, Span, Vector, Failure } from "@vivalence/typology";

specimen.describe("Controller: one span, one fold, one tree", () => {
  specimen.it("status is the fold of the controller's OWN span: open → RUNNING, close → DONE, the promise fulfils", async () => {
    const run = Controller({ stdout: new Span("run") });
    specimen.expect(run.status.is("IDLE")).toBe(true);
    run.stdout.open();
    specimen.expect(run.status.is("RUNNING")).toBe(true);
    run.stdout.branch("db").fault(new Error("a child's fault stays the child's"));
    specimen.expect(run.status.is("RUNNING")).toBe(true);
    run.stdout.close();
    specimen.expect((await run.settled).code).toBe("DONE");
  });

  specimen.it("a fault settles FAILED and absorbs the finally close; the promise rejects with the fault's record", async () => {
    const run = Controller({ stdout: new Span("run") });
    run.stdout.open().fault(new Error("boom")).close();
    specimen.expect(run.status.is("FAILED")).toBe(true);
    const failure = await run.settled.catch((error) => error);
    specimen.expect(failure).toBeInstanceOf(Failure);
    specimen.expect(failure.code).toBe("FAILED");
    specimen.expect(failure.message).toBe("boom");
  });

  specimen.it("SIGTERM is cooperative: STOPPING, proceed() says wind down, close settles STOPPED with the signal", async () => {
    const run = Controller({ stdout: new Span("run") });
    run.stdout.open();
    await run.kill("SIGTERM");
    specimen.expect(run.status.is("STOPPING")).toBe(true);
    specimen.expect(await run.proceed()).toBe(false);
    run.stdout.close();
    const failure = await run.settled.catch((error) => error);
    specimen.expect(failure.code).toBe("STOPPED");
    specimen.expect(failure.record.data).toEqual({ signal: "SIGTERM" });
  });

  specimen.it("SIGSTOP parks proceed() until SIGCONT", async () => {
    const run = Controller({ stdout: new Span("run") });
    run.stdout.open();
    await run.kill("SIGSTOP");
    let resumed = null;
    const waiting = run.proceed().then((go) => (resumed = go));
    await Promise.resolve();
    specimen.expect(resumed).toBe(null);
    await run.kill("SIGCONT");
    specimen.expect(await waiting).toBe(true);
  });

  specimen.it("SIGKILL is preemptive: abort marks, ABORTED absorbs the runner's fault, children die with the parent", async () => {
    const parent = Controller({ stdout: new Span("parent") });
    const child = Controller({ parent, stdout: parent.stdout.branch("child") });
    parent.stdout.open();
    child.stdout.open();
    await parent.kill("SIGKILL", "operator");
    specimen.expect(parent.abort.signal.aborted).toBe(true);
    specimen.expect(child.abort.signal.reason).toBe("operator");
    specimen.expect(child.status.is("ABORTED")).toBe(true);
    parent.stdout.fault(new Error("AbortError"));
    specimen.expect(parent.status.is("ABORTED")).toBe(true);
    specimen.expect((await parent.settled.catch((error) => error)).code).toBe("ABORTED");
  });

  specimen.it("a signal fans down to every child that answers it; a parent settling aborts its live children", async () => {
    const parent = Controller({ stdout: new Span("parent") });
    const child = Controller({ parent, stdout: parent.stdout.branch("child") });
    const other = Controller({ parent, stdout: parent.stdout.branch("other") });
    parent.stdout.open();
    child.stdout.open();
    other.stdout.open();
    await parent.kill("SIGSTOP");
    specimen.expect(child.status.is("PAUSED")).toBe(true);
    child.stdout.close();
    specimen.expect(parent.children.has(child)).toBe(false);
    parent.stdout.close();
    specimen.expect(other.status.is("ABORTED")).toBe(true);
    specimen.expect(other.abort.signal.reason).toBe("parent DONE");
  });

  specimen.it("adoption: an aborted AbortController settles ABORTED at construction; a client leaf stands under the root use", async () => {

  });

  specimen.it("the tree reflects: toJSON folds children, the wire Record is what a tap hears", () => {
