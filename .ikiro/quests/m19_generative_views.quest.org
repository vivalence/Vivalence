import { Path, crypto } from "@vivalence/typology";

export class View {
  kind = "svelte";
  schema = null;
  mount = null;
  bundles = [];
  outputs = [];
  bundler = null;
  bundling = null;
  sealed = false;

  constructor(mount, schema = null, kind = "svelte") {
    if (mount instanceof View) return mount;
    if (typeof mount === "object" && !(mount instanceof Path)) {
      schema = mount.schema ?? schema;
      kind = mount.kind ?? kind;
      mount = mount.mount;
    }
    this.kind = kind;
    this.schema = schema;
    this.mount = new Path(mount);
  }

  fill(descriptor = {}) {
    descriptor.data ??= {};
    this.schema?.fill(descriptor);
    return descriptor.data;
  }

  withBundler(bundler, { sealed = false } = {}) {
    this.bundler = bundler;
    this.sealed = sealed;
    this.bundling ??= this.compile();
    return this;
  }

  async compile() {
    if (!this.bundler) throw new Error("view missing bundler");
    this.outputs = await this.bundler(this.mount.absolute);
    const base = this.mount.trace?.absolute ?? "";
    this.bundles = await Promise.all(
      this.outputs.map(async (output) => ({
        type: "js",
        mount: output.path.startsWith(base) ? output.path.slice(base.length) : output.path,
        integrity: this.sealed ? await crypto.digest(output.text) : null,
        bytes: output.text.length,
      })),
    );
    return this;
  }

  record() {
    return { kind: this.kind, hash: null, mount: this.mount.nature, bundles: this.bundles };
  }

  serve = async (name) => {
    if (this.bundler && !this.sealed) await this.compile();
    else await this.bundling;
    const base = this.mount.trace?.absolute ?? "";
    const output = this.outputs.find((candidate) => candidate.path === base + name);
    if (!output) return null;
    const entry = this.bundles.find((candidate) => candidate.mount === name);
    return { text: output.text, type: "application/javascript", integrity: entry?.integrity ?? null };
  };

  toJSON() {
    return this.record();
  }
}

import { Socket, fromm } from "@vivalence/typology";

const MIME = {
  html: "text/html", css: "text/css", js: "application/javascript", mjs: "application/javascript",
  json: "application/json", png: "image/png", jpg: "image/jpeg", jpeg: "image/jpeg",
  svg: "image/svg+xml", woff2: "font/woff2", wasm: "application/wasm",
  txt: "text/plain", pdf: "application/pdf", mp3: "audio/mpeg", mp4: "video/mp4",
  webp: "image/webp", ico: "image/x-icon", gif: "image/gif",
};

function mime(path) {
  return MIME[path.split(".").pop()?.toLowerCase()] || "application/octet-stream";
}

export function file(root) {
  const fn = async (ctx) => {
    let i = 0, parts = [];
    while (ctx.params[i] !== undefined) parts.push(ctx.params[i++]);
    const filepath = `${root}/${parts.join("/")}`;
    try {
      const file = await Deno.open(filepath, { read: true });
      ctx.response.type = mime(filepath);
      return file.readable;
    } catch {
      ctx.response.status = 404;
      return null;
    }
  };
  Object.defineProperty(fn, "length", { value: 1 });
  return fn;
}

export function websocket(handler) {
  const fn = (ctx) => {
    const { socket, response } = Deno.upgradeWebSocket(ctx.request.raw);
    handler(socket, ctx);
    return response;
  };
  Object.defineProperty(fn, "length", { value: 1 });
  fn.websocket = true;
  fn.attend = handler;
  return fn;
}

export function multiplex(vector) {
  const sockets = new Set();
  const fn = websocket((ws, ctx) => {
    const socket = new Socket(ws, vector, {
      bearer: ctx?.request?.url?.searchParams?.get("token") ?? null,
    });
    sockets.add(socket);
    ws.addEventListener("close", () => sockets.delete(socket));
  });
  fn.sockets = sockets;
  return fn;
}

export function bundles(sources) {
  return async (input, ctx) => {
    const name = fromm.params(ctx.params).path.absolute;
    for (const source of sources) {
      const served = await source(name);
      if (!served) continue;
      ctx.response.type = served.type;
      if (served.integrity) ctx.response.headers.set("x-viva-integrity", `sha256-${served.integrity}`);
      return served.text;
    }
    ctx.response.status = 404;
    return null;
  };
}

import { join } from "@std/path";
import { bundle, crypto } from "@vivalence/typology";
import { BufferEntity } from "@vivalence/typology/entities";

export const kinds = {
  svelte: {
    ext: "svelte",
    compile: async ({ entry, bundler }) => {
      const outputs = await bundler(entry);
      return outputs.find((output) => output.path === entry).text;
    },
  },
  html: {
    ext: "html",
    compile: async ({ source }) =>
      [
        `const html = ${JSON.stringify(source)};`,
        `export default (target) => {`,
        `  target.innerHTML = html;`,
        `  return { destroy: () => { target.innerHTML = ""; } };`,
        `};`,
      ].join("\n"),
  },
};

export const ARTIFACT = new RegExp(`^/?[0-9a-f]{16}\\.(${Object.keys(kinds).join("|")})\\.mjs$`);

let compiled;
export async function bundler() {
  if (compiled) return compiled;
  const paladin = (await import("@vivalence/paladin")).default;
  if (!paladin.scope.repository) throw new Error("!paladin.scope.repository");
  const reporoot = paladin.scope.repository.absolute;
  const imports = {
    "@vivalence/typology": join(reporoot, "subsystems/typology/mod.client.js"),
    "@vivalence/drapes": join(reporoot, "subsystems/drapes/mod.js"),
  };
  compiled = (entry) =>
    bundle.svelte(entry, { prod: paladin.is.prod, imports, baseUrl: new URL(import.meta.url) });
  return compiled;
}

export async function store(daemon) {
  if (!daemon.mountpoint) throw new Error("store: daemon carries no mountpoint");
  const paladin = (await import("@vivalence/paladin")).default;
  const directory = daemon.mountpoint.absolute + "/spawned";
  await paladin.state.dir(directory);
  return directory;
}

export function spawner({ directory, bundler }) {
  const spawn = async (source, kind = "svelte") => {
    if (typeof source !== "string" || !source.trim()) throw new Error("spawn: source is empty");
    if (!kinds[kind]) throw new Error(`spawn: unknown kind "${kind}" — one of ${Object.keys(kinds)}`);
    const hash = await crypto.digest(source);
    const stem = hash.slice(0, 16);
    const artifact = join(directory, `${stem}.${kind}.mjs`);
    let text = await Deno.readTextFile(artifact).catch(() => null);
    if (text === null) {
      const entry = join(directory, `${stem}.${kinds[kind].ext}`);
      await Deno.writeTextFile(entry, source);
      text = await kinds[kind].compile({ source, entry, bundler });
      await Deno.writeTextFile(artifact, text);
    }
    const mount = `/${stem}.${kind}.mjs`;
    return {
      kind,
      hash,
      mount,
      bundles: [{ type: "js", mount, integrity: await crypto.digest(text), bytes: text.length }],
    };
  };

  spawn.inspect = async (hash) => {
    for (const { ext } of Object.values(kinds)) {
      const source = await Deno.readTextFile(join(directory, `${hash.slice(0, 16)}.${ext}`)).catch(
        () => null,
      );
      if (source !== null) return { hash: await crypto.digest(source), source };
    }
    throw new Error(`inspect: unknown view: ${hash}`);
  };

  spawn.serve = async (name) => {
    const stem = name.startsWith("/") ? name.slice(1) : name;
    if (!ARTIFACT.test(stem)) return null;
    const text = await Deno.readTextFile(join(directory, stem)).catch(() => null);
    if (text === null) return null;
    return { text, type: "application/javascript", integrity: await crypto.digest(text) };
  };

  return spawn;
}

export function factory(mode, daemon) {
  return async (descriptor = {}, view = null) => {
    const buffer = daemon.entities.em.create(BufferEntity, {
      mode: mode.entity.id,
      data: view ? (descriptor.data ?? {}) : mode.app.fill(descriptor),
      view: view ?? null,
      index: descriptor.index ?? 0,
    });
    if (descriptor.thread) {
      const thread = await daemon.entities.thread.findOne(descriptor.thread);
      buffer.thread = thread;
      buffer.index = thread.counter++;
    }
    if (descriptor.literals)
      buffer.literals.add(await daemon.entities.literal.findByIdentifiers(descriptor.literals));
    if (descriptor.symbols)
      buffer.symbols.add(await daemon.entities.symbol.findByIdentifiers(descriptor.symbols));
    return buffer;
  };

<script module>
  const verified = new Map();

  async function digest(text) {
    const bytes = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(text));
    return Array.from(new Uint8Array(bytes))
      .map((byte) => byte.toString(16).padStart(2, "0"))
      .join("");
  }

  async function loader(view) {
    const entry = view.bundles?.find((candidate) => candidate.mount === view.mount) ?? view.bundles?.[0];
    if (entry?.integrity && verified.has(entry.integrity)) return verified.get(entry.integrity);
    const response = await fetch(view.url);
    if (!response.ok) throw new Error(`view fetch ${response.status}`);
    const text = await response.text();
    if (entry?.integrity) {
      const output = await digest(text);
      if (output !== entry.integrity)
        throw new Error(`integrity sha256 ${output.slice(0, 12)} is not ${entry.integrity.slice(0, 12)}`);
    }
    const url = URL.createObjectURL(new Blob([text], { type: "text/javascript" }));
    try {
      const module = await import(/* @vite-ignore */ url);
      if (entry?.integrity) verified.set(entry.integrity, module);
      return module;
    } finally {
      URL.revokeObjectURL(url);
    }
  }
</script>
