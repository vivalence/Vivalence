* The shape (settled across the rotation)
=Request = { turns, tools?, settings?, output?: { object: <schema> } }=
- =turns=      — the conversation (dialogue/object only).
- =tools=      — capabilities offered (record name→spec); definitions for the provider, bodies for the loop.
- =settings=   — model-behaviour config: maxTokens · thinking · thinkingBudget · temperature · tool_choice (the last set INTERNALLY by dressAsObject to force respond).
- =output=     — return-shaping, keyed by semantic. =output.object= is ALWAYS a v-schema (the structured target). Absent → plain dialogue.
Four orthogonal, stable members. Nothing heterogeneous, nothing dead, extends by adding a key (=output.speech=) not a member.
Hallucination-INTERNAL, never on the wire: =rounds= (tool-loop budget) · =tune= (already resolved to THIS faculty at findOne).

async function render(faculty, request, rounds, tools) {
  let turns = request.turns;
  for (let round = 0; round < rounds; round++) {
    const turn = await faculty.via.render({ ...request, turns });
    if (turn?.meta?.stop !== "tool_use") return turn;
    const results = await execute(tools, turn.parts);
    turns = [...turns, turn, { role: "user", parts: results }];
  }
  throw new Error(`[hallucination] '${faculty.type}' tool loop exceeded ${rounds} rounds`);
}

async function* stream(faculty, request, rounds, tools) {
  let turns = request.turns;
  for (let round = 0; round < rounds; round++) {
    let turn = null;
    for await (const packet of await faculty.via.stream({ ...request, turns })) {
      turn = soma.pour(turn, packet);
      yield packet;
    }
    if (!turn || turn.meta?.stop !== "tool_use") return;
    const results = await execute(tools, turn.parts);
    const resultTurn = { role: "user", parts: results };
    yield* soma.drain(resultTurn);
    turns = [...turns, turn, resultTurn];
  }
  throw new Error(`[hallucination] '${faculty.type}' tool loop exceeded ${rounds} rounds`);
}
#+end_src

export function dressAsObject(dialogue) {
  return {
    ...dialogue,
    type: "object",
    channels: { in: dialogue.channels?.in ?? ["text"], out: ["object"] },
    via: {
      render: async (request) => {
        const schema = request.output?.object;
        const respond = { valence: "Return the final result as structured data.", input: schema };
        const turn = await dialogue.via.render({
          ...request,
          tools: { ...request.tools, respond },
          settings: { ...request.settings, tool_choice: { type: "any" } },
        });
        const done = turn.parts.find((part) => part.type === "tool_use" && part.name === "respond");
        if (!done) return turn;
        const parsed =
          typeof done.input === "string" ? (done.input ? JSON.parse(done.input) : {}) : done.input;
        const data = schema ? v.fill(schema, parsed) : parsed;
        return { role: "assistant", parts: [{ type: "object", data }], meta: { stop: "end_turn" }, object: data };
      },
    },
  };
}
