* the law

#+BEGIN_EXAMPLE
two principals, one verify path. a bearer on the lighthouse is a JWT of type "access" (an identity) or
type "system" (a runtime, a ghost). a system token is EXCHANGED, never carried: the holder posts the
shared key ONCE at /auth/system and rides the same Authorization: Bearer every other caller uses.

the signing secret never leaves the lighthouse host. SECRET_VIVA_JWT signs; SECRET_VIVA_LIGHTHOUSE_KEY is
exchanged. a consumer on the other side of the world carries only the key.

the lighthouse runs the daemon's gates. shard.secure.authority(local) · authenticate() · authorize() —
the same shard, a LOCAL provider that verifies in-process instead of over HTTP. /entities is authorized;
a system principal is unscoped; an identity principal is scoped to its own row and may only READ.
writes on daemon and identity are system-only. the one new word is a claim gate beside authorize().
a scope WINS over the caller's where and data — on the daemon and the lighthouse alike.

entity-level defaults publish nothing sensitive. Identity.authentication is `hidden` on the schema —
MikroORM never serializes it, gated or not; it stays loadable for identify(). the row carries NO tokens.

signup is a lighthouse static: open | invite | closed, default invite. a system principal may always
sign up a user. an invite is a system-minted single-use JWT {type: "invite", jti}, burned on redemption.

the store holds hashes. tokens.json keys refresh entries by sha256(token); a stolen file redeems
nothing. sessions per identity are capped; the newest wins.

verify pins the algorithm and the issuer. state is per service, never module-global.
#+END_EXAMPLE

    ...payload,
    createdAt: Date.now(),

  delete db.data.refresh[refreshToken];

import { JSONFilePreset } from "lowdb/node";
import { timingSafeEqual } from "node:crypto"; // `@std/crypto` is mapped without its subpaths; node:crypto resolves under deno.jsonc (measured)

const ACCESS = "1d";
const REFRESH = "180d";
const REFRESH_MS = 180 * 24 * 60 * 60 * 1000;
const SYSTEM = "1h";
const INVITE = "7d";
const INVITE_MS = 7 * 24 * 60 * 60 * 1000;
const SESSIONS = 8;
const ALGORITHMS = ["HS256"];

const encode = (text) => new TextEncoder().encode(text);

const digest = async (text) => {
  const bytes = new Uint8Array(await crypto.subtle.digest("SHA-256", encode(text)));
  return Array.from(bytes, (byte) => byte.toString(16).padStart(2, "0")).join("");
};

// === CHANGE START — was: module-global `db`/`secret`, verify with no algorithm or issuer pin ===
export default async function (service) {
  if (!service.secrets?.jwt) throw new Error("JWT secret not configured in service.secrets.jwt");
  if (service.secrets.jwt.length < 24) throw new Error("service.secrets.jwt must be at least 24 characters");

  const secret = encode(service.secrets.jwt);
  const issuer = service.statics?.serve?.absolute ?? `lighthouse/${service.manifest.slug}`;

  const tokenfile = service.mountpoint.branch("/tokens.json").absolute;
  const db = await JSONFilePreset(tokenfile, { refresh: {}, invites: {} });
  await db.read();
  db.data.refresh ??= {};
  db.data.invites ??= {}; // a pre-m60 file has no invites bag; lowdb's default only seeds a missing file

  const create = (payload, expiration = ACCESS) =>
    new jose.SignJWT(payload)
      .setProtectedHeader({ alg: "HS256" })
      .setIssuedAt()
      .setIssuer(issuer)
      .setJti(payload.jti ?? crypto.randomUUID()) // iat is seconds: two mints in one second must still be two tokens
      .setExpirationTime(expiration)
      .sign(secret);

  const verify = async (token) => {
    const { payload } = await jose.jwtVerify(token, secret, { algorithms: ALGORITHMS, issuer });
    return payload;
  };

  const jwt = { create, verify };

  // === refresh — keyed by sha256(token); the file redeems nothing on its own ===
  const rft = {
    create: async (payload) => {
      const token = await create({ ...payload, type: "refresh" }, REFRESH);
      const now = Date.now();
      db.data.refresh[await digest(token)] = { ...payload, createdAt: now, expiresAt: now + REFRESH_MS };
      const owned = Object.entries(db.data.refresh)
        .filter(([, entry]) => entry.id === payload.id)
        .sort(([, a], [, b]) => a.createdAt - b.createdAt);
      for (const [key] of owned.slice(0, Math.max(0, owned.length - SESSIONS))) delete db.data.refresh[key];
      await db.write();
      return token;
    },
    verify: async (token) => {
      const key = await digest(token);
      const entry = db.data.refresh[key];
      if (!entry) return false;
      if (entry.expiresAt && Date.now() > entry.expiresAt) {
        delete db.data.refresh[key];
        await db.write();
        return false;
      }
      return true;
    },
    revoke: async (token) => {
      delete db.data.refresh[await digest(token)];
      await db.write();
    },
    cleanup: async () => {
      const now = Date.now();
      let changed = false;
      for (const bag of [db.data.refresh, db.data.invites]) {
        for (const [key, entry] of Object.entries(bag)) {
          if (entry.expiresAt && now > entry.expiresAt) {
            delete bag[key];
            changed = true;
          }
        }
      }
      if (changed) await db.write();
    },
  };

  // === invites — a system-minted, single-use door into signup ===
  const invites = {
    mint: async (by) => {
      const jti = crypto.randomUUID();
      const now = Date.now();
      db.data.invites[jti] = { by, createdAt: now, expiresAt: now + INVITE_MS };
      await db.write();
      return create({ jti, type: "invite" }, INVITE);
    },
    // verifies, then burns — a second redemption is a miss
    burn: async (token) => {
      const payload = await verify(token).catch(() => null);
      if (!payload || payload.type !== "invite" || !db.data.invites[payload.jti]) return null;
      const entry = db.data.invites[payload.jti];
      delete db.data.invites[payload.jti];
      await db.write();
      return entry;
    },
  };

  const system = {
    enabled: !!key,
    exchange: async (candidate, holder) => {

#+BEGIN_SRC js :tangle ../../commons/lighthouses/multiplayer/server/lib/throttle.js :mkdirp yes
// a token bucket per KEY around ONE verb. login/signup key by the username attempted, system by the
// holder (the address is not on ctx today — see the quest's deferred list). `limit` a minute, then 429.
export function throttle(handler, { limit = 10, window = 60_000, key = (input) => input?.username } = {}) {
  const buckets = new Map();
  return async (input, ctx) => {
    const now = Date.now();
    const slot = key(input) ?? "*";
    const bucket = buckets.get(slot) ?? { count: 0, reset: now + window };
    if (now > bucket.reset) Object.assign(bucket, { count: 0, reset: now + window });
    bucket.count += 1;
    buckets.set(slot, bucket);
    if (bucket.count > limit) {
      ctx.response.status = 429;
      ctx.response.headers.set("retry-after", String(Math.ceil((bucket.reset - now) / 1000)));
      return { status: "ERROR", error: { code: "THROTTLED", message: "too many attempts — wait a minute" } };
    }
    return handler(input, ctx);
  };
}
