#+begin_src javascript
// gestalten/shape/cortex.js  — zero imports; reads passed structures, returns plain objects (no Cortex import → no cycle)
export const strip = (cortex) =>
  [...cortex.faculties.values()].flat().map((faculty) =>
    ({ type, tune, context, channels, via: Object.keys(faculty.via) }));   // native faculties; derivations stay client-side

export const wire = (connection, strip = []) =>          // connection PRE-BRANCHED to /cortex at the seam
  strip.map((meta) => ({ ...meta, via: providers(connection, meta) }));

const providers = (connection, meta) => {                // the only transport-aware bit
  const round = (request) => ({ type: meta.type, tune: meta.tune, request });   // forward EXACT tune (Q2)
  via.render = (request) => connection.call("/render", round(request));
  via.stream = (request) => connection.observe("/stream", { method: "POST", body: round(request) });
};
#+end_src
