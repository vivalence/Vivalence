#+begin_src javascript
// prototypes/pipe.js — the Event's third face: scan/hold. one combinator, stepper OR foldp by the step.
reactive(seed = null, step = (_, value) => value) {
  const store = atom(seed);                             // nanostores atom (import from "nanostores")
  this.tap((value) => store.set(step(store.get(), value)));
  return store;
}

// prototypes/tracks.js — Objected facet (+ Faulted.raise now emits)
export class Objected {
  span = null; payload = null; schema = null;
  constructor(options, span) { if (options) Object.assign(this, options); this.span = span; }
  set(payload) { this.payload = payload; this.span?.emit?.(); return this; }   // the mutation emits
  get json() { return { payload: this.payload ?? null, schema: this.schema ?? null }; }
}

// prototypes/span.js — pipe flows DOWN; emit is local; record ≠ trace; absolute = ancestry string; log = terse verb
to(pipe) { this.pipe = pipe; return this; }
branch(signature) { const child = super.branch(signature); child.pipe ??= this.pipe; return child; }  // inherit DOWN
emit() { this.pipe?.send(this.record); return this; }                                                 // LOCAL, inert w/o pipe
log(subject, payload, transport) {                                                                    // SNIFF — redesign queued
  const write = (nature, value, wire) => {
    const child = this.branch(nature);
    if (wire) { const t = child.track.transport(); if (wire.request) t.send(wire.request); if (wire.response) t.receive(wire.response); }
    if (value instanceof Error) child.track.fault().raise(value.message, value.code);
    else child.track.object(value);                                                                   // always emits (even undefined→null line)
    return child;
  };
  if (typeof subject === "string") return write(subject, payload, transport);
  let child; for (const [nature, value] of Object.entries(subject)) child = write(nature, value, payload);
  return child;
}
get absolute() {                                                                                      // ancestry-only string; NOT super.absolute (drags finn)
  const natures = [];
  for (const node of this.heritage()) if (node.nature) natures.unshift(node.nature);
  return ("/" + natures.join("/")).replace(/\/{2,}/g, "/");
}
get record() { /* {nature, absolute, timing, transport?, transition?, subject?, fault?, object?} — NO children */ }
get json()   { const result = this.record; if (this.gauges.length) result.children = this.gauges.map((g) => g.json); return result; }

* Lessons (self-corrections)

- The reactive design took SIX beef rotations because I kept adding machinery instead of finding the algebra. When beef says =gigabrain lispable haskelly=, the ask is FRP/monadic minimalism (Event · hold · foldp), not a richer object. The floor was already there: Pipe was an Event, the span TREE was already the accumulator — reactive just had to be =hold=.
- =record= ≠ =trace=: a =get trace()= on a =Signature= subclass shadows the parent-pointer field =heritage()= walks → infinite recursion. Naming collides with the base's structural fields (=trace=, =gauges=, =nature=) — check them before adding a getter.
- =absolute= "overengineered": the elegant precedent was already on disk — =Path=/=Url=/=Signal= all string-override =absolute=; only base =Signature= returns the array (and its =array= drags =finn=). Read the sibling prototypes before inventing.
- =buffer.mode=, never =terminal.mode= (which is undefined). The sketch's handle was aspirational; the Explore-agent map + beef's =use buffer.mode!!= are canonical.
- Snapshot-first paid: the goldens EXPOSED the finn-drag and the double-slash as JSON before any UI consumed them — exactly the "get a grip on the shape" the strand asked for.
- Sniff-as-nature was a conflation beef named precisely: payload keys are DATA (display labels), the nature is the tree position. Kept for the demo, flagged superseded — did not pretend it was the final design.
