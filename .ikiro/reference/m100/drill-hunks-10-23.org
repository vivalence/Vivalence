// subsystems/typology/prototypes/pipe.js   ← Locked Q4: one mechanism (tap a sink), two shapes over it
export class Pipe {
  listeners = new Set();
  send(value) { for (const listener of this.listeners) listener(value); }
  tap(target) {
    const listener = fromm.sink(target).write;
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }
  from(origin) { (async () => { for await (const value of fromm.source(origin).read) this.send(value); })(); return this; }
  fold(seed, step) { const store = atom(seed); this.tap((value) => store.set(step(store.get(), value))); return store; }
  async *[Symbol.asyncIterator]() {
    const queue = new Queue();
    const untap = this.tap(queue);
    try { yield* queue.drain(); } finally { untap(); }
  }
}
