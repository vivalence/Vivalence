#+BEGIN_SRC js
// <pkg>/services/whatsapp/whatsapp.viva.js
const graph = "https://graph.facebook.com/v25.0";

const hex = (buffer) => [...new Uint8Array(buffer)].map((byte) => byte.toString(16).padStart(2, "0")).join("");

const signature = async (secret, body) => {
  const key = await crypto.subtle.importKey("raw", new TextEncoder().encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  return `sha256=${hex(await crypto.subtle.sign("HMAC", key, body))}`;
};

export const manifest = { type: "service", slug: "whatsapp", name: "WhatsApp Cloud", version: "0.0.1", traits: [] };

export function provider(service) {
  const { token, secret, number } = service.secrets;
  const authorized = { authorization: `Bearer ${token}` };
  return {
    verify: async (body, claimed) => (await signature(secret, body)) === claimed,
    send: (to, text) =>
      fetch(`${graph}/${number}/messages`, {
        method: "POST",
        headers: { ...authorized, "content-type": "application/json" },
        body: JSON.stringify({ messaging_product: "whatsapp", to, type: "text", text: { body: text } }),
      }),
    media: async (id) => {
      const { url } = await (await fetch(`${graph}/${id}`, { headers: authorized })).json();
      return new Uint8Array(await (await fetch(url, { headers: authorized })).arrayBuffer());
    },
  };
}
#+END_SRC

// commons/social/whatsapp/attachment.js
import { decodeHex } from "@std/encoding";
import { Vector } from "@vivalence/typology";
import { inbox } from "./provider.js";

const refused = () => new Response(null, { status: 403 });

const signed = async (secret, signature, bytes) =>
  crypto.subtle.verify(
    "HMAC",
    await crypto.subtle.importKey(
      "raw",
      new TextEncoder().encode(secret),
      { name: "HMAC", hash: "SHA-256" },
      false,
      ["verify"],
    ),
    decodeHex(signature.slice("sha256=".length)),
    bytes,
  );

const answer = {
  GET: (ctx) =>
    ctx.request.url.query["hub.verify_token"] === ctx.mask.secrets.verify
      ? new Response(ctx.request.url.query["hub.challenge"])
      : refused(),
  POST: async (ctx) => {
    const signature = ctx.request.headers.get("x-hub-signature-256") ?? "";
    const bytes = await ctx.request.bytes();
    const genuine = await signed(ctx.mask.secrets.app, signature, bytes).catch(() => false);
    if (!genuine) return refused();
    inbox.enqueue(ctx.input);
    return new Response(null, { status: 200 });
  },
};

export const attachment = new Vector().open("/webhook", (ctx) => answer[ctx.request.method](ctx));

// commons/social/whatsapp/provider.js
import { encodeBase64 } from "@std/encoding";
import { Queue } from "@vivalence/typology";
import { heard } from "./heard.js";

export const inbox = new Queue();

const answered = async (response) => {
  if (!response.ok) throw new Error(`whatsapp: ${response.status} ${await response.text()}`);
  return response;
};

export function provider(service) {
  const headers = { authorization: `Bearer ${service.secrets.token}` };

  const fetched = async (id) => {
    const held = await fetch(`${service.statics.graph}/${id}`, { headers })
      .then(answered)
      .then((response) => response.json());
    const bytes = await fetch(held.url, { headers })
      .then(answered)
      .then((response) => response.bytes());
    return { data: encodeBase64(bytes), mime: held.mime_type };
  };

  return {
    async *messages(signal) {
      for await (const payload of inbox.drain(signal))
        yield* await heard(payload, service.statics.owner, fetched);
    },
    send: (address, text) =>
      fetch(`${service.statics.graph}/${service.statics.number}/messages`, {
        method: "POST",
        headers: { ...headers, "content-type": "application/json" },
        body: JSON.stringify({
          messaging_product: "whatsapp",
          to: address,
          type: "text",
          text: { body: text },
        }),
      })
        .then(answered)
        .then((response) => response.json()),
  };
}
