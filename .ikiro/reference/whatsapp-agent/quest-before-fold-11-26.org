* the law

- One bot is one mode. One owner: the service drops every other sender before it fetches anything.
- A service is transport: it knows an address, never a User or a Thread. The mode owns the turn.
- The hook's guard is the signature over the exact bytes of the body. It checks, queues, answers 200, nothing else.
- A message is a Turn of a Thread. No store of messages beside Thread.
- LLMs only through the mode's harness.

#+BEGIN_SRC js
// commons/social/personal/boot.js
import { received } from "./received.js";

const pulled = async (modeDie, messages, signal) => {
  while (!signal.aborted)
    try {
      for await (const message of messages(signal)) await received(modeDie, message);
    } catch (error) {
      modeDie.mode.stdout.fault(error);
    }
};

export const boot = (modeDie, mode) => {
  const release = new AbortController();
  const pulls = mode.statics.platforms.map((platform) =>
    pulled(modeDie, modeDie.providers[platform].messages, release.signal),
  );
  return () => {
    release.abort();
    return Promise.all(pulls);
  };
};
#+END_SRC

// commons/social/personal/received.js
export const received = (modeDie, message) =>
  modeDie.datamap.shard.scope(async () => {
    const { parts, ...provider } = message;
    const user = await modeDie.entities.user.findOneOrFail({ slug: modeDie.mode.statics.user });
    modeDie.entities.em.setFilterParams("user", { user: user.id });
    const mine = { user: user.id, mode: modeDie.mode.id };
    const seen = await modeDie.entities.turn.count({
      thread: mine,
      meta: { provider: { platform: message.platform, id: message.id } },
    });
    if (seen) return;
    const newest = { orderBy: { createdAt: "DESC" }, limit: 1 };
    const [thread] = await modeDie.entities.thread.find(mine, newest);
    const [turn] = thread ? await modeDie.entities.turn.find({ thread: thread.id }, newest) : [];
    const fresh = parts[0].text === "/new";
    const silent = turn && Date.now() - turn.createdAt.getTime();
    const lapsed = silent > modeDie.mode.statics.idle * 1000;
    const current =
      !thread || (turn && (fresh || lapsed)) ? await modeDie.entities.thread.create(mine) : thread;
    await modeDie.entities.em.flush();
    if (fresh) return;
    await modeDie.entities.turn.chain({
      thread: current.id,
      role: "user",
      parts,
      meta: { provider },
    });
  });
#+END_SRC

export const decoded = async (bytes) => {
  const decoder = new OggOpusDecoder({ sampleRate: 16000, forceStereo: false });
  await decoder.ready;
  const { channelData } = await decoder.decodeFile(bytes);
