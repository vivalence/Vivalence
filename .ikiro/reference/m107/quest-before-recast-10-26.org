export const COVERS = [
  [(app) => app.client.runtime === null, panels.booting.Booting],
  [(app) => app.client.runtime.lighthouse === null || app.client.runtime.lighthouse.manifest === null, panels.booting.Booting],

  [() => true, Layout],
];

    terminals();
    status();
    authorized();

    <box style={{ flexDirection: "column", flexGrow: 1, width: "100%", height: "100%" }}>

    renderer.destroy();
