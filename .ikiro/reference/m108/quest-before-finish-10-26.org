const SHOWN = ["log", "fault"];

const shown = (record) => {
  if (!SHOWN.includes(record.track)) return;
  globalThis.console[record.track === "fault" ? "error" : "log"](record.path.join(" › "), record.track, record.data);
};

export const console = async (appDie, next) => {
  for (const record of appDie.client.controller.stdout.history) shown(record);
  const untap = appDie.client.controller.stdout.pipe.tap(shown);
  try {
    await next();
  } finally {
    untap();
  }
};

const encoder = new TextEncoder();

export const ledger = async (appDie, next) => {
  const file = await paladin.ledger.log(appDie.client.instance.manifest.slug).open("client", "spans");

  try {
    await next();
  } finally {
    untap();
    file.close();
  }
};
