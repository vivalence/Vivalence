import { Stall, v } from "@vivalence/typology";
import * as belt from "../../../gestalten/belt/index.js";

export const stall = async (terminalDie, next) => {
  const off = terminalDie.terminal.$thread.subscribe((thread) => {
    if (terminalDie.terminal.stall !== null) terminalDie.terminal.stall.deactivate();
    if (thread === null) {
      terminalDie.terminal.stall = null;
      return;
    }
    terminalDie.terminal.stall = Stall({
      source: thread.$buffers,
      active: terminalDie.terminal.$buffer,
      phase: thread.$phase,
      pull: () => belt.thread.pull(thread),
      depth: () => v.thread.trait(thread, "QUEUEING").depth,
    });
    terminalDie.terminal.stall.on.release((buffer) =>
      terminalDie.terminal.client.runtime.entities.buffer
        .removeOne({ id: buffer.id })
        .catch((error) => terminalDie.controller.stdout.fault(error)),
    );
  });
  try {
    await next();
  } finally {
    off();
    if (terminalDie.terminal.stall !== null) terminalDie.terminal.stall.deactivate();
  }
};
