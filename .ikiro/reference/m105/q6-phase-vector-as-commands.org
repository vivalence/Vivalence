export const stall = async (terminalDie, next) => {
  let buffers = () => {};
  const seated = terminalDie.terminal.$thread.subscribe((thread) => {
    buffers();
    if (thread === null) return;
    terminalDie.terminal.stall.launch = terminalDie.terminal.$buffer.get();
    buffers = thread.$buffers.subscribe(() => terminalDie.terminal.stall.refill());
  });
  try {
    await next();
  } finally {
    seated();
    buffers();
  }
};

import { Vector, steer } from "@vivalence/typology";

const claims = (name) => async (stallDie, next) => {
  if (!stallDie.thread.traits.includes(name))
    throw new Error(`[stall] /${stallDie.thread.phase}: thread #${stallDie.thread.id} claims no ${name} — its traits: ${stallDie.thread.traits.join(" · ") || "none"}`);
  await next();
};

const blank = async (stallDie, next) => {
  if (stallDie.thread.$buffers.get().length === 0)
    await stallDie.runtime.entities.buffer.create({ mode: stallDie.thread.mode.id, thread: stallDie.thread.id, data: {} });
  await next();
};

const queue = async (stallDie, next) => {
  if (stallDie.thread.$buffers.get().length < stallDie.thread.trait.QUEUEING.depth) {
    const emission = await stallDie.thread.mode.connection.call(stallDie.thread.trait.AIMED.mount, { thread: stallDie.thread.id, ...stallDie.thread.trait.MASKED });
    for (const buffer of emission.output.buffer) await stallDie.runtime.entities.buffer.merge(buffer);
  }
  await next();
};

const escorting = async (stallDie, next) => {
  await next();
  if (stallDie.thread.$buffers.get().length === 0) stallDie.terminal.$buffer.set(stallDie.stall.launch);
};

const focus = (stallDie) => {
  const buffers = stallDie.thread.$buffers.get();
  if (buffers.includes(stallDie.terminal.$buffer.get())) return;
  stallDie.terminal.$buffer.set(buffers[0] ?? null);
};

export const phases = new Vector();

phases.open({ nature: "/inert", valence: "the mode manages its own buffers; the stall touches nothing. Example: /inert" }, () => {});

phases.branch({ nature: "/manual", valence: "an empty thread gets one blank buffer; the frame holds one of the thread's buffers. Example: /manual" })
  .use(blank)
  .do(focus);

phases.branch({ nature: "/continuous", valence: "the thread's emitter keeps QUEUEING.depth buffers queued; the frame holds one. Example: /continuous" })

  .use(claims("AIMED"), escorting, queue)
