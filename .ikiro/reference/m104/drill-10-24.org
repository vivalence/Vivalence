// systems/ghost/prototypes/repl.js
import paladin from "@vivalence/paladin";
import { ReplSignal, Vector, fromm, shard, steer } from "@vivalence/typology";
import * as clientCommand from "../subjects/client/commands/index.js";
import * as terminalCommand from "../subjects/terminal/commands/index.js";

export const execution = new Vector()
  .use(shard.controller.open)
  .do((replDie) => steer.dispatch.execute(replDie.repl.commands, replDie, replDie.signal));

export class Repl {
  constructor(client) {
    this.client = client;
  }

  get commands() {
    const { shell, quit, ...everywhere } = clientCommand;
    const terminal = this.client.activeTerminal;
    const commands = new Vector().graft(
      ...Object.values(everywhere),
      ...(paladin.is.tty ? [shell, quit] : []),
      ...fromm.modes(this.client.modes).flat.map((mode) => mode.repl),
      ...(terminal === null ? [] : Object.values(terminalCommand)),
    );
    for (const mode of this.client.runtime.entities.mode.$entities.get()) {
      commands.branch(`/mode/${mode.type}/${mode.slug}`, mode.repl);
      commands.branch(`/mode/${mode.slug}`, mode.repl);
    }
    return commands;
  }

  dispatch(line) {
    const signal = new ReplSignal(line);
    const terminal = this.client.activeTerminal;
    const replDie = {
      signal,
      client: this.client,
      runtime: this.client.runtime,
      terminal,
      controller: (terminal === null ? this.client.controller : terminal.controller).branch(signal.pathname),
      repl: this,
      output: undefined,
    };
    return steer.dispatch.execute(execution, replDie).then(() => replDie.output);
  }
}
