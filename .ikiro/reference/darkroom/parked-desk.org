const EXTENSION = { "image/png": "png", "image/webp": "webp", "image/jpeg": "jpg" };

export const desk = (ctx, id) => ctx.entities.buffer.findOneOrFail({ id, mode: ctx.mode.entity.id });

export const part = async (file) => ({ type: "image", media: file.mime, data: encodeBase64(await Deno.readFile(file.mount)) });

export const parts = async (ctx, references) =>
  Promise.all(references.map(async (reference) => ({ image: await part(await ctx.entities.file.findOneOrFail({ id: reference.file, mode: ctx.mode.entity.id })), role: reference.role })));

const digest = async (bytes) =>
  Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", bytes))).map((byte) => byte.toString(16).padStart(2, "0")).join("").slice(0, 16);

export async function settle(ctx, bytes, media, developed) {
  const path = `pictures/${await digest(bytes)}.${EXTENSION[media]}`;
  const file = await ctx.mode.call.files.receive({ path, bytes, mime: media });
  file.traits = [...new Set([...file.traits, "DEVELOPED"])];
  file.trait = { ...file.trait, DEVELOPED: developed };
  await ctx.entities.em.flush();
  return file;
}

export async function develop(ctx, held, rendered, provenance, { show = true } = {}) {
  const pictures = [];
  for (const image of rendered.images) {
    const bytes = Uint8Array.from(atob(image.data), (char) => char.charCodeAt(0));
    pictures.push(await settle(ctx, bytes, image.media, { ...provenance, revised: rendered.revised, provider: rendered.provider, model: rendered.model, settings: rendered.settings ?? {} }));
  }
  held.data = { ...held.data, batch: pictures.map((file) => file.id), ...(show && { shown: pictures[0].id, compare: undefined }) };
  await ctx.entities.em.flush();
  return {
    message: `${pictures.length} picture${pictures.length === 1 ? "" : "s"}${show ? `, showing ${pictures[0].id}` : ""}${rendered.revised ? ` · painted from: ${rendered.revised}` : ""}`,
    file: pictures,
    buffer: [held],
  };
}

export async function tree(ctx, held, limit) {
  const pictures = await ctx.entities.file.find({ mode: ctx.mode.entity.id, traits: { $like: '%"DEVELOPED"%' } }, { orderBy: { createdAt: "desc" }, limit });
  return {
    message: pictures.length ? `${pictures.length} pictures, newest first` : "no pictures yet",
    file: pictures,
    tree: pictures.map((file) => ({ id: file.id, prompt: file.trait.DEVELOPED.prompt, model: file.trait.DEVELOPED.model, parent: file.trait.DEVELOPED.parent ?? null, shown: file.id === held.data.shown })),
  };
}

import { Image } from "https://deno.land/x/imagescript@1.3.0/mod.ts";

export const EDGE = 512;

export async function thumbnail(bytes, edge = EDGE) {
  const image = await Image.decode(bytes);
  const scale = edge / Math.max(image.width, image.height);
  if (scale < 1) image.resize(Math.round(image.width * scale), Math.round(image.height * scale));
  return { type: "image", media: "image/webp", data: encodeBase64(await image.encodeWEBP(70)) };
}

import { createResource, For, from, Show } from "solid-js";
import { extend } from "@opentui/solid";
import { ScaledImage } from "./scaled.js";

extend({ scaled_image: ScaledImage });

const WIDTH = 96;
const ROWS = 13;

const tag = (id) => (id ? id.slice(-4) : "—");
const latched = (on, text) => ({ c: on ? "inv" : "m", t: on ? ` ${text} ` : text });

const rows = (desk, faculties, pictures) => [
  [["m", "next"], latched(desk.next === "iterate", `on ${tag(desk.shown)}`), latched(desk.next === "fresh", "fresh")],
  [],
  [["m", "model"], ...faculties.flatMap((faculty, index) => (desk.model === faculty.model ? [["inv", ` ${index + 1} ${faculty.model} `]] : [["P b", String(index + 1)], ["i", ` ${faculty.model}`]]))],
  [],
  [["m", "aspect"], ...["1:1", "3:2", "2:3", "16:9"].map((aspect) => latched(desk.aspect === aspect, aspect))],
  [],
  [["m", "count"], ...[1, 2, 4].map((count) => latched(desk.count === count, String(count)))],
  [],
  [["m", "refs"], ...desk.references.flatMap((reference) => [["i", `▣ ${tag(reference.file)}`], ["inv", ` ${reference.role} `]]), ["P b", "+"]],
  [],
  [["bd", `┌${"─".repeat(WIDTH)}┐`]],
  ...Array.from({ length: ROWS }, () => [["bd", "│"], ["sk", " ".repeat(WIDTH)], ["bd", "│"]]),
  [["bd", `└${"─".repeat(WIDTH)}┘`]],
  [],
  pictures.map((picture) => (desk.shown === picture.id ? ["inv", ` ${tag(picture.id)} `] : ["sk", ` ${tag(picture.id)} `])),
];

export function Darkroom(props) {
  const desk = from(props.buffer.$data);
  const files = from(props.terminal.client.runtime.entities.file.$entities);
  const [faculties] = createResource(() => props.buffer.mode.call.picture.faculties({}));
  const pictures = () => files().filter((file) => file.mode.id === props.buffer.mode.id && file.traits.includes("DEVELOPED")).sort((a, b) => a.createdAt.localeCompare(b.createdAt));
  const shown = () => files().find((file) => file.id === desk().shown);
  const pick = (key) => {
    const faculty = (faculties() ?? [])[Number(key) - 1];
    if (faculty) props.buffer.mode.call.picture.desk({ buffer: props.buffer.id, model: faculty.model });
  };
  return (
    <box flexDirection="column" onKeyDown={(event) => /^[1-4]$/.test(event.name) && pick(event.name)}>
      <For each={rows(desk(), faculties() ?? [], pictures())}>
        {(spans) => (
          <text>
            <For each={spans}>{(span) => <span class={span.c}>{span.t}</span>}</For>
          </text>
        )}
      </For>
      <Show when={shown()}>{(file) => <scaled_image src={file().url} position="absolute" top={11} left={1} width={WIDTH} height={ROWS} />}</Show>
    </box>
  );
}
