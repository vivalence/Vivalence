export const pour = (reading, signal) => {
  switch (signal.nature) {
    case "/done":
      return { ...reading, done: [...reading.done, signal.input.block] };
    case "/undo":
      return { ...reading, done: reading.done.slice(0, -1) };
    case "/face":
      return { ...reading, face: signal.input.face };
    case "/set":
      return { ...reading, settings: { ...reading.settings, ...signal.input } };
  }
};
