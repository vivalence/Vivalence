const blocks = () => props.buffer.data.article.blocks;
const mark = (id, done) => blocks().map((block) => (block.id === id ? { ...block, done } : block));
const done = (block) =>
  settle({ article: { ...props.buffer.data.article, blocks: mark(block.id, new Date().toISOString()) } });
const newest = () => blocks().filter((block) => block.done).sort((a, b) => b.done.localeCompare(a.done))[0];
const undo = () => settle({ article: { ...props.buffer.data.article, blocks: mark(newest().id, null) } });
const shown = () => blocks().filter((block) => !block.done);
