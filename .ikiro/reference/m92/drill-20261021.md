// subsystems/typology/gestalten/shape/keymap.js
import * as is from "../is/index.js";
import { Vector } from "../../prototypes/vector.js";
import { strip as stripVector } from "./strip.js";

export const strip = (keymap) =>
  stripVector(keymap, (pattern, effect) => {
    if (is.fn(effect)) throw new Error(`[keymap] ${pattern.nature}: a served keymap carries lines, not functions`);
    return { valence: pattern.valence, line: effect };
  });

export const wire = (node, keymap = new Vector()) => {
  for (const [key, child] of Object.entries(node.branches)) {
    const branch = keymap.branch({ nature: key, valence: child.effect === undefined ? undefined : child.effect.valence });
    if (child.effect !== undefined) branch.effect = child.effect.line;
    wire(child, branch);
  }
  return keymap;
};
#+END_SRC

// subsystems/typology/tests/keymap.test.js
specimen.it("strip ⟷ wire round-trips literal lines; the source is not mutated", () => {
  const source = new Vector().open({ nature: "leader/m", valence: "the mode picker" }, "/mode");
  const stripped = shape.keymap.strip(source);
  specimen.expect(stripped.branches.leader.branches.m.effect).toEqual({ valence: "the mode picker", line: "/mode" });
  specimen.expect(shape.keymap.strip(shape.keymap.wire(stripped))).toEqual(stripped);
  specimen.expect(shape.keymap.strip(source)).toEqual(stripped);
});
specimen.it("a function effect throws on strip", () => {
  specimen.expect(() => shape.keymap.strip(new Vector().open("leader/x", () => null))).toThrow("[keymap] x");
});
