  constructor(terminal) {
    this.terminal = terminal;
  }
  get composed() {
    return new Vector().graft(
      this.terminal.client.repl.commands,
      ...fromm.modes(this.terminal.client.modes).flat.map((mode) => mode.repl.commands),
      this.commands,
      ...this.terminal.client.runtime.entities.mode.$entities.get().map(scoped),
      ...(harnessed(this.terminal) ? [turn] : []),
    );
  }
  dispatch(line) {
    const signal = new ReplSignal(line);
    const replDie = {
      signal,
      client: this.terminal.client,
      runtime: this.terminal.client.runtime,
      terminal: this.terminal,
      controller: this.terminal.controller.branch(signal.pathname),
      commands: this.composed,
      output: undefined,
    };
    return steer.dispatch.execute(execution, replDie).then(() => replDie.output);
  }
}
