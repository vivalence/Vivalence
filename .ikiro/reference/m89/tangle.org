  level: Object.hasOwn(surface, "level") ? surface.level : telemetry.level,
  filters: Object.hasOwn(surface, "filters") ? surface.filters : telemetry.filters,
  ...(Object.hasOwn(surface, "tail") && { tail: surface.tail }),
});

export const admits = ({ level: floor, filters }) => (record) =>
  RANK.indexOf(level(record)) <= RANK.indexOf(floor) && (filters.length === 0 || filters.some((prefix) => record.path.startsWith(prefix)));

const KEEP = {
  faulty: () => trace.faulty,
  slower: (tail) => trace.slower(tail.threshold),
};

export const sieve = (surface, sink) => {
  const admit = admits(surface);
  const write = Object.hasOwn(surface, "tail") ? trace.decant(sink, KEEP[surface.tail.keep](surface.tail)) : trace.flow(sink);
  return (record) => {
    if (admit(record)) write(record);
  };
};

  const admit = admits(narrow(telemetry, telemetry.surfaces.inspector));
  return (record) => {
    if (admit(record)) store.set(trace.trim(trace.step(store.get(), record), telemetry.caps.story));

};

  if (node.children.length) out.children = node.children.map(stable);
  return out;
};

const faulted = (node) => node.fault !== null || node.children.some(faulted);

const wire = (node) => {
  const response = node.entries.find((entry) => entry.verb === "response");
  if (response !== undefined) return response.data;
  const request = node.entries.find((entry) => entry.verb === "request");
  return request === undefined ? null : request.data;
};

const elapsed = (node) => {
  const held = trace.duration(node);
  return held === null ? "" : `${held.toFixed(0)}ms`;
};

const TONES = [
  [faulted, "danger"],
  [(node) => trace.duration(node) > 500, "warning"],
  [() => true, "success"],
];

  <>
