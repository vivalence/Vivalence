#+BEGIN_SRC js
switch (packet.event) {
  case "/turn/open":
    speaking?.abort("interrupted");
    break;
  case "/turn/eager":
    early = answer([...history, soma.pour(structuredClone(heard), { event: "/turn/close" })], { held: true });
    break;
  case "/turn/resume":
    early?.abort("resumed");
    break;
  case "/turn/close":
    history = [...history, heard];
    speaking = early?.matches(heard) ? early.release() : answer(history);
    break;
}
#+END_SRC
