enum ApiProcessState {
  CREATED, STARTING, RUNNING, STOPPING, 
  FINISHED, ERRORED, EXHAUSTED, MIXED
}

class Process {
  private status: ApiProcessState = ApiProcessState.CREATED
  private runner?: Runner | WorkerRunner
  private cronJob?: Cron
  private watcher?: Watcher

  async start(reason?: string, restart?: boolean) {
    // Guard conditions
    if (this.blocked) return  // Blocked processes can't start
    if (this.status === RUNNING && !this.config.overrun) return
    if (this.restarts >= this.config.restartLimit && restart) {
      this.setStatus(ApiProcessState.EXHAUSTED)
      return
    }

    this.setStatus(ApiProcessState.STARTING)
    
    // Create appropriate runner
    this.runner = this.config.worker 
      ? new WorkerRunner(this.pup, this.config)
      : new Runner(this.pup, this.config)
    
    // Start file watcher if configured
    if (this.config.watch) this.setupWatch(this.config.watch)
    
    try {
      // Execute and wait for completion
      const result = await this.runner.run((pid) => {
        this.setStatus(ApiProcessState.RUNNING)
        this.pid = pid
        this.started = new Date()
      })
      
      // Handle exit codes
      if (result.code === 0) {
        this.setStatus(ApiProcessState.FINISHED)
      } else {
        this.setStatus(ApiProcessState.ERRORED)
      }
    } catch (e) {
      this.setStatus(ApiProcessState.ERRORED)
    }
  }

  async stop(reason: string): Promise<boolean> {
    this.setStatus(ApiProcessState.STOPPING)
    const abortTimers = new AbortController()
    
    // Graceful termination sequence
    delay(this.config.terminateGracePeriod * 1000).then(() => {
      this.killRunner(reason, "SIGTERM")  // Try graceful stop first
    })
    
    delay(this.config.terminateTimeout * 1000).then(() => {
      this.killRunner(reason, "SIGKILL")  // Force kill if needed
    })
    
    // Wait for process to actually stop
    return new Promise((resolve) => {
      const onFinish = (ev) => {
        if (ev.status?.pid == this.pid && 
           [FINISHED, EXHAUSTED, ERRORED].includes(this.status)) {
          abortTimers.abort()
          resolve(true)
        }
      }
      this.pup.events.on("process_status_changed", onFinish)
    })
  }
}

private watchdog = async () => {
  try {
    for (const process of this.allProcesses()) {
      const status = process.getStatus()
      const config = process.getConfig()

      // 1. Handle initial autostart
      if (config.autostart && status.status === CREATED) {
        process.start("autostart")
      }

      // 2. Handle pending restarts from file changes
      if (status.status !== STOPPING && process.isPendingRestart()) {
        process.start(process.pendingRestartReason)
      }

      // 3. Restart policy enforcement
      if (status.status === FINISHED || status.status === ERRORED) {
        const msSinceExited = Date.now() - status.exited?.getTime()
        const restartDelay = config.restartDelayMs ?? 
          (config.watch ? 500 : 10000)  // Faster restart for watchers
        
        if (msSinceExited > restartDelay) {
          const restartPolicy = config.restart ?? 
            (config.autostart ? "always" : undefined)
          
          if (restartPolicy === "always" ||
             (restartPolicy === "error" && status.status === ERRORED)) {
            process.start("restart", true)
          }
        }
      }

      // 4. Timeout enforcement  
      if (status.status === RUNNING && config.timeout && status.started) {
        const secondsRunning = (Date.now() - status.started.getTime()) / 1000
        if (secondsRunning > config.timeout) {
          process.stop("timeout")
        }
      }
    }

    // 5. Update application state and refresh plugin tokens
    this.updateApplicationState()
    this.refreshPluginTokens()
    
  } catch (e) {
    this.logger.error("watchdog", "Watchdog error: " + e.message)
  }

  // 6. Reschedule if not terminating
  if (!this.requestTerminate) {
    this.watchdogTimer = setTimeout(() => this.watchdog(), 1000)
  }
}
