#+begin_src javascript
export const controlShard = (manifest, provider) => {
  const vector = new Vector()
    .use(shards.context.attach("manifest", manifest))
    .use(shards.context.attach("provider", provider));
    
  // Lifecycle primitives
  vector
    .open("/start", async (ctx) => {
      const process = await ctx.provider.start(ctx.manifest);
      return { pid: process.pid, status: "running" };
    })
    .open("/stop", async (ctx) => {
      await ctx.provider.stop();
      return { status: "stopped" };
    })
    .open("/restart", async (ctx) => {
      await ctx.provider.stop();
      const process = await ctx.provider.start(ctx.manifest);
      return { pid: process.pid, status: "running" };
    });
    
  return vector;
};

export async function gaiaService(variant) {
  const gaia = new Gaia();
  
  // Load circuits
  for (const circuit of variant.circuits) {
    for (const service of circuit.services) {
      const cake = new ServiceCake(service);
      const die = await gaia.spawn(cake);
      await die.start();
    }
  }
  
  // Install as system service
  if (variant.traits.includes("DAEMON")) {
    await installService({
      name: "gaia",
      cmd: `deno run -A gaia.js`,
      system: true
    });
  }
  
  return gaia;
}
