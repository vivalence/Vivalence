import type { ColorScale } from '../core/color-system';

export interface ThemeGeneratorOptions {
  prefix?: string;
  target?: 'css' | 'js' | 'scss';
}

export class ThemeGenerator {
  static generateColorVariables(colors: Record<string, ColorScale>, options: ThemeGeneratorOptions = {}) {
    const { prefix = '--dv-' } = options;
    
    return Object.entries(colors).reduce((acc, [colorName, scale]) => {
      Object.entries(scale).forEach(([shade, value]) => {
        acc[`${prefix}${colorName}-${shade}`] = value;
      });
      return acc;
    }, {} as Record<string, string>);
  }

  static createThemeClass(name: string, variables: Record<string, string>) {
    return `
      .${name} {
        ${Object.entries(variables)
          .map(([key, value]) => `${key}: ${value};`)
          .join('\n')}
      }
    `;
  }
}

    this._element?.remove();
  }

export interface ComponentMetadata {
  id: string;
  type: string;
  renderer: 'always' | 'onlyWhenVisible';
  params?: Record<string, unknown>;
}

export class ComponentRegistry {
  private _components = new Map<string, ComponentMetadata>();
  private _factories = new Map<string, (...args: any[]) => any>();

  register<T extends object>(
    type: string, 
    factory: (...args: any[]) => T
  ) {
    this._factories.set(type, factory);
  }

  create(metadata: ComponentMetadata) {
    const factory = this._factories.get(metadata.type);
    if (!factory) {
      throw new Error(`No factory registered for component type: ${metadata.type}`);
    }

    const instance = factory(metadata);
    this._components.set(metadata.id, metadata);
    
    return instance;
  }

  getMetadata(id: string) {
    return this._components.get(id);
  }
}

import { EventEmitter } from '../events/event-emitter';

export interface LayoutNode {
  id: string;
  type: 'panel' | 'group' | 'grid';
  children?: LayoutNode[];
  params?: Record<string, unknown>;
}

export class LayoutStore {
  private _layout: LayoutNode;
  private _eventEmitter = new EventEmitter();

  constructor(initialLayout?: LayoutNode) {
    this._layout = initialLayout ?? { id: 'root', type: 'grid', children: [] };
  }

  getNode(id: string): LayoutNode | undefined {
    return this.findNode(this._layout, id);
  }

  private findNode(node: LayoutNode, id: string): LayoutNode | undefined {
    if (node.id === id) return node;
    return node.children?.find(child => this.findNode(child, id));
  }

  updateNode(id: string, updates: Partial<LayoutNode>) {
    const node = this.getNode(id);
    if (node) {
      Object.assign(node, updates);
      this._eventEmitter.emit('layoutChanged', { node });
    }
  }

  serialize(): string {
    return JSON.stringify(this._layout);
  }

  deserialize(data: string) {
    this._layout = JSON.parse(data);
    this._eventEmitter.emit('layoutLoaded', { layout: this._layout });
  }
}

      return null;
    }
  }
}

type EventListener = (...args: any[]) => void;

export class EventEmitter {
  private _listeners = new Map<string, Set<EventListener>>();

  on(event: string, listener: EventListener): () => void {
    const listeners = this._listeners.get(event) ?? new Set();
    listeners.add(listener);
    this._listeners.set(event, listeners);

    return () => this.off(event, listener);
  }

  off(event: string, listener: EventListener) {
    this._listeners.get(event)?.delete(listener);
  }

  emit(event: string, data?: any) {
    this._listeners.get(event)?.forEach(listener => listener(data));
  }
}
