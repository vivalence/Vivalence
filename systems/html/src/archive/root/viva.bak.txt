import { Stall, Buffer } from "@vivalence/drapes";
import { dataspace } from "$client";

  const stall = new Stall();
  const signal = new Signal(event.url.pathname);
  // console.log("1");
  // console.log("1", signal);
  // console.log(event.url.pathname, event.url);
  // console.log("2");

  const [take, apply, match] = controller.traverse(perspective, signal);

  // console.log("3", match);

  if (!take) throw new NotFound(signal);

  stall.withPull(async () => {
    // console.log("8");
    const context = new Context({
      stall,
      signal,
      match,
      params: fromm.match(match).parameters,
    });
    // console.log("parmas", context.params);

    await apply(context, async (ctx) => (ctx.take = await take(ctx)));
    // console.log("4", context);

    return context.take || [];
  });

  return { stall };
};
