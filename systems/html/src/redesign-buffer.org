    this.daemon = daemon;

      "/userspace/entities/state/find",
      { where: { session: session.id }, orderBy: { position: "asc" } },
    );

    this.$entries.set(states);
    this.$cursor.set(session.cursor ?? 0);
  }

  withPull(pull) {
    this.handlers.pull = pull;
    return this;
  }

  async append(entryState) {
    const entry = await this.daemon.connection.call(
      "/userspace/entities/state/append",
      { session: this.session.id, ...entryState },
    );
    this.$entries.set([...this.$entries.get(), entry]);
    return entry;
  }

    const status = this.$status.get();
    if (["STOP", "NEXT"].includes(status)) return;
    this.$status.set("NEXT");

    const modals = this.$modals.get();
    const cursor = this.$cursor.get();

    if (cursor < modals.length - 1) {
      this.$cursor.set(cursor + 1);
      // persist cursor
      this.daemon.connection.call("/userspace/entities/session/update", {
        id: this.session.id,
        cursor: cursor + 1,
      });
    }

    this.$status.set("IDLE");
    this.pull();
  }

  async pull() {
    const status = this.$status.get();
    if (["STOP", "PULLING"].includes(status)) return;
    if (!this.handlers.pull) return;

    const modals = this.$modals.get();
    const cursor = this.$cursor.get();
    const remaining = modals.length - cursor - 1;
    if (remaining > 2) return; // threshold

    this.$status.set("PULLING");

    try {
      const newEntries = await this.handlers.pull(this);
      // entries already appended server-side and via this.append
      this.$status.set("IDLE");
    } catch (error) {
      this.$error.set(error);
      this.$status.set("STOP");
      console.error("[stall pull error]", error);
    }
  }

  get entries() {
    return this.$entries.get();
  }
}

// routes/[...viva]/boot.js
import { Signal, fromm, Blacklist } from "@vivalence/typology";
import { controller, Context } from "@vivalence/vector";
import { Buffer, Stall } from "@vivalence/html/typology";
import { perspective } from "./perspective.js";
import { dataspace } from "$client";
import { replaceState } from "$app/navigation";
import { get } from "svelte/store";
import { page } from "$app/stores";

export async function boot(pathname) {
  const signal = new Signal(pathname);
  const [take, apply, match] = controller.traverse(perspective, signal);
  if (!take) return null;

  const params = fromm.match(match).parameters;
  const buffer = new Buffer(signal);

  // 1. resolve daemon + mode
  buffer.daemon = await dataspace.daemon
    .findOne((d) => d.manifest.slug === params.daemon);

  buffer.mode = await dataspace.mode.findOne((m) =>
    m.daemon.manifest.slug === params.daemon &&
    m.type === params.type &&
    m.slug === params.mode,
  );

  // 2. resolve intent
  const url = new URL(get(page).url);
  const intentId = url.searchParams.get("intent");

  if (intentId) {
    buffer.intent = await buffer.daemon.connection.call(
      "/userspace/entities/intent/findOne",
      { where: { id: intentId } },
    );
  } else {
    buffer.intent = await buffer.daemon.connection.call(
      "/userspace/entities/intent/create",
    );
    url.searchParams.set("intent", buffer.intent.id);
    replaceState(url, {});
  }

  // 3. resolve session (find active or create)
  let session = await buffer.daemon.connection.call(
    "/userspace/entities/session/findOrCreate",
    { intent: buffer.intent.id, user: buffer.intent.user },
  );
  buffer.session = session;
  buffer.phase = session.phase;

  // 4. boot stall from session
  await buffer.stall.boot(session, buffer.daemon);

  // 5. resolve valence (if in path)
  if (params.valence) {
    buffer.valence = await dataspace.valence.findOne(
      (v) => v.slug === params.valence && buffer.mode.id === v.mode.id,
    );
  }

  // 6. wire traits
  if (buffer.mode.implements("LANGUAGED")) {
    buffer.converse = async (message) => {
      const response = await buffer.mode.connection.call("/converse", {
        message,
        session: session.id,
      });

      for (const entry of response.entries) {
        buffer.stall.$entries.set([...buffer.stall.$entries.get(), entry]);
        buffer.processEntry(entry);
      }

      return response;
    };
  }

  if (buffer.mode.implements("PRODUCTIVE") && buffer.valence) {
    buffer.stall.withPull(async (stall) => {
      const products = stall.$modals.get().map((e) => e.data?.PRODUCT?.id).filter(Boolean);
      const blacklist = { products };

      await buffer.mode.connection.call("/produce", {
        session: session.id,
        valence: buffer.valence.id,
        scope: { intent: buffer.intent.id },
        blacklist,
      });

      // server created State entries — reload
      const states = await buffer.daemon.connection.call(
        "/userspace/entities/state/find",
        {
          where: { session: session.id },
          orderBy: { position: "asc" },
        },
      );
      stall.$entries.set(states);
    });
  }

  // initial pull
  await buffer.stall.pull();

  return buffer;
}

// daemon/mode/traits/languaged.js
import { Agent } from "@vivalence/typology";
import { Type } from "@sinclair/typebox";

export const LANGUAGED = async (mode, daemon) => {
  mode.brain = mode.brain || daemon.hallucinator;

  // collect tools from other traits
  if (!mode.tools) mode.tools = new (await import("@vivalence/vector")).Vector();

  mode.aperture.open("/converse", async (input, ctx) => {
    const session = await ctx.daemon.entities.session.findOne(input.session);
    if (!session) throw new Error("session not found");

    // 1. persist user message as State
    const userState = ctx.daemon.entities.state.create({
      session: session.id,
      position: session.counter++,
      type: "message",
      traits: ["USER"],
      data: { USER: { content: input.message } },
    });
    await ctx.daemon.entities.em.flush();

    // 2. load recent entries for LLM context
    const history = await ctx.daemon.entities.state.find(
      { session: session.id, type: "message" },
      { orderBy: { position: "asc" }, limit: 50 },
    );

    const messages = history.map((state) => {
      const isUser = state.traits.includes("USER");
      return {
        role: isUser ? "user" : "assistant",
        content: state.data[isUser ? "USER" : "AGENT"]?.content || "",
      };
    });

    // 3. build agent with mode-registered tools
    const agent = new Agent(mode.manifest.slug)
      .withBrain(mode.brain)
      .withTools(mode.tools)
      .withInput(Type.Object({ message: Type.String() }))
      .withContext("identity", mode.systemPrompt || =You are ${mode.manifest.name}.=)
      .withContext("environment", =${JSON.stringify(ctx.daemon.docs || {})}=)
      .withHistory(messages)
      .withTemplate((input) => input.message);

    // 4. execute
    const result = await agent.do({ message: input.message });

    // 5. persist agent response as State
    const newEntries = [userState];

    const agentState = ctx.daemon.entities.state.create({
      session: session.id,
      position: session.counter++,
      type: "message",
      traits: ["AGENT"],
      data: { AGENT: { content: result.text } },
    });
    newEntries.push(agentState);

    // 6. process tool results — other traits may have pushed states
    if (result.toolStates) {
      for (const ts of result.toolStates) {
        ts.session = session.id;
        ts.position = session.counter++;
        const stateEntity = ctx.daemon.entities.state.create(ts);
        newEntries.push(stateEntity);
      }
    }

    await ctx.daemon.entities.em.flush();

    return { entries: newEntries };
  });
};
