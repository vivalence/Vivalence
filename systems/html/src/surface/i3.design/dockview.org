// DockviewContainer.svelte
<script lang="ts">
  import { onMount, onDestroy } from 'svelte';
  import { 
    DockviewComponent,
    DockviewApi,
    IDockviewPanelProps,
    IContentProps,
    DefaultPanel,
    PanelCollection
  } from 'dockview';
  import 'dockview/dist/styles/dockview.css';

  // Define your panel components
  const ExplorerPanel = {
    component: DefaultPanel,
    props: {
      title: 'Explorer',
      content: '<div class="p-4">File Explorer Content</div>'
    }
  };

  const EditorPanel = {
    component: DefaultPanel,
    props: {
      title: 'Editor',
      content: '<div class="p-4">Editor Content</div>'
    }
  };

  const TerminalPanel = {
    component: DefaultPanel,
    props: {
      title: 'Terminal',
      content: '<div class="p-4">Terminal Content</div>'
    }
  };

  let dockviewApi: DockviewApi | null = null;
  let container: HTMLElement;

  // Component registry
  const components = {
    default: (props: IDockviewPanelProps) => {
      const component = new DefaultPanel(props);
      return component;
    }
  };

  // Initial layout configuration
  const layout = {
    type: 'dockview',
    content: [
      {
        type: 'branch',
        size: 200,
        data: [
          {
            id: 'explorer',
            component: 'default',
            props: ExplorerPanel.props
          }
        ]
      },
      {
        type: 'branch',
        size: 600,
        data: [
          {
            id: 'editor',
            component: 'default',
            props: EditorPanel.props
          },
          {
            id: 'terminal',
            component: 'default',
            props: TerminalPanel.props,
            position: 'bottom'
          }
        ]
      }
    ]
  };

  onMount(() => {
    // Initialize Dockview
    const dockview = new DockviewComponent({
      components,
      container,
      layout
    });

    dockviewApi = dockview.api;

    // Add event listeners
    dockview.onDidLayoutChange((event) => {
      console.log('Layout changed:', event);
    });

    dockview.onDidActiveChange((event) => {
      console.log('Active panel changed:', event);
    });
  });

  onDestroy(() => {
    dockviewApi?.dispose();
  });

  // Methods to control the layout programmatically
  function addNewPanel() {
    if (!dockviewApi) return;
    
    const id = `panel-${Math.random().toString(36).substr(2, 9)}`;
    dockviewApi.addPanel({
      id,
      component: 'default',
      props: {
        title: `New Panel ${id}`,
        content: `<div class="p-4">Content for ${id}</div>`
      }
    });
  }

  function maximizePanel(id: string) {
    const panel = dockviewApi?.getPanel(id);
    if (panel) {
      panel.setMaximized(true);
    }
  }

  function saveLayout() {
    if (!dockviewApi) return;
    const currentLayout = dockviewApi.toJSON();
    localStorage.setItem('dockview-layout', JSON.stringify(currentLayout));
  }

  function loadLayout() {
    if (!dockviewApi) return;
    const savedLayout = localStorage.getItem('dockview-layout');
    if (savedLayout) {
      dockviewApi.fromJSON(JSON.parse(savedLayout));
    }
  }
</script>

<div class="w-full h-full flex flex-col">
  <div class="flex gap-2 p-2 bg-gray-100 border-b">
    <button 
      class="px-3 py-1 bg-blue-500 text-white rounded hover:bg-blue-600"
      on:click={addNewPanel}
    >
      Add Panel
    </button>
    <button 
      class="px-3 py-1 bg-green-500 text-white rounded hover:bg-green-600"
      on:click={saveLayout}
    >
      Save Layout
    </button>
    <button 
      class="px-3 py-1 bg-yellow-500 text-white rounded hover:bg-yellow-600"
      on:click={loadLayout}
    >
      Load Layout
    </button>
  </div>
  
  <div 
    bind:this={container} 
    class="flex-1"
    style="min-height: 0;" /* Important for proper flexbox behavior */
  />
</div>
