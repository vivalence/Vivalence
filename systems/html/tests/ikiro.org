// @vivalence/html/typology/repository.js
import { atom, computed } from "nanostores";

export class Repository {
  $items = atom([]);
  $has = computed(this.$items, (items) => items.length > 0);

  constructor(schema, lifecycle) {
    this.schema = schema;      // typebox schema
    this.lifecycle = lifecycle; // async (entity) => entity
  }

  async spawn(seed) {
    const entity = this.schema.create ? this.schema.create(seed) : seed;
    if (this.lifecycle) await this.lifecycle(entity);
    this.$items.set([...this.$items.get(), entity]);
    return entity;
  }

  async findOne(predicate) {
    return this.$items.get().find(predicate) || null;
  }

  toJSON() {
    return this.$items.get();
  }
}

// @vivalence/html/typology/lighthouse.js
import { map, computed } from "nanostores";

export class Lighthouse {
  $session = map({ authority: null, identity: null, state: "ANONYMOUS" });
  
  isIdentified = computed(this.$session, (s) => s.state === "IDENTIFIED");

  constructor(connection) {
    this.connection = connection;
  }

  async login(username, password) {
    const result = await this.connection.call("/auth/login", { username, password });
    if (result.authority) {
      this.$session.set({
        authority: result.authority,
        identity: result.identity,
        state: "IDENTIFIED",
      });
    }
    return result;
  }

  async refresh() {
    const { refresh } = this.$session.get().authority || {};
    if (!refresh) return { success: false };
    const result = await this.connection.call("/auth/refresh", { refresh });
    if (result.access) {
      this.$session.setKey("authority", { ...this.$session.get().authority, access: result.access });
    }
    return result;
  }

  logout() {
    this.$session.set({ authority: null, identity: null, state: "ANONYMOUS" });
  }
}
